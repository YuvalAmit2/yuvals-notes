\section{Motivation and Review}

\begin{notebox}[Motivation]
    Distinguish topological objects, and ultimately classify them.
\end{notebox}

Let's see some examples where the power of algebraic topology helps us prove important topological properties.

\begin{thm}
    The unit ball $\bar{\mb D}^{n+1}$ is not homeomorphic to $S^n$.
\end{thm}

\begin{proof}
    The $n^{\text{th}}$ reduced homology group $\tilde H_n$ distinguishes them. Namely, $\tilde H_n\paren{\bar{\mb D}^{n+1}} = \{0\}$, while $\tilde H_n(S^n) = \Z$.
\end{proof}

\begin{thm}[][label=thm:boundaryfixedthm]
    There does not exist a continuous map $f:\bar{\mb D}^{n+1} \to S^n$ such that $f(x)=x$ for every $x \in S^n$ (note that $S^n$ is the boundary of $\bar{\mb D}^{n+1}$).
\end{thm}

\begin{proof}
    Suppose such an $f$ exists. Consider the inclusion map $i: S^n \to \bar \D^{n+1}$. Then, the map $f \circ i : S^n \to S^n $ is the identity map. Moreover, $i$ and $f$ induce maps on the homology groups 
    $$\tilde H_n(S^n) \arr[\tilde H_n(i)] \tilde H_n(\bar\D^{n+1}) \arr[\tilde H_n(f)] \tilde H_n(S^n).$$
    By functoriality,
    $$\tilde H_n(f) \circ \tilde H_n(i) = \tilde H_n(f \circ i) = \tilde H_n(\id) = \id.$$
    Hence, the composition
    \begin{align*}
        \Z \to0 \to \Z
    \end{align*}
    would be the identity map $\Z \to \Z$, which is impossible.
\end{proof}

\begin{thm}[Brouwer's Fixed Point Theorem]
    Every continuous function $f:\bar{\D}^n \to \bar{\D}^n$ has a point $x$ such that $f(x) = x$.
\end{thm}

\begin{proof}
    Suppose $f$ does not have a fixed point. Then, for every $x \in \bar{\D}^n$, we may consider the unique straight line from $f(x)$ to $x$. Let $g(x)$ be the point where the line $f(x) \to x$ hits $S^{n-1}$. Observe that $g(x) = x$ for every $x \in S^{n-1}$, and $g$ is continuous. Thus, we have a contradiction by \Cref{thm:boundaryfixedthm} (applied with $n-1$ in place of $n$).
\end{proof}

\subsection{Topology Review}

\begin{defn}
    In $\R^n$, we have the usual metric $d(x,y)$. We denote $B(x,r)$ to be the open ball around $x$ with radius $r$.
\end{defn}

\begin{defn}
    A topological space $X$ is called \term{compact} if every open cover has a finite subcover.
\end{defn}

\begin{thm}
   A subset $X \subset \R^n$ is compact if and only if $X$ is closed and bounded. 
\end{thm}

\begin{proof}
    Suppose $X$ is compact. The set $\{B(0,n)\}_{n=1}^\infty$ of open balls is necessarily an open cover of $X$, hence has a finite subcover, thus $X$ is bounded. Moreover, if $\R^n-X$ is not open, let $p \in \R^n\setminus X$ be such that $B(p,\veps) \cap X \neq \emptyset$ for every $\veps>0$. Then, take the cover of $X$ by 
    $$\brax{B(x,\frac{d(p,x)}{2})}_{x \in X}.$$
    One sees that this does not have a finite subcover, a contradiction.

    For the other direction, note that closed+bounded implies sequential compactness. Let $\{U_\alpha\}_{\alpha \in I}$ be an open cover of $X$. 
\begin{claim}
    There exists $\delta>0$ such that for every $x \in X$, $B(x,\delta) \subset U_\alpha$ for some $\alpha \in I$.
\end{claim}

\begin{proof}
    Contrarily, if this were not the case, there exists a sequence $\{x_i\} \subset X$ such that $B(x_i,1/i) \not \subset U_\alpha$ for all $\alpha \in I$. Then, $\{x_i\}$ has a subsequence converging to some $p$. This $p$ must be contained in some $U$ in the open cover, and it follows that there exists $N$ with $B(x_N,1/N) \subset U$, a contradiction.
\end{proof}

Since $X$ is bounded in $\R^n$, for every $\delta>0$ the number of disjoint balls of radius $\delta/2$ around points in $X$ is bounded. Let $m$ be this upper bound. Take a point $p_1 \in X$. The ball $B(p_1,\delta)$ is contained in some cover element $U_1$. If $U_1$ covers $X$, we're done, but otherwise repeat with some $p_2 \in X \setminus U_1$ and $U_2$. Continue to repeat until we have $p_{m+1}$ and $U_{m+1}$. We claim that $U_1,\dots,U_{m+1}$ must cover $X$: indeed, if it doesn't, then the balls $\{B(p_i,\delta/2)\}_{i=1}^{m+1}$ must all be disjoint, and this is a contradiction.
\end{proof}

\begin{defn}
    A topological space $X$ is called \term{connected} if $X = U_1 \sqcup U_2$ with $U_1,U_2$ open implies $U_1 = \emptyset$ or $U_2 =\emptyset$.
\end{defn}

\begin{defn}
    $X$ is called \term{path connected} if for every $p,q \in X$, there exists a continuous map $\delta:[0,1] \to X$ such that $\delta(0)=p$ and $\delta(1) = q$.
\end{defn}

\begin{prop}
    Path connected $\implies$ connected.
\end{prop}

\begin{defn}
    A topological space $X$ is called \term{Hausdorff} if for every distinct $p,q \in X$, there exist open sets $U \ni p$ and $V \ni q$ such that $U \cap V = \emptyset$.
\end{defn}

\begin{thm}
    Let $f: X \to Y$ be bijective and continuous, with $X$ compact, $Y$ Hausdorff. Then, $f$ is a homeomorphism.
\end{thm}

\begin{proof}
    It suffices to show $f$ is a closed map. Let $C \subset X$ be closed. Since $X$ is compact, one can see that $C$ is compact (theorem). Therefore, $f(C)$ is compact (theorem), hence closed (theorem).
\end{proof}

\begin{prop}
    If $X$ is Hausdorff and $A \subset X$ is compact, then $A$ is closed.
\end{prop}

\begin{exercise}
    Give an example of a space $X$ such that $A \subset X$ is compact but not closed. This space cannot be Hausdorff.
\end{exercise}

\begin{defn}
    Let $X$ be a topological space, and $\sim$ be an equivalence relation on $X$. The \term{quotient-topology} is defined as follows: let $\pi: X \to X/\!\!\sim$ be the projection map. Then, $U \subset X/\!\!\sim$ is open whenever $\pi^{-1}(U)$ is open.
\end{defn}

\begin{prop}
    A map $f:(X/\!\!\sim) \to Y$ is continuous if and only if $f \circ \pi$ is continuous.
\end{prop}

\begin{ex}
    Let $\R\P^{n}$ be the space of lines through the origin in $\R^{n+1}$: this is the quotient space of $\R^{n+1}\setminus \{0\}$ by the equivalence $x \sim y \iff$ there exists $\lambda \in \R$ with $\lambda x=y$. This is homeomorphic to the quotient space of $S^n$ by the relation $x \sim -x$.
\end{ex}

\begin{defn}
    Let $(X_\alpha,\mc U_\alpha)_{\alpha \in I}$ be topological spaces. The \term{product topology} on $\prod_{I}X_\alpha$ has a basis of sets of the form
    $$\prod_{\alpha \in I}U_\alpha,  \quad U_\alpha \in \mc U_\alpha,$$
    and $U_\alpha =X_\alpha$ for all but finitely many $\alpha \in I$.
\end{defn}

\begin{thm}[Tychonoff's Theorem][label=thm:tychonoff]
    The product of compact spaces is compact.
\end{thm}

\section{CW Complexes}

\begin{defn}
    A \term{CW complex} is a topological space $X$, together with a sequence of subspaces $X^0 \subset X^1 \subset \cdots $ such that
    \begin{enumerate}
        \item $X = \bigcup X^n$
        \item $U\subset X$ is open if and only if $U \cap X^n$ is open for all $n \geq 0$
        \item $X^0$ is a discrete set of points
        \item For every $n \geq 1$, there exists an indexing set $I(n)$, disks $\{D^n_\alpha\}_{\alpha \in I(n)}$, and continuous maps
        $\varphi^n_\alpha : \partial D^n_\alpha 
         \to X^{n-1}$ such that 
         $$X^n = \bigsqcup_{\alpha \in I(n)}\bar D_\alpha^n\sqcup X^{n-1}  /\sim$$
         where $x \sim \varphi_\alpha^n(x)$ for all $\alpha \in I(n)$, $x \in \partial D_\alpha^n$.
    \end{enumerate}
    We call $X^n$ the $n^{\text{th}}$ \term{skeleton} of $X$, and $\varphi_\alpha^n$ is called an \term{attaching map}. For a CW complex $X$, the sets $\{X^n\}$ and $\{ \varphi_\alpha^n\}$ are called a \term{CW structure} on $X$. The quotient maps $\Phi_\alpha^n:\bar D_\alpha^n \to X^n$ are called the \term{characteristic maps}. If $X^{n-1} \subsetneq X^n$, but $X^i=X^j$ for all $i, j\geq n$, we say that $X$ is \term[]{$n$-dimensional}.
\end{defn}


\begin{ex}
    A discrete set of points is a $0$-dimensional CW complex.
\end{ex}

\begin{ex}
    A graph is a $1$-dimensional CW complex.

\begin{ex}
    $\R^n$ is a CW complex. For example, $\R^2$ has the points of $\Z^2$ as $0$-cells, the edges of the $\Z^2$-grid as $1$-cells, and a $2$-cell attached along each square of the grid.

\begin{ex}
    $S^n$ is a CW complex. Namely, $X^0$ is a single point, $X^1=\cdots = X^{n-1}=X^0$, and $X^n$ is obtained by attaching a single $n$-cell via the (only possible) attaching map $\varphi_1^n : \partial D_1^n \to X^{n-1}$.
\end{ex}

\begin{ex}
    $\R\P^n=S^n/(x \sim -x)$ is a CW complex. When $n=0$, $\R\P^n$ is just a point, hence has a CW structure. Suppose $\R\P^{n-1}$ has a CW structure. Let $\bar D_\alpha^n$ be the upper hemisphere. We obtain the CW structure on $\R \P^n$ by attaching one $n$-cell to $\R\P^{n-1}$ via the attaching map 
    $$\varphi_\alpha^n(x_1,\dots,x_n,0) = [(x_1,\dots,x_n)] \in \R\P^{n-1}.$$
\end{ex}

\begin{lem}
    Let $X$ be a CW-complex. Then, $U \subset X$ is open $\iff$ for every $n$ and $\alpha \in I(n)$, $(\Phi^n_\alpha)^{-1}(U)$ is open.
\end{lem}

\begin{proof}
    Recall that $U$ is open in $X$ if and only if $U \cap X^n$ is open for all $n$, if and only if for the projection map
    $$P^n: X^{n-1} \sqcup \bigsqcup _{\alpha \in I(n)}D_\alpha^n \to X^n,$$
    we have
    $$(P^n)^{-1}(U \cap X^n)$$
    is open. This is the case if and only if for each $\alpha \in I(n)$,
    $$(\Phi_\alpha^n)^{-1}(U \cap X^n)$$
    and $U \cap X^{n-1}$ are open. So, $(\implies)$ is verified, and $(\impliedby)$ follows by induction on $n$.
\end{proof}

\begin{lem}
    Let $K\subset X$ be compact. Then, $K$ intersects only finitely many cells in $X$. 
\end{lem}

\begin{proof}
    Suppose not. Then, we can choose an infinite set $S \subset K$ such that no two points in $S$ belong to the same cell. We claim $S$ is closed. Indeed, note that $S \cap X^0$ is closed since $X^0$ is discrete. By induction, suppose $S \cap X^{n-1}$ is closed. Then, $S \cap X^n$ is closed $\iff$ $(\Phi_\alpha^n)^{-1}(S)$ is closed, and by induction $(\Phi_\alpha^n)^{-1}(S) \cap \partial D_\alpha^n$ is closed, so indeed $(\Phi_\alpha^n)^{-1}(S)$ is closed as the union of closed sets. Thus, $S \cap X^n$ is closed for every $n$, hence $S$ is closed. 
    
    In fact, we've shown that every subset of $S$ is closed, so $S$ is discrete. Moreover, $S$ is compact as a closed subset of a compact set. Thus, $S$ is compact and discrete, hence finite.
\end{proof}

\begin{defn}
    Let $A \subset X$ be a set, and fix $\veps :=\{\veps_\alpha^n \in \R^+\}$ for every $n$ and $\alpha \in I(n)$. Then, an \term{$\veps$-neighborhood} of $A$ is the open set $N_\veps(A) \subset X$ defined as follows: let
    \begin{align*}
        N_\veps^0(A) := A \cap X^0
    \end{align*}
    and then for every $\alpha \in I(n)$, set
    $$(\Phi_\alpha^n)^{-1}(N_\veps^n(A)) := \braks{B(A \cap \Int(D_\alpha^n),\veps_\alpha^n) \cap \Int(D_\alpha ^n)} \cup \braks{ (1-\veps _\alpha^n,1](\Phi_\alpha^n)^{-1}\paren{N_\veps^{n-1}(A)}},$$
    and
    $$N_\veps(A) = \bigcup N_\veps^n(A).$$
\end{defn}

\begin{prop}
    $$N_\veps(A) \cap X^n = N_\veps^n(A)$$
\end{prop}

\begin{thm}
    Let $X$ be a CW complex. Then, $X$ is Hausdorff.
\end{thm}

\begin{proof}
    Let $p \neq q \in X$, and suppose that $p,q \in X^n$ but $q \not \in X^{n-1}$. So, $q$ is in some open cell $e_\alpha^n$. If $p \in X^{n-1}$, we may choose $\veps$ sufficiently small so that $N_\veps(X^{n-1})$ and $N_\veps(q)$ are disjoint. If $p \not \in X^{n-1}$, then $p,q$ are distinct points in $e_\alpha^n$ which is Hausdorff, hence separated by $\veps$-neighborhoods. These $\veps$-neighborhoods lift to the global structure of $X$ so that $N_\veps(p) \cap N_\veps(q) = \emptyset$.
\end{proof}

\begin{defn}
    Let $X,Y$ be topological spaces. Two (continuous) maps $f,g:X \to Y$ are \term{homotopic} if there exists $H:X \times [0,1] \to Y$ continuous such that $H(x,0)=f(x)$ and $H(x,1)=g(x)$ for every $x \in X$.

    For a subset $A \subset X$, two maps $f,g:X \to Y$ are called \term{homotopic relative to} $\bm A$ if $f|_A=g|_A$, and there exists a homotopy $H:X \times [0,1]\to Y$ between $f,g$ such that $H(x,t)=f(x)=g(x)$ for all $x \in A$, $t \in [0,1]$.
\end{defn}
Homotopy is denoted $f \sim g$ or $f \overset{H}{\sim} g$.

\begin{defn}
    Two maps $f:X \to Y$ and $g:Y \to X$ are called \term[homotopy]{homotopy equivalent} if $g \circ f \sim \id_X$ and $f \circ g \sim \id_Y$. In this case, $X$ and $Y$ are said to be \termnt{homotopy equivalent}.

    A subset $A \subset X$ is called a \term{retract} of $X$ if there exists $r:X \to A$ continuous such that $r(x)=x$ for every $x \in A$. Moreover, $A$ is called a \term{deformation retract} if $i \circ r \sim \id_X$ relative to $A$, where $i:A \hookrightarrow X$ is the inclusion. $X$ is called \term{contractible} if $\id_X \sim $ constant.
\end{defn}

\end{exercise}

\begin{exercise}

\end{exercise}

\begin{defn}
    $X$ is called \term{locally contractible} if for every $p \in X$ and $U \ni p$ open, there exists an open $V \subset U$ with $p \in V$ such that $V$ is contractible.
\end{defn}

\chapter{The Fundamental Group}

\section{Fundamental Group}

\begin{defn}
    Let $X$ be a topological space and $p \in X$. The \term{fundamental group} of $X$ at $p$ is
    $$\pi_1(X,p):=\brax{ \gamma:[0,1]\to X \mid \gamma(0)=\gamma(1)=p } / \braks{ \gamma \overset{\text{rel}(\{0,1\})}{\sim}\mu }.$$
    Equivalently, let $q$ be a fixed point in $S^1$, and we have
    $$\pi_1(X,p) = \brax{\gamma :S^1 \to X \mid \gamma(q)=p}/\braks{\gamma \overset{\text{rel(q)}}{\sim} \mu}$$
    This definition is in some ways better, since changing $S^1$ to $S^n$ gives a more generalized definition of the \term[homotopy group]{$n$-th homotopy group} $\pi_n(X,p)$.
\end{defn}

\begin{defn}
    A path connected space $X$ is called \term{simply connected} if there exists $p \in X$ such that $\pi_1(X,p)=\{ [p]\}$.
\end{defn}

\begin{prop}
    If $X$ deformation retracts to $p \in X$, then
    $$\pi_1(X,p) =\{[p]\}.$$
\end{prop}

\begin{proof}
    Let $\gamma \in \pi_1(X,p)$. Let $H$ be a homotopy between $\id_X$ and $p$ relative to $\{p\}$. Take
    $$\widetilde{H}(s,t) := H(\gamma(s),t).$$
    Then, $\gamma \overset{\widetilde H}{\sim}p$.
\end{proof}

\begin{defn}
    For $\gamma,\mu \in \pi_1(X,p)$, define
    $$\gamma \star \mu(t) = \case{\mu(2t) & 0 \leq t \leq 1/2 \\ \gamma (2t-1) & 1/2 \leq t \leq 1}.$$
    Then, $\star$ induces a well-defined group structure on $\pi_1(X,p)$.
\end{defn}

Indeed, in order to show that this concatenation of curves defines a group structure on $\pi_1(X,p)$, one must prove the following claim:

\begin{claim}
    If $[\gamma] = [\gamma']$ and $[\mu] = [\mu ']$, then $[\gamma \star \mu] = [\gamma ' \star \mu ']$. Thus, $[\gamma] \star [\mu]$ is a well-defined binary operation on $\pi_1(X,p)$. Moreover, $\pi_1(X,p)$ is a group, with identity the constant map $[p]$, and $[\gamma(t)]^{-1}=[\gamma(1-t)]$.
\end{claim}

\begin{exercise}
    Show that if $\varphi :[0,1] \to [0,1]$ is a continuous map with $\varphi(0)=0$ and $\varphi(1)=1$, then $\gamma \circ \varphi \sim \gamma$ relative to $\{0,1\}$.
\end{exercise}

\begin{thm}[][label=thm:pathconnectedisom]
    Let $X$ be path connected, $p,q\in X$. Then, $\pi_1(X,p) \cong \pi_1(X,q)$.
\end{thm}

\begin{proof}
Since $X$ is path connected, we have a continuous curve $\mu:[0,1] \to X$ with $\mu(0)=p$ and $\mu(1)=q$.
    \begin{claim}
        Let $\Omega_p$ denote the space of curves starting and ending at $p$. Then, the map
    $$T_\mu :\Omega_p \to \Omega_q$$
    defined by $T_\mu(\gamma) := \mu \star \gamma \star \mu^{-1}$
    induces an isomorphism $\pi_1 (X,p) \arr[\sim] \pi_1(X,q)$.
    \end{claim}
    Indeed, suppose $[\gamma] = [\gamma ']$. Then, $[T_\mu (\gamma)] = [T_\mu(\gamma')]$. Moreover,
    $$\mu \star (\gamma _1 \star \gamma_2)\star \mu^{-1}\sim\mu \star \gamma _1 \star \mu^{-1}\star \mu \star \gamma_2 \star \mu^{-1},$$
    hence $T_\mu$ is a homomorphism. Moreover, the map $\psi =\mu(1-t)$ induces a clear inverse homomorphism $T_{\psi}:\pi_1(X,q) \to \pi_1(X,p)$, so it follows that $T_\mu$ is an isomorphism.
\end{proof}

\begin{prop}
    Let $(X,p)$ and $(Y,q)$ be (pointed) topological spaces, and $f: (X,p) \to (Y,q)$ be continuous (i.e. a continuous map $f:X \to Y$ with $f(p)=q$). Then, the map $f_*:=\pi_1(f):\pi_1(X,p) \to \pi_1(Y,q)$ defined by
    $$f_*([\gamma]) = [f \circ \gamma]$$
    is a well-defined group homomorphism. Moreover, if
    $$(X,p) \arr[f] (Y,q) \arr[g] (Z,r)$$
    is a chain of continuous maps, then $(g\circ f)_* = g_* \circ f_*$.
\end{prop}

\begin{proof}
    Let $[\gamma] = [\gamma ']$. Then, $f\circ \gamma \sim f\circ \gamma'$ via $f \circ H$, relative to $\{0,1\}$, so this is well-defined, and
    $$(f \circ \gamma ) \star (f \circ \mu) = f\circ(\gamma \star \mu)$$
    at the curves level, so $f_*$ is a homomorphism. Finally, $(g \circ f)\circ\gamma = g \circ (f \circ \gamma)$ gives
    $(g \circ f)_* =g_* \circ f_*$.
\end{proof}

\begin{thm}[Functoriality of the Fundamental Group]
The map $\pi_1$ on spaces and the continuous maps between them is a functor from the category of pointed spaces to the category of groups. That is,
$$\pi_1 : \w{Top_\bullet} \to \w{Grp}$$
is a covariant functor.
\end{thm}

\begin{defn}
    $X$ is called \term{strongly contractible} if there exists $p \in X$ such that $X$ deformation retracts to $p$.
\end{defn}

\begin{prop}
    If $X$ is strongly contractible, then $\pi_1(X,q) = \{e\}$ for all $q \in X$.
\end{prop}

\begin{proof}
When $q=p$, we know $\id _X \overset{\w{rel} p}{\sim}(x \mapsto p)$, and it follows that $\pi_1(X,p)$ is trivial. Then, the proposition follows for arbitrary $q$ by \Cref{thm:pathconnectedisom}, since strongly contractible implies path connected.
\end{proof}

\begin{notebox*}[Recall]
    $X$ is called contractible if $X \sim \{q\}$, i.e. $\id_X \sim (x \maps q)$.
\end{notebox*}

\begin{thm}
    If $X$ is contractible, then $\pi_1(X,q) = \{e\}$ for all $q \in X$.
\end{thm}

\section{\texorpdfstring{The Fundamental Group of $S^1$}{The Fundamental Group of S1}}

\begin{thm}[][label=thm:circlelift]
    Let $\gamma:[0,1] \to S^1$ be continuous with $\gamma(0)=1$. Then, there exists a unique $\theta:[0,1] \to \R$ such that $\theta (0)=0$, and
    $$e^{i\theta(t)}=\gamma(t).$$
    Moreover, we can do this continuously along homotopies. That is, if $H:[0,1]\times [0,1] \to S^1$ has $H(0,s)=1$, then there exists a unique $\theta: [0,1] \times [0,1] \to \R$ such that $\theta(0,s)=0$ for all $s$, and $e^{i\theta}=H$ for all $t,s$.
\end{thm}


\begin{cor}
    Let $\gamma_1,\gamma_2:[0,1] \to S^1$ be such that $\gamma_i(0)=\gamma_i(1)=1$, $i=1,2$. Let $\theta_1,\theta_2$ be their lifts as in \Cref{thm:circlelift}. Then,
    $$\gamma_1 \overset{\w{rel}\{0,1\}}{\sim}\gamma_2 \iff \theta_1(1)=\theta_2(1)$$
\end{cor}

\begin{proof}
($\implies$) Suppose $\gamma_1 \sim \gamma_2$ relative to $\{0,1\}$, via a homotopy $H$. Let $\theta:[0,1]\times [0,1] \to \R$ be the lift of $H$ as in \Cref{thm:circlelift}. Then, $\theta_1(x)=\theta(x,0)$ and $\theta_2(x)=\theta(x,1)$. Consider the path $s \mapsto \theta(1,s)$. Then, since $H(1,s)=1$, and $e^{i\theta}=H$, we have $\theta(1,s) \in 2 \pi \Z$. So, as $\theta(1,s)$ is continuous with a discrete image, it must be constant, hence $\theta_1(1)=\theta_2(1)$.

$(\impliedby)$ Take $\theta(t,s)= (1-s)\theta_1(t) + s\theta_2(t)$. Then, $\theta(t,s)$ is a homotopy between $\theta_1$ and $\theta_2$, with $\theta(1,s)=\theta_1(1)$, and $\theta(0,s)=0$. Now, we may take $H(t,s):= e^{i\theta(t,s)}$, and $H$ is the desired homotopy from $\gamma_1$ to $\gamma_2$ relative to $\{0,1\}$; it is relative to $\{0,1\}$ because $\theta(0,s)=0$ and $\theta(1,s)=\theta_1(1)=\theta_2(1)$ for all $s$.
\end{proof}

\begin{cor}
For any $p \in S^1$,
    $$\pi_1(S^1,p) \cong \Z,$$
    with $t \mapsto e^{2\pi i t}:[0,1]\to S^1$ being the generator.
\end{cor}

\begin{proof}
    Clearly, $(e^{2\pi i t})^k \overset{\w{rel}\{0,1\}}{\sim} e^{2\pi i kt}$, where the LHS denotes concatenation $k$ times. On the other hand, each element in $\pi_1(S^1,1)$ is represented by $t \mapsto e^{2\pi i kt}$ for exactly one $k \in \Z$. Indeed, if $\theta$ is the lift of an element in $\pi_1(S^1,1)$, then $\theta(1) \in 2\pi \Z$, and the bijective correspondence is induced by $\gamma \leftrightarrow \frac{\theta(1)}{2\pi}$. The case of general $p$ follows by \Cref{thm:pathconnectedisom}.
\end{proof}

\begin{prop}
    For all $k \geq 2$, $\pi_1(S^k,p)$ is trivial.
\end{prop}

\begin{proof}
    Let $\gamma:[0,1] \to S^k$ be continuous. We claim $\gamma$ is homotopic, relative to $\{0,1\}$, to a curve $\tilde \gamma$ that misses a point $q$. The proposition follows from the claim: indeed, we have $\psi:S^k \setminus \{q\} \arr[\sim] \R^k $, so consider $\mu:=\psi \circ \tilde \gamma$, which is homotopic to the constant map, and we may push it back to $S^k$ via $\psi^{-1}$ to show $[\gamma]=[\tilde \gamma]=[p]$, where $p = \gamma(0)$.

    To prove the claim, we approximate $\gamma$ via smooth maps, and smooth maps always miss points by \href{https://en.wikipedia.org/wiki/Sard%27s_theorem}{Sard's theorem}.
\end{proof}

\section{Van Kampen's Theorem}

\begin{thm}[Van Kampen's Theorem]
    Let $(X,p)$ be a connected pointed space, and suppose
    $$X = \bigcup_{\alpha \in I}(U_\alpha,p)$$
    where $U_\alpha \subset X$ is open and $p \in U_\alpha$. Suppose that for every $\alpha,\beta \in I$, $U_\alpha \cap U_\beta$ is path connected. Then, the map 
    \begin{align*}
        \Phi: &*_{\alpha}\pi_1(U_\alpha,p) \to \pi_1(X,p)
        \\ &*_{\alpha}[\gamma_\alpha]\mapsto \star_\alpha[\gamma _\alpha]
    \end{align*}
    is surjective. Moreover, if $U_\alpha \cap U_\beta \cap U_\delta$ is path-connected for all $\alpha,\beta, \delta \in I$, then
    $$\pi_1(X,p) \cong *_\alpha\pi_1(U_\alpha, p)/\aangles{ i_{\alpha ,\beta}(\gamma) i_{\beta,\alpha}(\gamma^{-1})},$$
    where
    $$i_{\alpha,\beta}:\pi_1(U_\alpha \cap U_\beta, p) \to \pi_1(U_\alpha, p)$$
    is the homomorphism induced by the inclusion, and $\aangles{-}$ is the normal closure. In particular, if $(X,p) = U_1 \cup U_2$ with $U_1 \cap U_2$ path connected, then
    $$\pi_1(X,p) = \pi_1(U_1,p) \underset{\pi_1(U_1\cap U_2,p)}{*} \pi_1(U_2,p)$$
\end{thm}

\begin{ex}
    Take $X= \bigvee_{\alpha \in I}(S^1,p)$, which is the space defined by attaching $I$ copies of $S^1$ at one shared point $p$. For each $\alpha$, take $U_\alpha = S^1_\alpha \cup N_\veps (p)$. Then,
    $$U_\alpha \cap U_\beta \cap U_\delta = N_\veps(p)$$
    for every $\alpha,\beta,\delta \in I$ is path connected (and contractible). Thus, by van Kampen's theorem,
    $$\pi_1(X,p) \cong *_\alpha \pi_1(S^1_\alpha,p)\cong *_\alpha \Z$$
    Indeed, the normal subgroup is trivial because $N_\veps(p)$ is contractible.
\end{ex}

Let $(X,p)$ be a pointed space, $X = \bigcup _{\alpha \in I}(U_\alpha,p)$, suppose $U_\alpha \cap U_\beta$ is path connected for all $\alpha, \beta$, and let $\Phi$ be the map
$$\Phi:*\pi_1(U_\alpha,p) \to \pi_1(X,p).$$
We claim $\Phi$ is surjective. Let $\gamma:[0,1] \to X$ be a loop at $p$, and we want to decompose it into loops in $(U_\alpha,p)$. Since $\gamma$ is continuous, the set
$$\{ \gamma^{-1}(U_\alpha)\} _{\alpha \in I}$$
is an open cover of $[0,1]$. Thus, there exists $\delta>0$ called the \href{https://en.wikipedia.org/wiki/Lebesgue%27s_number_lemma}{Lebesgue number}, such that any open subset of $[0,1]$ with diameter $<\delta$ is contained in $\gamma^{-1}(U_\alpha)$ for some $\alpha\in I$. Assume $\delta = 1/k$ for $k \in \N$, and partition the interval $[0,1]$ into $2k$ sub-intervals of length $1/(2k)$. Thus, each sub-interval $[\frac{i-1}{2k},\frac{i}{2k}]$ is mapped via $\gamma$ into some $U_{\alpha_i}$. Let $t_i = i/(2k)$ and $p_i = \gamma (t_i) \in U_{\alpha_i }\cap U_{\alpha_{i+1}}$. Since $U_{\alpha_i}\cap U_{\alpha_{i+1}}$ is path connected, there exists a curve $\mu_i$ in $U_{\alpha_i}\cap U_{\alpha_{i+1}}$ from $p_i$ to $p$. Let $\gamma_i = \gamma|_{[t_{i-1},t_i]}$. Then,
$$\gamma = \gamma_{2k} \star \cdots \star \gamma_1 \sim \gamma_{2k}\cdots \gamma_{3}\mu_2^{-1}\mu_2 \gamma_2 \mu_{1}^{-1}\mu _1 \gamma_1,$$
proving surjectivity.

Now, in order to prove the desired isomorphism, it suffices to show $$\ker (\Phi) =  \angles{\angles{i_{\alpha \beta}(\gamma) \cdot i_{\beta\alpha}(\gamma^{-1})}}=:N.$$
Indeed, first observe that 
$$\Phi(i_{\alpha \beta}(\gamma) \cdot i_{\beta\alpha}(\gamma^{-1})) = [\gamma] [\gamma^{-1}]=e,$$
so clearly $N \subset \ker \Phi$.

For the other direction, we want to show that if $\gamma_1 \star \cdots \star \gamma_{k}$ is homotopic to $[e]$ as loops in $X$, then $\gamma_1 \star \cdots \star \gamma_k \in N$. This is provable, but the proof was confusing in lecture... I didn't feel like finishing it.
\end{proof}

\begin{ex}[The Fundamental Group of $\R^3 \setminus S^1$]
    We can imagine the sphere of which $S^1$ is a cross section ($S^2$), and deformation retract everything outside that sphere down to it (that is, we can ignore everything far outside our circle). We claim that $\bar B \setminus S^1$ deformation retracts to $S^2\cup L$, where $B$ is the closed ball corresponding to our sphere $S^2$, and $L$ is the diameter of the sphere perpendicular to the disk bounded by $S^1$ (so $L$ passes through the center of $S^1$, missing $S^1$ itself).

    Then, $L$ can be homotoped within the interior of $S^2$ to a closed loop touching a point of $S^2$. That is, our space retracts to $S^2 \lor S^1$, and by van Kampen's theorem
    $$\pi_1(\R^3 \setminus S^1) = \pi_1(S^2 \lor S^1) = \pi_1(S^2)*\pi_1(S^1)=\Z.$$
\end{ex}

\begin{ex}

\end{ex}

\begin{ex}

The idea is as follows: we thicken one of our circles so that it becomes a torus, and we surround our torus and circle by a sphere so that the torus touches the boundary of the sphere. Then, the second circle is ``jump-roped'' around the torus so that it reaches the boundary of the sphere at the same point as the torus.
\end{ex}

\begin{ex}
    Consider the space $S^n$, for $n \geq 2$, and a point $p \in S^n$. We claim the fundamental group is trivial. Indeed, we may pass $S^{n-1}$ through our point, which splits $S^n$ into an ``upper-hemisphere'' $U$ and ``lower-hemisphere'' $L$. \end{ex}

\begin{prop}
Suppose $Y$ is obtained from a path-connected pointed space $(X,x_0)$ by attaching a $2$-cell, i.e.
\[
Y= X\sqcup \bar D^2/ (\varphi(x) \sim x),
\]
where $x \in \partial \bar D^2$ and $\varphi:\partial D^2 \to X$.  Let $p \in X$ be a point such that there exists $\tilde p \in \partial D^2$ with $\varphi(\tilde p)=p$. Let $\mu_p$ be a path in $X$ between $x_0$ and $p$. Then,
\[
\pi_1(Y,x_0) = \pi_1(X,x_0)/ \aangles{\mu_p^{-1} \star \varphi \star \mu_p}.
\]
\end{prop}
\begin{proof}
    Let $p'$ be a point in the interior of $D$ obtained by moving radially inward from $\tilde p$. Let $\tilde \varphi$ be a loop at $p'$, let $\varphi_1$ be a loop in $D^2$ with $\tilde \varphi$ in the interior, and $\varphi_2$ be a loop with $\tilde \varphi$ in the exterior. Let $G$ be the region within the loop $\varphi_1$, and $P$ the region outside the loop $\varphi_2$. Then, $p'$ is contained in $G \cap P$, and we may write $Y=G \cup P$. Moreover, $G \cap P$ is an annulus which is homotopy equivalent to $S^1$. It follows that
    \[
    \pi_1(Y,p') = \pi_1(G,p')*_{\pi_1(S^1,p')}\pi_1(P,p').
    \]
    Moreover, $\tilde \varphi$ is homotopic to $\bar{p'p}^{-1}\varphi \bar{p'p}$, so
    \[
    \pi_1(Y,p')\cong \pi_1(P,p')/\aangles{\bar{p'p}^{-1}\varphi \bar{p'p}}.
    \]
    But, note that $P$ deformation retracts via $\mu_p\bar{p'p}^{-1}$ to give the desired result.
\end{proof}

\begin{defn}
    Let $X$ be path connected. A space $\tilde{X}$ with a continuous map $p:\tilde{X}\to X$ is called a \term{covering space} for $X$ if for all $x\in X$ there exists an open set $U\ni x$ such that $p^{-1}(U)=\coprod_{\alpha\in I_{x}}U_{\alpha}$ for some index set $I_{x}$, where $p:U_{\alpha}\to U$ is a homeomorphism for all $\alpha \in I_x$.
\end{defn}

\end{ex}

\end{ex}

\end{ex}

\end{ex}

\begin{prop}[][label=prop:liftingtoloop]
    Suppose

    (\tilde X, \tilde{x_0}) \arr[p] (X,x_0)

    is a covering, with $X$ path connected, and fix $[\gamma] \in \pi_1(X,x_0)$. Then, $[\gamma] \in p_*\paren{\pi_1(\tilde X,\tilde{x_0})}$ if and only if $\gamma$ lifts to a loop at $\tilde{x_0}$.

\begin{thm}[][label=thm:lifting-criterion]
    Suppose $Y$ is path connected and locally path connected. Then, a map $f:(Y,y_0) \to (X,x_0)$ can be lifted to $\tilde f:(Y,y_0) \to (\tilde X,\tilde {x_0})$ if and only if
    \[
    f_*\paren{\pi_1(Y,y_0)} \subset p_*\paren{\pi_1(\tilde X,\tilde {x_0})}
    \]
\end{thm}


\begin{proof}
    If a lift exists, i.e. there exists $\tilde f$ such that $p \circ \tilde f = f$, then
    \[
    f_* =p_* \circ \tilde{f_*} \implies \Img(f_*) \subset \Img (p_*),
    \]
    proving the first direction. For the other direction, fix $y \in Y$, and take a path $\gamma$ connecting $y_0$ to $y$. Then, $f \circ \gamma$ is a curve in $X$ starting at $x_0$, so by the homotopy lifting property there exists $\tilde{\mu}(t)$ lifting $f(\gamma(t))$. Define $\tilde f_\gamma(y):=\tilde{\mu}(1)$. We now need to show that this $\tilde f$ is independent of the choice of $\gamma$. 

    Let $\gamma_1,\gamma_2$ be two such paths from $y_0$ to $y$. Define $\mu_i(t)=f(\gamma_i(t))$, and let $\tilde{\mu_i}$ be the lifts. We want to show $\tilde{\mu_1}(1)=\tilde{\mu_2}(1)$. Note that $\mu_1^{-1} \star \mu_2$ is a loop in $X$ at $x_0$, and 
    \[
    [\mu_1^{-1} \star \mu_2] = f_*\paren{[ \gamma_1^{-1} \star \gamma_2 ]}\in f_*(\pi_1 (Y,y_0)) \subset p_*(\pi_1 (\tilde X, \tilde{x_0})).
    \]
    Thus, $\mu_1,\mu_2,$ and $\mu_1^{-1} \star \mu_2$ have all been lifted to $\tilde X$, and these lifts are unique. It follows that $\tilde{\mu_1}(1)=\tilde{\mu}_2(1)$. Since we have well-definedness, we may set $\tilde f(y)=\tilde f_{\gamma}(y)$ for any $\gamma$.

    % We're now left to show continuity. Fix $y \in Y$, and let $U \ni f(y)$ be as in the covering space definition. Take $U_\alpha \subset \tilde X$ containing $\tilde f(y)$, so that $U \arr[p]U_\alpha$ is a homeomorphism. Select $V \subset f^{-1}(U)$ to be path connected with $y\ in V$.
    Continuity is left to the reader.
\end{proof}

\begin{thm}[Unique Lifting Property][label=thm:uniq-lifting-prop]
    Suppose $Y$ is path connected, $f:Y \to X$ is continuous, and $\tilde f,\tilde f':Y \to \tilde X$ are two lifts of $f$. If $\tilde f(y)=\tilde f'(y)$ for some $y \in Y$, then $\tilde f \equiv \tilde f'$. 
\end{thm}

\begin{proof}
    Consider the set $A:=\brax{ y \setbar \tilde f(y)= \tilde f'(y)}$, which we assume is non-empty. It suffices to show that $A$ is clopen. Fix $y \in A$, and select $U\ni y$ as in the definition of a covering space. Let $U_\alpha \subset \tilde X$ map homeomorphically to $U$ via $p$, and such that $\tilde f(y) = \tilde f'(y) \in U_\alpha$. We claim $y \in f^{-1}(U) \subset A$. Indeed, since $p$ is a homeomorphism from $U_\alpha$ to $U$, one sees that this is the case.
\end{proof}

\section{Universal Covers}

\begin{defn}
    A \term{universal cover} of $X$ is a simply connected cover of $X$.
\end{defn}

\begin{que}
    Does there always exist a universal cover?
\end{que}

\begin{defn}
A space $X$ is called \term{semi-locally simply connected} if for every point $x \in X$, there exists a neighborhood $U \ni x$ such that the image of the map $i_*:\pi_1(U,x)\to \pi_1(X,x)$ induced by the embedding $i:U \hookrightarrow X$ is trivial.
\end{defn}

\begin{rmk}
    Simply connected implies semi-locally simply connected.
\end{rmk}

\begin{prop}
If $X$ has a universal cover, then $X$ is semi-locally simply connected.
\end{prop}

\begin{proof}
    Fix $x \in X$, and let $U \ni x$ be as in the definition of the covering space $\tilde X$, which is our universal covering. Let $\tilde x \in \tilde X$ be such that $p(\tilde x)=x$, and let $\tilde U_\alpha \ni \tilde x$ be such that $p: \tilde U_\alpha \to U$ is a homeomorphism. Then, any loop $\gamma$ at $x$ in $U$ lifts to a loop $\tilde \gamma$ at $\tilde x$ in $\tilde U_\alpha$. However, since $\tilde X$ is simply connected by our assumption, we know $\tilde \gamma$ is homotopically trivial in $\tilde X$, and composing this homotopy with $p$ shows $\gamma$ is homotopically trivial in $X$.
\end{proof}

\begin{thm}[Existence of the Universal Cover]
    Suppose $X$ is path connected, locally path connected, and semi-locally simply connected. Then, $X$ has a universal cover.
\end{thm}

\begin{proof}
    We want to construct the universal cover $\tilde X$. Suppose, for now, that such a universal cover existed, and $p:\tilde X \to X$ is the projection. Fix $x_0 \in X$, and $\tilde {x_0}\in p^{-1}(x_0)$. Given a curve $\gamma$ at $x_0$, we can lift it to $\tilde \gamma$ starting at $\tilde{x_0}$. Now, consider
    \[
    \Gamma:= \{\gamma:[0,1] \to X, \gamma(0)=x_0\}.
    \]
    Then, we have an induced map $F:\Gamma \to \tilde X$ given by $F(\gamma):=\tilde \gamma(1)$, where $\tilde \gamma$ is the corresponding lift. Since $\tilde X$ is path connected, $F$ is surjective. Moreover, if $F(\gamma_1)=F(\gamma_2)$, we must have $\gamma_1(1)=\gamma_2(1)$, and $\tilde{\gamma_2}^{-1} \star \tilde{\gamma_1}$ is a loop in $\tilde X$ and therefore homotopically trivial, which means $\gamma_2^{-1} \star \gamma_1$ is as well. That is, $\gamma_1\overset{\w{rel\{0,1\}}}{\sim}  \gamma_2$. Conversely, if $\gamma_1 \sim \gamma_2$, then by the homotopy lifting property, $F(\gamma_1)=F(\gamma_2)$. Thus, $F$ induces a bijection $(\Gamma/\sim) \leftrightarrow \tilde X$, where $\sim$ is homotopy relative to $\{0,1\}$. This hints that, to find our universal cover $\tilde X$, we should use the collection $\Gamma / \sim$.
    
    Indeed, define
    \[
    \tilde X:=\brax{ [\gamma]_{\w{rel}\{0,1\}} \setbar \gamma:[0,1]\to X,\gamma(0)=x_0 },
    \]
and let $p:\tilde X \to X$ be the projection $p([\gamma])=\gamma(1)$, which is well-defined. Define a basis $B$ for $X$ by
\[
B := \brax{U \text{ open, path connected, and } i_*(\pi _1(U,x)) \subset \pi_1(X,x) \text{ is trivial for all } x \in U}.
\]
Note that if $V \subset U\in B$ is open and path-connected, then $V \in B$. Thus, if $U_0 \cap U_1$ is non-empty for $U_0,U_1\in B$, then for any $x \in U_0 \cap U_1$, there exists an open path connected $V \ni x$, $V \subset U_0 \cap U_1$, because $X$ is locally path connected, and then $V \in B$. Moreover, $B$ covers $X$ (by semi-local simple connectivity and local path connectivity), hence $B$ indeed defines a basis for $X$.

Now, to obtain a topology for $\tilde X$: for every $[\gamma] \in \tilde X$ and $U \in B$ such that $\gamma(1) \in U$, take
\[
U_{[\gamma]}:= \brax{ [\mu\star\gamma] \setbar \mu(0)=\gamma(1), \Img (\mu) \subset U }.
\]
Then, $p$ induces a bijection $U_{[\gamma]} \to U$ for similar reasons as before. Moreover, if $[\gamma_2] \in U_{[\gamma]}$, then $U_{[\gamma_2]}=U_{[\gamma]}$. Thus
\[
\tilde B := \brax{ U_{[\gamma]} \setbar U \in B, \gamma(1) \in U}
\]
defines a basis for $\tilde X$: if $U^1_{[\gamma_1]} \cap U^2_{[\gamma_2]}\neq \emptyset$, then we may take $\gamma_3$ in the intersection so that $U^1_{[\gamma_1]}=U^1_{[\gamma_3]}$ and $U^2_{[\gamma_2]}=U^2_{[\gamma_3]}$. Thus, since $B$ is a basis for $X$, we may take $V \in B$ with $\gamma_3(1) \in V \subset U^1 \cap U^2$, and then $V_{[\gamma_3]} \in \tilde B$ with $[\gamma_3] \in V_{[\gamma_3]} \subset U^1_{[\gamma_3]} \cap U^2_{[\gamma_3]}$, proving $\tilde B$ is a basis.

One now checks that $\tilde X$, with the topology induced by $\tilde B$, indeed defines a covering. Finally, to prove that it's simply connected: fix $[\gamma]$ a curve on $X$, and define
\[
[\gamma]_t(s) := \case{ \gamma(s) & 0 \leq s \leq t \\ \gamma(t)  & t \leq s \leq 1},
\]
which demonstrates that $\tilde X$ is path-connected. Moreover, if $[\gamma] \in p_* \paren{\pi_1(\tilde X,\tilde x_0)}$, and $\tilde \gamma$ is the lift, then we know that $[\gamma] = [x_0]$, and so $\tilde \gamma$ is homotopically trivial, proving simply connectedness.
\end{proof}

    We have $p:(\tilde X, \tilde{x_0}) \to (X,x_0)$, the quotient map $\tilde p:(\tilde X,\tilde{x_0}) \to (X_H,x_0^H)$, where $x_0^H:=\braks{\tilde{x_0}}$, and we have $p_H : (X_H,x_0^H)\to (X,x_0)$. We claim $p_H$ is a covering map.

        For $g \in \pi_1(X,x_0)$, if $g(U_\alpha)\cap U_\beta \ne \emptyset$, then $g(U_\alpha)=U_\beta$.
    \end{claim}

\begin{defn}
    Let $\tilde X \arr[p] X$ be a covering map, and let $T : \tilde X \to \tilde X$ be a homeomorphism such that $p \circ T = p$. Then, $T$ is called a \term{deck transformation}.
\end{defn}

\begin{defn}
    Let $\tilde X$ be a topological space, and $G$ act on $\tilde X$ on the left via continuous maps. The action of $G$ is called \term{properly discontinuous} if for all $\tilde x \in \tilde X$, there exists $\tilde U \ni \tilde x$ such that for all $g \in G$, $g(\tilde U)\cap \tilde U \neq \emptyset \implies g=e$.
\end{defn}

\begin{thm}[][label=thm:quotientaction]
    Let $\paren{\tilde X,\tilde{x_0}}$ be a simply connected, locally path connected, path connected space. Let $G$ act on $\tilde X$ properly discontinuously, and set $X:= \tilde X/G$ with the natural projection $p:\tilde X \to X$. Then, $\tilde X$ is the universal cover of $X$, and setting $x_0 = p(\tilde x_0)$ gives
    \[
    \pi_1(X , x_0)\cong G
    \]
\end{thm}

\begin{proof}
Take $x \in X$, $\tilde x \in p^{-1}(x)$, and let $\tilde U$ be a neighborhood of $\tilde x$ as in the definition, and $U = p(\tilde U)$. Then, $p:\tilde U \to U$ is a homeomorphism, and $p^{-1}(U)=\bigsqcup_{g \in G}g\tilde U.$ Thus, $\tilde X$ is the universal cover, and both $\pi_1(X,x_0)$ and $G$ act on $\tilde X$ by deck transformations. For all $\tilde x_0' \in p^{-1}(x_0)$, there exists a unique $g \in G$ such that $g(\tilde{x}_0) = \tilde x_0'$, and there exists a unique $\gamma \in \pi_1(X,x_0)$ such that $F_\gamma(\tilde x_0)=\tilde x_0'$. The map $\gamma \mapsto g$ is the desired isomorphism.
\end{proof}

\end{ex}

\begin{defn}
Let $v_0,\dots,v_n \in \R^m$ be ordered points such that no points $v_{i_1},\dots,v_{i_k}$ lie in any $k-2$ dimensional affine space. The $n$-\term{simplex} $[v_0,\dots,v_n]$ is the convex hull of $v_0,\dots,v_n$. That is,
\[
[v_0,\dots,v_n]= \brax{ \sum t_iv_i \setbar t_i\geq 0, \sum t_i=1 }
\]
\end{defn}

\begin{ex}
The \term{standard $n$-simplex} $\Delta^n \subset \R^{n+1}$ is the simplex $[e_0,\dots,e_{n}]$, where $e_0,\dots,e_n$ is the standard basis of $\R^{n+1}$. For instance, the standard $1$-simplex is:

\end{figure}
\end{ex}



\begin{defn}
Given an $n$-simplex $[v_0,\dots,v_n]$, the $i$-th \term{face} is the $(n-1)$-simplex $[v_0,\dots,v_{i-1},v_{i+1},\dots,v_n]$. The \term{boundary} $\partial[v_0,\dots,v_n]$ of a simplex is the union of its faces. The \term{interior} is the simplex minus its boundary.
\end{defn}

\begin{defn}
A topological space $X$ is called a \term{$\Delta$-complex} if there exists a collection of continuous maps $\brax{\sigma_\alpha : \Delta^{n(\alpha)}\to X}_{\alpha \in I}$ with the following properties:
\begin{enumerate}[(i)]
    \item For every $\alpha \in I$, the restriction $\sigma_\alpha|_{\mathring \Delta^{n(\alpha)}}$ is injective, and for all $x \in X$, there exists a unique $\alpha$ such that $x \in \Img (\sigma_{\alpha}|_{\mathring \Delta^{n(\alpha)}})$.
    \item $U \subset X$ is open if and only if $\sigma_{\alpha}^{-1}(U)$ is open for all $\alpha \in I$
    \item For all $\alpha \in I$, and every $0 \leq i \leq n(\alpha)$, the restriction of $\sigma_\alpha$ to the $i$-th face is in the collection of maps. That is, there exists $\beta\in I$ such that $\sigma_\beta =\sigma_\alpha|_{[e_0,\dots,\hat e_i,\dots,e_{n(\alpha)}]}.$ This is the case if and only if, denoting by $\iota_i:\Delta^{n(\alpha)-1}\to \Delta^{n(\alpha)}$  the map sending $(x_0,\dots,x_{n(\alpha)-1}) \mapsto (x_0,\dots,x_{i-1},0,x_{i},\dots,x_{n(\alpha)-1})$, we have $\sigma_\beta = \sigma _\alpha \circ \iota_i$.
\end{enumerate}
\end{defn}

\begin{defn}
Let $X$ be a $\Delta$-complex, with simplices $\{\sigma_\alpha : \Delta^{n(\alpha)} \to X\}_{\alpha \in I}$, let $I(n)=\{\alpha \mid n(\alpha)=n\}$, and $I = \bigcup I(n)$. The \term{group of simplicial $n$-chains} $\Delta_n(X)$ is the free abelian group on $\{\sigma_\alpha\}_{\alpha \in I(n)}$. That is,
\[
\Delta_n(X)= \brax{ \sum_{\alpha \in I(n)} m_\alpha \sigma_\alpha \setbar m_\alpha \in \Z, m_\alpha=0 \text{ for all but finitely many }\alpha}.
\]
Such an expression $\sum m_\alpha \sigma_\alpha$ is called a \term{simplicial chain}. 
\end{defn}

\begin{defn}
The $n\th$ \term{boundary map} $\partial_n :\Delta_n(X) \to \Delta_{n-1}(X)$ is the homomorphism defined on the basis by 
\[
\partial_n(\sigma_\alpha)=\sum_{i=0}^{n}(-1)^{i}\sigma_\alpha\big |_{[e_0,\dots,\hat e_i,\dots,e_n]}
\]
\end{defn}

\begin{prop}[][label=prop:cycle-condition]
$\partial_{n-1}\circ \partial _n\equiv 0$.
\end{prop}

\begin{proof}
We break the sum up corresponding to when the second removed basis element comes before or after the first:
\begin{align*}
    \partial_{n-1}\circ \partial_n(\sigma_\alpha) &= \partial_{n-1}\paren{ \sum_{i=0}^{n} (-1)^i \sigma_\alpha|_{[e_0,\dots,\hat{e_i},\dots,e_n]} }
    \\ &=\sum_{i=0}^{n}\paren{\sum_{j=0}^{i-1}(-1)^{i+j}\sigma_\alpha|_{[e_0,\dots,\hat{e_j},\dots,\hat{e_i},\dots,e_n]}+ \sum_{j=i+1}^n(-1)^{i+j+1} \sigma_\alpha|_{[e_0,\dots,\hat{e_i},\dots,\hat{e_j},\dots,e_n]}}.
\end{align*}
Now, for every $k<\ell \in \{0,\dots,n\}$, the $(n-2)$-simplex
\[
\sigma_\alpha|_{[e_0,\dots,\hat{e_k},\dots,\hat{e_\ell},\dots,e_n]}
\]
appears twice, for $(k,\ell)=(i,j)$ and $(k,\ell)=(j,i)$, but they appear with opposite sign. Therefore, they cancel, and the resulting sum is zero.
\end{proof}


\begin{defn}
A \term{chain complex} is a sequence of abelian groups $\{C_n\}_{n=0}^\infty$ with homomorphisms $\partial_n : C_n \to C_{n-1}$ such that $\partial_{n-1} \circ \partial_n \equiv 0$. 

The elements of $\ker(\partial_n)\subset C_n$ are called \term[$n$-cycle]{$n$-cycles}, and the elements of $\Img(\partial_{n+1})\subset C_n$ are called \term{$n$-boundaries}. 

Note that the condition $\partial_{n} \circ \partial_{n+1} \equiv 0$ means that every $n$-boundary is an $n$-cycle, since $\Img(\partial_{n+1})\subset \ker(\partial _n)$. The $n\th$ \term{homology group} is the quotient $H_n(C):=\ker(\partial_n)/\Img(\partial_{n+1})$.
\end{defn}

\begin{defn}
By \Cref{prop:cycle-condition}, the simplicial $n$-chains $\{\Delta_n(X)\}_{n=0}^\infty$ along with the boundary maps $\partial_n:\Delta_n(X)\to \Delta_{n-1}(X)$ form a chain complex. The $n$-th \term{simplicial homology group} of $X$ is
\[
H_n^\Delta (X):= \ker(\partial_{n}(X))/\Img \paren{\partial_{n+1}(X)}
\]
\end{defn}

Finally, $H_0^\Delta (S^1) \cong \Z$ because $\Delta_0(X) \cong \Z$ (a single vertex) and $\partial_1 \equiv 0$.

\begin{rmk}
There are generally a few problems with simplicial homology. First, it's not immediately clear that the simplicial homology groups are independent of the choice of $\Delta$-structure. While this is true, it's tedious to show. Furthermore, doing computations with a $\Delta$-structure can be annoying, especially as we're taking more and more complicated spaces. We therefore look for a better solution.
\end{rmk}

\section{Singular Homology}

\begin{defn}
Let $X$ be a topological space. The group \term{$C_n(X)$} of \term{singular $n$-chains} in $X$ is the free abelian group on
\[
\brax{ \sigma: \Delta ^n \to X \setbar \sigma \text{ continuous}}.
\]
That is
\[
C_n(X) := \brax{ \sum_{i=1}^m k_i\sigma_i \setbar \sigma_i : \Delta^n \to X, k_i \in \Z, m \in \N}
\]
Each such $\sigma$ is called a \term{singular simplex}.

If $\sigma$ is a singular simplex, we define its \term{boundary} to be
\[
\partial \sigma = \sum_{i=0}^n(-1)^i\sigma \circ \iota_i.
\]
This extends to a homomorphism $\partial_n:C_n(X) \to C_{n-1}(X)$, and $\partial_{n-1}\circ \partial _n=0$. The \term[singular homology,$H_n$]{$n\th$ singular homology group $H_n(X)$} of $X$ is the $n\th$ homology of the chain complex $C_n(X)$. That is
\[
H_n(X)=\ker \partial _n/\Img\partial_{n+1}.
\]
The elements of $\ker \partial _n$ are called \term[cycle,cycles]{$n$-cycles}, and the elements of $\Img \partial_{n+1}$ are called \term[boundary,boundaries]{$n$-boundaries}.
\end{defn}

\begin{rmk}
If $f:X \to Y$ is continuous, and $\sigma:\Delta^n \to X$ is a singular $n$-simplex in $X$, then $f \circ \sigma$ is a singular $n$-simplex in $Y$. Hence $f_*: C_n(X) \to C_n(Y)$  sending $\sigma \mapsto f \circ \sigma$ defines a homomorphism. Moreover
\[
\partial_n \circ f_* = f_* \circ \partial _n
\]
\end{rmk}

\begin{defn}
Given a space $X$, define $\veps:C_0(X)\to \Z$ by 
\[
\veps\paren{\sum_{i=1}^mk_i\sigma_i}=\sum_{i=1}^mk_i,
\]
which is a homomorphism.
\end{defn}

\begin{prop}
For every $a \in C_1(X)$, $\veps(\partial a)=0$.
\end{prop}

\begin{proof}
If $a= \sum_{i=1}^m k_i\sigma_i$ with $\sigma_i:\Delta^1 \to X$, then
\[
\partial a = \sum_{i=1}^mk_i(\sigma_i(1)-\sigma_i(0)),
\]
hence
\[
\veps(\partial a)=\sum k_i - \sum k_i = 0.
\]
\end{proof}

This means $\veps$ defines a map $H_0(X) \to \Z$.

\begin{prop} \label{prop:path-connected-zero-reduced}
If $X$ is path connected, then $\veps :H_0(X) \to \Z$ is an isomorphism.
\end{prop}

\begin{proof}
Consider a 0-chain $\sum_{i=1}^m k_i p_i$, where $p_i \in X$. Then for every $i = 2,\dots,m$, consider a $1$-simplex $\gamma_i:[0,1]\to X$ with $\gamma_i(0)=p_1$ and $\gamma_i(1)=p_i$. Note $\partial \gamma_i = p_i -p_1$. Thus
\[
\sum_{i=1}^m k_ip_i - \partial \paren{ \sum_{i=2}^m k_i \gamma_i } = \paren{\sum_{i=1}^m k _i}p_1.
\]
Thus every element in $H_0(X)$ is represented by $kp$.  Fixing some $p_0 \in X$, for all $p \in X$ we have
\[
mp - mp_0 = \partial(m\gamma),
\]
where $\gamma$ is a curve from $p_0$ to $p$. Thus for all $x \in C_0(X)$, $x \in \braks{\veps(x)p_0}$. Therefore if $\veps(x)=\veps(y)$, then $[x]=[y]$, so $\veps$ is injective.

Furthermore, $\veps\paren{[mp_0]}=m$.
\end{proof}

\begin{defn}
The \term[reduced homology]{reduced homology groups} of $X$ are the homologies of the complex
\[
\cdots \to C_n(X) \arr[\partial_n]C_{n-1}(X) \arr \cdots \arr C_1(X)\arr[\partial _1]C_0(X) \arr[\veps] \Z \to 0
\]
We write this as
\[
\tilde H_n(X)= \case{ H_n(X) & n \geq 1 \\ \ker \veps /\Img \partial _1  & n=0.}

So by \Cref{prop:path-connected-zero-reduced}, $\tilde H_0(X)=0$ if $X$ is path connected.
\end{defn}

\begin{thm}
For $X = \{p\}$, $\tilde H_n(X)=0$ for all $n$.
\end{thm}

\begin{proof}
We know $\tilde H_0(X)=0$. Now observe that 
\[
\partial \sigma_n = \sum_{i=0}^n (-1)^i \sigma_{n-1} = \case{ \sigma_{n-1}  & n \text{ even} \\ 0 & n \text{ odd}},
\]
as there is only one $n$-simplex $\sigma_n$, and only one $(n-1)$-simplex $\sigma_{n-1}$. 

Now if $n$ is even then $\ker \partial _n = 0$, so $H_n(X)=0$. On the other hand if $n$ is odd, then $\ker \partial _n = \Img \partial_{n+1}$, so again $H_n(X)=0$.

\begin{thm}[Homotopy Invariance of Homology] \label{thm:homo-invar-of-hom}
Suppose $f,g:X \to Y$ are homotopic. Then $f_{*n},g_{*n}:H_n(X)\to H_n(Y)$ are equal. In particular, if $f:X \to Y$ is a homotopy equivalence, then $f_{*n}$ is an isomorphism.
\end{thm}

To prove this, we need a helpful algebraic notion first.
\begin{defn}
Let $\{C_n\},\{D_n\}$ be chain complexes, and $\brax{f_n:C_n \to D_n}_{n=0}^\infty,\brax{g_n:C_n \to D_n}_{n=0}^\infty$ chain maps. A sequence of homomorphisms $\brax{p_n:C_n \to D_{n+1}}_{n=0}^\infty$ is called a \term{chain homotopy} between $f$ and $g$ if $\partial_{n+1} p_n + p_{n-1}\partial _n = g_n-f_n$. In this case, we say that $f$ and $g$ are \term{chain homotopic}.
\end{defn}

\begin{claim} \label{clm:chain-homo-gives-isom}
If $f,g : \brax{C_n} \to \brax{D_n}$ are chain homotopic maps, then the induced maps $f_{*n}$ and $g_{*n}$ are equal.
\end{claim}

\begin{proof}
Let $[z] \in H_n(C)$, and we want to show $g_{*n}(z)-f_{*n}(z)=0$, i.e. that it's a boundary. Well if $p$ is a chain homotopy between $f$ and $g$, then
\[
g_{*n}(z)-f_{*n}(z) = \partial p_n(z)+ p_{n-1}(\partial z)= \partial p_n(z),
\]
as desired.
\end{proof}

\begin{proof}[Proof of Homotopy Invariance of Homology]
Let $f,g :X \to Y$ be homotopic, so we have a homotopy $F:X \times [0,1]\to Y$. By \Cref{clm:chain-homo-gives-isom}, it suffices to find a chain homotopy between $f_*$ and $g_*$.

Let $\sigma : \Delta^n \to X$ be a singular $n$-simplex. Then $f_* (\sigma) = f\circ \sigma$, and similarly for $g$. In fact, we may use the homotopy $F$ to have
\[
\Delta^n \times I \arr[\sigma \times \id]X \times I \arr[F]Y.
\]
This will be used to obtain a chain homotopy.
\end{proof}

\begin{cor}[The Mayer–Vietoris Sequence]
Suppose $X = A \cup B$ for $A,B$ open. Then there exists a long exact sequence
\[
\cdots \arr H_{n+1}(X) \arr[d_{n+1}] H_n(A\cap B)\arr H_n(A)\oplus H_n(B) \arr H_n(X)\arr[d_n]H_{n-1}(A\cap B)\arr \cdots
\]

\end{cor}

\begin{cor} \label{cor:hom-circle-zero}
For all $n \geq 0$, we have $\tilde H_n(S^n)=\Z$, and $\tilde H_k(S^n)=0$ for $k \neq n$.
\end{cor}

\begin{proof}
Write $S^n = U\cup L$, where $U$ is the thickened upper-hemisphere and $L$ the thickened lower-hemisphere. Now we want to use Mayer–Vietoris, and we observe
\begin{align*}
    \tilde H_k(U \cap L) &= \tilde H_k(S^{n-1}), \\
    \tilde H_k(U) &= \tilde H_k(L)=0.
\end{align*}
Hence we have the exact sequence
\[
\tilde H_k(S^{n-1}) \to 0 \to \tilde H_k(S^n) \to \tilde H_{k-1}(S^{n-1}) \to 0 \to \tilde H_{k-1}(S^n),
\]
thus for all $k$
\[
\tilde H_k(S^n) \cong \tilde H_{k-1}(S^{n-1}).
\]
The result then follows by induction on $n$, with the base case $S^0$ trivial.
\end{proof}

	\arrow["\partial", from=1-3, to=3-3]
	\arrow["S"', from=3-1, to=3-3]
\end{tikzcd}\]
\end{prop}

\begin{proof}
We compute
\begin{align*}
\partial S(\sigma) &=\partial b_\sigma (S\partial \sigma) \\
 &= S \partial \sigma - b_\sigma \partial S \partial \sigma \qquad (\Cref{rmk:sigma-as-partials-and-bari}) \\ 
 &= S \partial \sigma - b_\sigma S \partial \partial \sigma \qquad (\text{Induction}) \\ 
 &= S \partial \sigma,
\end{align*}
thus $S$ is a chain map.
\end{proof}

\begin{prop}
$S$ is chain homotopic to the identity.
\end{prop}

\begin{proof}
Define $T$ inductively by $T([\emptyset]):=0$ and $T(\sigma):=b_\sigma(\sigma - T(\partial \sigma))$. We claim
\begin{equation} \label{eq:chain-hom-identity-relation}
\partial T + T\partial = \id - S.
\end{equation}
Checking this for the base case is an exercise. Now inductively,
\begin{align*}
\partial T (\sigma) 
&= \partial b_\sigma ( \sigma - T(\partial \sigma)) \\
&= \sigma - T\partial \sigma - b_\sigma\partial(\sigma - T\partial \sigma) \\
&= \sigma - T\partial \sigma - b_\sigma \partial \sigma + b_\sigma (\partial \sigma - S \partial \sigma)- b_\sigma (T \partial \partial \sigma) \\
&= \sigma - T\partial \sigma-b_\sigma (S \partial \sigma) \\
&= \sigma - T\partial \sigma-S(\sigma),
\end{align*}
and rearranging gives \eqref{eq:chain-hom-identity-relation}.
\end{proof}

\begin{defn}
We define
\begin{align*}
S &: C_n(X) \to C_n(X) \\
T &:C_n(X) \to C_{n+1}(X)
\end{align*}
by pushing forward $S$ and $T$ via simplices. That is, defining $S,T$ amounts to defining them on $\sigma : \Delta^n \to X$ continuous. For instance
\begin{align*}
S(\sigma) &= \sigma_*\paren{S [ e_0,\dots,e_n]} \\
&= \sigma_* \paren{\sum m_{\alpha}[w_0^\alpha,\dots,w_n^\alpha]} \\
&= \sum{m_\alpha} \sigma |_{[w_0^\alpha,\dots,w_n^\alpha]},
\intertext{and similarly}
T(\sigma) &= \sigma_*(T[e_0,\dots,e_n]).
\end{align*}
\end{defn}

\begin{prop}
$S$ is a chain map, and $T$ is a chain homotopy between $S$ and $\id$.
\end{prop}

\begin{proof}
We compute
\begin{align*}
\partial S(\sigma) &= \partial \sigma_* \paren{S [e_0,\dots,e_n]} \\
&= \sigma_*\paren{\partial S[e_0,\dots,e_n]} \\
&= \sigma_*(S \partial [e_0,\dots,e_n]) \\
&= \sigma_*\paren{S\paren{\sum (-1)^i\braks{e_0,\dots,\hat{e_i},\dots,e_n}}} \\
&= \sum (-1)^i \sigma_*\paren{S\braks{e_0,\dots,\hat{e_i},\dots,e_n}} \\
&= \sum (-1)^i S\sigma|_{[e_0,\dots,\hat{e_i},\dots,e_n]} \\
&= S(\partial \sigma).
\end{align*}
The proof for $T$ is similar.
\end{proof}

We are now equipped to prove \Cref{thm:chain-homology-A-map}.
\begin{proof}[Proof of \Cref{thm:chain-homology-A-map}]
Since $\brax{\Int(U)\setbar U \in \cA}$ is an open cover of $X$, for every $\sigma :\Delta^n\to X$, there exists $m=m(\sigma)$ such that $S^{m(\sigma)}(\sigma)\in C_n^{\cA}(X)$. Now choose $m$ to be minimal.

Define $D:C_n(X)\to C_{n+1}(X)$ by $D(\sigma)=D_{m(\sigma)}(\sigma)$, where $D_m:=\sum_{j=0}^{m-1}TS^j$ satisfies $\partial D_m + D_m \partial = \id - S^m$, so then we define $\rho:C_n(X) \to C_n(X)$ by the relation
\begin{align*}
\partial D + D\partial = \id - \rho.
\end{align*}
We claim $\rho$ has all the desired properties. Namely,
\begin{enumerate}
    \item $\rho$ is a chain map: \begin{align*}
    \partial \rho(\sigma)= \partial (\id \sigma)-\partial \partial D\sigma - \partial D \partial \sigma = \partial \sigma -\partial D \partial \sigma = \rho(\partial \sigma)
    \end{align*}
    \item $\rho$ is well-defined: \begin{align*}
    \rho(\sigma)=\sigma - \partial D \sigma - D \partial \sigma = \sigma - \partial D_{m(\sigma)}\sigma -D\partial \sigma,
    \end{align*}
    and $\partial D_{m(\sigma)}\sigma + D_{m(\sigma)}\partial\sigma =\sigma-S^{m(\sigma)}\sigma$, so
    \begin{align*}
    \rho(\sigma)= \cdots=  S^{m(\sigma)}\sigma + D_{m(\sigma)}\partial \sigma - D\partial \sigma.
    \end{align*}
    Clearly $S^{m(\sigma)}\sigma \in C_n^{\cA}(X)$. Moreover 
    \begin{align*}
    D_{m(\sigma)}\partial \sigma - D\partial \sigma = \sum_{i=0}^n(-1)^i\sum_{j=m(\tau_i)}^{m(\sigma)-1}TS^j(\tau_i) \in C_n^{\cA}(X),
    \end{align*}
    where $\tau_i:=\sigma \circ \iota_i$ is the $i$-th face of $\sigma$,
    hence $\rho (\sigma) \in C_n^{\cA}(X)$.
    \item $i\circ \rho$ is chain homotopic to the identity: $\id - i\circ \rho = \partial D + D\partial$
    \item $\rho \circ i$ is chain homotopic to the identity: $\rho(i(\sigma))=\sigma$.
\end{enumerate}
This concludes the proof.
\end{proof}

\begin{defn}
Let $A \subset X$. The \term[]{chains in $X$ relative to $A$}, denoted \term[]{$C_n(X,A)$}, is
\begin{align*}
C_n(X,A):= C_n(X)/C_n(A).
\end{align*}
Observe that $\partial _n:C_n(X) \to C_{n-1}(X)$ descends naturally to $\partial_n:C_n(X,A)\to C_{n-1}(X,A)$, so we denote by \term[]{$H_n(X,A)$} the homology groups of this complex. Namely, $[Z]\in C_n(X,A)$ is a \term[]{relative cycle} if $\partial Z \in C_{n-1}(A)$, and $[Z]=0$ in $H_n(X,A)$ if and only if there exists $Z_{n+1}\in C_{n+1}(X)$ such that $\partial Z_{n+1}=Z-a$ for $a \in C_n(A)$.
\end{defn}

\subsection{Homology of quotients}

\begin{thm} \label{thm:quotient-homology-isom}
Suppose $A \subset X$ is closed and $A$ is a deformation retract of $A \subset U \subset X$, with $U$ open. Then the quotient map
\begin{align*}
q_{n*}: H_n(X,A) \to H_n(X/A, A/A) \cong \tilde H_n (X/A)
\end{align*}
is an isomorphism.
\end{thm}

To prove this, we need a few lemmas.

\begin{lem} \label{lem:HnUA-zero}
If $A \subset U$, and $U$ deformation retracts to $A$, then $H_n(U,A)=0$.
\end{lem}

\begin{proof}
Since $U$ deformation retracts to $A$, the embedding $i:A \to U$ induces an isomorphism of homologies $i_{n*}:H_n(A) \to H_n(U)$. Now we have the long exact sequence
\begin{align*}
H_n(A) \arr[\sim] H_n(U) \to H_n(U,A) \to H_{n-1}(A) \arr[\sim] H_{n-1}(U),
\end{align*}
but isomorphisms are bijective so the middle section becomes
\begin{align*}
H_n(U)\arr[0]H_n(U,A)\arr[0]H_{n-1}(A).
\end{align*}
Now the image of the first zero map (which is just $\{0\}$) is the kernel of the second (which is everything), so it must follow that $H_n(U,A)=0$.
\end{proof}

\begin{cor} \label{cor:XA-XU-homology-isom}
Let $A \subset U \subset X$ be such that $U$ deformation retracts to $A$. Then
\begin{align*}
H_n(X,A) \to H_n(X,U)
\end{align*}
is an isomorphism.
\end{cor}

\begin{proof}
We have the short exact sequence of complexes
\begin{align*}
0 \to C_n(U,A) \to C_n(X,A)\to C_n(X,U)\to 0,
\end{align*}
which induces the long exact sequence
\begin{align*}
H_n(U,A) \to H_n(X,A) \to H_n(X,U)\to H_{n-1}(U,A),
\end{align*}
but \Cref{lem:HnUA-zero} says that this is
\begin{align*}
0 \to H_n(X,A) \to H_n(X,U) \to 0,
\end{align*}
hence $H_n(X,A)\arr[\sim] H_n(X,U)$.
\end{proof}

We're now prepared to prove \Cref{thm:quotient-homology-isom}.

\begin{proof}[Proof that $H_n(X,A)\cong \tilde H_n(X/A)$]
By \Cref{cor:XA-XU-homology-isom} and \Cref{thm:excision}, we have isomorphisms
\begin{align*}
H_n(X,A)\underset{\ref{cor:XA-XU-homology-isom}}{\arr[\sim]}
H_n(X,U)\underset{\ref{thm:excision}}{\arr[\sim]} H_n(X \sm A, U \sm A).
\end{align*}
Next one sees
\begin{align*}
H_n(X \sm A, U \sm A) \underset{p_*}{\arr[\sim]} H_n \paren{\frac{X}{A}\sm \frac{A}{A},\frac{U}{A}\sm \frac{A}{A}} \underset{\ref{thm:excision}}{\arr[\sim]} H_n\paren{\frac{X}{A},\frac{U}{A}},
\end{align*}
and finally
\begin{align*}
H_n\paren{\frac{X}{A},\frac{U}{A}} \underset{\ref{cor:XA-XU-homology-isom}}{\arr[\sim]} H_n \paren{\frac{X}{A}, \frac{A}{A}}.
\end{align*}
\end{proof}

We now want to remove the finite assumption from \Cref{thm:homology-isom}. To do this, we need to generalize the following claim for infinite $\Delta$-complexes.

\begin{claim}
Let $(X_\alpha,x_\alpha)$ be pointed spaces such that for all $\alpha$, there exists $U_\alpha \subset X_\alpha$ open which deformation retracts to $x_\alpha$. Then
\begin{align*}
\tilde H_n\paren{\bigvee _{\alpha \in I} (X_\alpha,x_\alpha) }= \bigoplus \tilde H_n(X_\alpha).
\end{align*}
\end{claim}

\begin{proof}
Observe
\begin{align*}
\bigoplus_{\alpha \in I} \tilde H_n(X_\alpha) &\cong \bigoplus_{\alpha \in I} H_n(X_\alpha,x_\alpha) \cong H_n\paren{\coprod_{\alpha \in I} X_\alpha, \coprod_{\alpha \in I} \{x_\alpha\} } \underset{p_*}{\arr[\sim]} H_n\paren{\frac{\coprod_{\alpha \in I}X_\alpha}{\coprod_{\alpha \in I}\{x_\alpha\}}, \{\w{pt}\}} \\
&\cong \tilde H_n\paren{\bigvee (X_\alpha,x_\alpha)}.
\end{align*}
\end{proof}

\begin{claim}
Let $X$ be a $\Delta$-complex. Then $i:X^k \inj X$ induces isomorphism on homologies up to the $k-1$ homology: that is, for all $n \le k-1$, \begin{align*}
i_{*n}:H_n(X^k) \to H_n(X)
\end{align*} 
is an isomorphism.
\end{claim}

\begin{proof}
We've seen that for every $k_2 \ge k$, the map $i_{k_2}:X^k \to X^{k_2}$ induces isomorphism on $H_n$ for $n \le k-1$. 

To see that $i_*$ is injective: if not, then there exists a cycle $z \in C_n(X^k)$ such that $[z]\ne 0\in H_n(X^k)$, but $[z]=0$ in $H_n(X)$. Hence there exists $z_{n+1} \in C_{n+1}(X)$ such that $\partial z_{n+1}=z$. Now if $z_{n+1} = \sum_{i=1}^T \alpha_i\sigma_i$, then $\Img (\sigma_i)$ is compact, and so $\Img(\sigma_i) \subset X^{k_i}$. Thus there exists $K$ such that $z_{n+1} \in C_{n+1}(X^K)$, contradicting $i_{K*}$ being injective.

Surjectivity is similar.
\end{proof}

\begin{rmk}
This claim is sufficient to prove the desired isomorphism $H_n^\Delta \cong H_n$, since
\begin{align*}
H_n^\Delta(X) \cong H_n^\Delta(X^{n+1}) \cong H_n(X^{n+1}) \cong H_n(X)
\end{align*}
\end{rmk}

\section{Mapping Degrees}

\begin{defn}
Consider $f:S^n \to S^n$. Let $\alpha$ be a generator of $H_n(S^n) \cong \Z$. Then $f_{n*}(\alpha) = k\alpha$. The \term[]{degree} of $f$ is \term[]{$\deg (f)$} $:= k$.
\end{defn}

\begin{prop}
The degree has some basic properties:
\begin{enumerate}
    \item $\deg(\id)=1$.
    \item $\deg (f \circ g) = \deg (f)\cdot \deg (g)$.
    \item If $f \sim g$ are homotopic, then $\deg (f)=\deg (g)$.
    \item If $f$ is not surjective, then $\deg(f)=0$.
    \item For each $i \in \{1,\dots,n+1\}$, consider $A_i:S^n \to S^n$ defined by \begin{align*}
    (x_1,\dots,x_i,\dots ,x_{n+1}) \maps (x_1,\dots,-x_i,\dots,x_{n+1}).
    \end{align*}
    Then $\deg (A_i) = -1$.
    \item The antipodal map $A:x \mapsto (-x)$ has degree $(-1)^{n+1}$. In particular, $A \sim \id$ if and only if $n$ is odd.
\end{enumerate}
\end{prop}

\begin{proof}
(1) and (2) are easy to show. (3) follows from \Cref{thm:homo-invar-of-hom}. (4) follows from $S^n \sm \{p\}$ being contractible.

For (5), assume WLOG that $i=n+1$. Let $U = S^n\cap \{x_{n+1}\ge 0\}$ and $L = S^n\cap \{x_{n+1}\le 0\}$. Let $\sigma_1 : \Delta^n \to U$ be a homeomorphism such that $\partial \Delta^n$ is mapped to $U \cap \{x_{n+1}=0\}$. Then take $\sigma_2 = A_{n+1}\circ\sigma_1$, and observe $\sigma_2$ is a homeomorphism of $\Delta^n \to L$.

We claim $\sigma_1 - \sigma_2$ is a generator of $H_n(S^n)$. Clearly $\partial(\sigma_1-\sigma_2)=0$ because they agree on the boundary, thus $[\sigma_1-\sigma_2] \in H_n(S^n)$. Moreover, we have
\begin{align*}
H_n(S^n) &\arr[\sim] H_n(S^n,L) \\ 
\sigma_1 -\sigma_2 &\mapsto \sigma_1
\end{align*}
but additionally
\begin{align*}
H_n(\Delta^n,\partial\Delta^n) &\arr[\sigma_1] H_n(S^n,L) \\
\id & \mapsto \sigma_1.
\end{align*}

\end{align*}
