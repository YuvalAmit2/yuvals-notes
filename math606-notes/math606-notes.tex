\section{Introduction: From Complex Analysis to Algebraic Geometry}

``Complex analysis is more rigid, even magical, when compared to real analysis."

For example, if $U \subset \C$ is open. and $f:U \to \C$ has a single complex derivative at each point $x \in U$, then $f$ is smooth (has infinitely many derivatives) and analytic (locally representable by a power series). Since power series are close to polynomials, it's not surprising that the local study of power series (analysis) and polynomials (geometry) is similar.

\begin{defn}
    A \term{complex analytic set} is defined locally as 
    $$\{x \in U \mid f_1(x) =\cdots = f_m(x) =0\},$$ 
    where $f_i: U \to \C$ are holomorphic.

    A \term{complex algebraic set} is defined in the same way, replacing ``holomorphic'' by ``polynomial''.
\end{defn}

\subsection{Classification of Real and Complex Manifolds}

Real manifolds were defined and studied as sub-manifolds of $\R^n$ long before their definitions using charts and atlases. We can think of ``embedding theorems'' as a way of seeing an abstract manifold $X$ by embedding it in coordinate space, writing down global equations for it.

\begin{thm}[Whitney's Theorem, $\sim$1935]
    If $M$ is a smooth manifold of dimension $n$, then there exists a smooth ($C^\infty$) embedding $f: M \hookrightarrow \R^{2n+1}$.
\end{thm}

Suppose $X$ is a complex manifold (more generally, a complex analytic space, which is a space locally defined by a system of holomorphic functions). Then, there exists a holomorphic embedding $f:X \hookrightarrow \C^N$ for some $N \iff X$ is a Stein manifold.

\begin{defn}
    A complex manifold $X$ is \term{Stein} if for any discrete subset $\{x_n\}_{n \geq 1}$ for points in $X$, and sequence $\{c_n\}_{n\geq1}$ of complex numbers, there exists a holomorphic function $f:X \to \C$ such that $f(x_n) =c_n$ for all $n \geq 1$.
\end{defn}

\begin{thm}
    The only compact complex manifolds that embed holomorphically in $\C^N$ are points.
\end{thm}

\begin{proof}
    We have the projection maps $z_i:X \to \C$ for $i=1,\dots,N$. Because $X$ is compact, $z_i$ is a closed map. By the open mapping theorem, if $z_i$ is not constant, then $z_i$ is open. This is impossible since $\C$ is connected, so $z_i=c_i$ is constant for all $i$. Hence, $X=\{(c_1,\dots,c_n)\}$ is a point.
\end{proof}

Since $\C^N$ can't serve as an embedding space for compact complex manifolds, we can try to use projective space $\mb P^N$ instead. On $\mb P^N = \brax{[z_0:\cdots :z_n]}$ we have homogeneous coordinates $z_i$, and analytic/algebraic subsets can be defined using homogeneous polynomials.

\begin{thm}[Chow's Theorem, 1949]
    If a complex manifold (or analytic space) embeds holomorphically as a complex submanifold of $\mb P^N$ for some $N$, then $X$ is a smooth algebraic variety. That is,
    $$X = \brax{ [z] \in \mb P^N \setbar f_1(z) =\cdots = f_m(z) = 0},$$
    where $f_1,\dots,f_m$ are polynomials in $z_0,\dots,z_n$
\end{thm}

\begin{rmk}
    There is a 1-variable analogue of this result: if $f:\C \to \C$ is an entire function such that $\lim_{z \to \infty}f(z)$ exists in $\C \cup \{\infty\}$, then $f$ is a polynomial.

    The condition on $f$ means that $f:\C \to \C$ extends to a holomorphic function $\tilde f:\mb P^1 \to \mb P^1$, where $\mb P^1 = \C \cup \{\infty\}$
\end{rmk}

Soon after, Serre proved comparison theorems for sheaves and sheaf cohomology on projective varieties.

\begin{thm}[Serre's ``GAGA'' Theorem, 1955, Géométrie Algébrique et Géométrie Analytique]
    If $\mc F$ is a coherent analytic sheaf on a projective variety $X_{\text{hol}}$, there is a unique coherent algebraic sheaf $\mc F_{\text{alg}}$ on $X_{\text{alg}}$ such that $\mc F = (\mc F_{\text{alg}})_{\text{hol}}$. Moreover, the natural maps
    $$H^{i}(X_{\text{alg}}, \mc F_{\text{alg}}) \to H^i(X_{\text{hol}},\mc F)$$
    are isomorphisms for all $i \geq 0$.
\end{thm}

\begin{defn}
    A \termnt{Riemann surface} is a 1-dimensional complex manifold $X$. Underlying $X$ is a 2-dimensional orientable real manifold $X_\R$. If $X$ is compact, then $X_\R$ is diffeomorphic to a $g$-torus, which we represent as
    $$H_1(X_\R, \Z) \simeq \Z^{2g}$$
\end{defn}

Consider an irreducible polynomial $p \in \C[z,w]$, and the subset of $\C^2$ given by
$$V(p):= \brax{(z,w) \in \C^2 \setbar p(z,w) =0}.$$
In general, $V(p) \subset \C^2$ is neither compact nor a smooth manifold. Riemann showed that $V(p)$ can be modified to obtain a compact complex manifold $X(p)$, which is the Riemann surface of the algebraic function $p(z,w)$. Moreover, $X(p)$ comes with two meromorphic functions $f,g$ which extend $z$ and $w$.

Later, Riemann defined abstract manifolds. We obtain compact Riemann surfaces $X$, which do not obviously come from polynomials $p$, but it turns out that these are all algebraic varieties. This was shown by studying the \textit{function theory} on $X$. For every compact $X$,
$$\mc O(X) := \{ f:X \to \C \text{ holomorphic}\} \simeq \C$$
is just the constant functions. Instead, we consider the field $m(X)$ of meromorphic functions on $X$ (these are just holomorphic functions $f:X \to \P^1$).

\begin{ex}
    For the projective line, $m(\P^1) \simeq \C(z)$, where $z$ is the affine coordinate on $\C \subset \P^1$.
\end{ex}

Studying meromorphic functions instead of holomorphic functions makes things more interesting. For instance, we'll prove the following theorem later, which is rather difficult.
\begin{thm}
    If $X$ is a compact Riemann surface, then there exists a non-constant meromorphic $\pi : X \to \P^1$.
\end{thm}

Consider the sequence
$$X \arr[\pi] \P^1 \arr[z]\C.$$
Pulling back $z$ via $\pi$ to $m(X)$ gives an inclusion of fields
$$\C(z) \arr[\sim] \C(\pi) \hookrightarrow \mc M(X).$$
So, $\mc M(X) \supset \C(z)$ is a field extension, which turns out to be finite. Thus, the primitive element theorem yields
$$\mc M(x) = \C(\pi)(g).$$
Let $\tilde p \in \C(\pi)[z]$ be the minimal polynomial of $g$. Then, clearing denominators, we get a polynomial $p$ with $p(\pi,g)=0$. Moreover, $X$ is the Riemann surface associated to $p$. We therefore see that $X$ is determined by $m(X)$, an ``algebraic function field in one variable''. This is the point of view in algebraic geometry, where $X$ is a smooth algebraic curve over $\C$.

\subsection{The Riemann-Roch Problem}

Given a compact Riemann surface $X$, let $f \in m(X)$ be meromorphic, and
$$D:=\div (f) = \text{``Zeros}(f)'' - ``\text{Poles}(f)''.$$
More formally,
$$D \in \bigoplus_{p \in X}\Z[p].$$

Note that we must have $\deg D=0$, where
$$\deg D = \deg \paren{\bigoplus n_p[p]}=\sum n_p.$$

In fact, $D$ determines $f$ up to scalar. Indeed, if $\div f = \div g$, then $\frac{f}{g} \in \mc O(X)$ is non-zero and holomorphic on $X$, and is therefore constant.

\begin{notethm*}[Riemann's Idea]
    Given any divisor $D$, consider the complex vector space 
    $$H^0(D) := \brax{ f \in m(x) \setbar \div f \geq -D }.$$
    Then, $\dim_\C H^0(D) < \infty$.
\end{notethm*}

\begin{notethm*}[Riemann-Roch Problem]
    Compute $\dim_\C H^0(D).$
\end{notethm*}
Riemann showed
$$\dim H^0(D) \geq 1-g+\deg D.$$
This problem in its entirety is still open, and is a motivating problem for this course.

\section{Riemann Surfaces}

\begin{defn}
Let $X$ be a 2-dimensional real topological manifold. We consider pairs $(U,\varphi)$ where $U \subset X$ is open, and $\varphi:U\to \C$ is a homeomorphism on its image. Two such pairs $(U_1,\varphi_1)$ and $(U_2,\varphi_2)$ are said to be \term{compatible} if
$$\varphi_2 \circ \varphi_1^{-1} :\varphi_1(U_1\cap U_2) \to \varphi_2(U_1\cap U_2)$$
is holomorphic. In this case, its inverse is also holomorphic.

A \term{complex structure} on $X$ is a family of pairs $\{(U_i,\varphi_i)\}_{i \in I}$ which are compatible, and $X = \bigcup _IU_i$. We typically assume such a family is maximal. The pairs $(U_i,\varphi_i)$ are called \term[holomorphic chart]{holomorphic charts}. In a coordinate neighborhood $U$, we usually identify $U$ with $\varphi(U)$, and write $z$ for the map $\varphi$ as one does with 1 complex variable in $\C$. 

A \term[Riemann surfaces]{Riemann surface} is a connected 2-dimensional real manifold with a complex structure on it. Given a Riemann surface $X$, a function $f:X \to \C$ is \term{holomorphic} if $f$ is continuous, and for every chart $(U,\varphi)$ of $X$, 
$$f \circ \varphi^{-1}: \varphi(U) \to \C$$
is holomorphic. This also defines holomorphic functions on open subsets $\Omega \subset X$. Let $\bm{\mc O(\Omega)}$ denote the $\C$-algebra of all holomorphic functions on $\Omega$. If $X,Y$ are Riemann surfaces, and $f:X \to Y$ is continuous, we say that $f$ is \term{holomorphic} if for every chart $(V,\psi)$ on $Y$, the function 
$$\psi\circ f:f^{-1}(V)\to \psi(V)$$
is holomorphic. A holomorphic map $f:X \to Y$ is an \term[analytic isomorphisms]{analytic isomorphism} (or \term[biholomorphic,biholomorphisms]{biholomorphism}) if there exists a holomorphic map $g:Y \to X$ such that $f\circ g =\id_Y$ and $g \circ f=\id_X$.
\end{defn}

\begin{rmk}
    A theorem of Rado asserts that any Riemann surface $X$ is a 2nd-countable topological space, so has a countable basis for its topology.
\end{rmk}

\begin{ex}
    Any connected open set $\Omega$ in $\C$ is a Riemann surface. Similarly, any connected open subset $\Omega$ of a Riemann surface $X$ is a Riemann surface.
\end{ex}

\begin{ex}[The Riemann sphere $\P^1$]
    The \term{Riemann sphere} $\P^1=\C \cup \{\infty\}$ is the one-point compactification of $\C$, where the open neighborhoods of $\infty$ are complements of compact subsets $K \subset \C$. With this topology, $\P^1$ is homeomorphic to the 2-sphere $S^2 \subset \R^3$. Set $U_1 := \P^1\setminus \{\infty \} = \C$ and $U_2:=\P_1\setminus \{0\}=\C^*\cup \{\infty\}$, and define the charts $\varphi_i:U_i\to \C$ by $\varphi_1=\id_\C$, and
    $$\varphi_2(z) = \case{1/z & z \in \C^*, \\ 0 & z=\infty.}$$
    Then,
    $$\varphi_2\circ \varphi_1^{-1}:\varphi_1(U_1\cap U_2) = \C^* \to \C^* = \varphi_2(U_1 \cap U_2)$$
    sends $z \mapsto 1/z$, and is a biholomorphic map on $\C^*$.
\end{ex}

\begin{ex}[Complex tori]
    Let $\omega_1$ and $\omega_2$ be non-zero complex numbers, which are linearly independent over $\R$. Define $\Lambda =\Z\omega_1 + \Z\omega_2$ to be a \term{lattice} in $\C$, and let $\pi:\C \surj \C/\Lambda$ be the canonical projection, giving $\C/\Lambda$ the quotient topology. Since $\pi$ is a covering map, $\C/\Lambda$ inherits a complex structure from the complex structure on $\C$. Note that $\C/\Lambda$ is compact, and is homeomorphic to the real $2$-torus $S^1 \times S^1$.

    More explicitly, let $\C = \bigcup V_i$ be an open cover of $\C$ by disks such that $\pi|_{V_i}$ is a homeomorphism onto $U_i := \pi (V_i)\subset \C/\Lambda$. Let $\varphi_i:U_i \to V_i \subset \C$ be the inverse $\paren{\pi|_{V_i}}^{-1}$. Then, the family $\{(U_i,\varphi_i)\}$ is a complex structure on $X := \C/\Lambda$. Indeed, let $x_0 \in U_i \cap U_j$, $z_i:=\varphi_i(x_0)$, and $z_j := \varphi_j(x_0)$. Then, $\lambda:=z_i-z_j \in \Lambda$, and
    $$\varphi_i \circ \varphi_j^{-1}: \varphi_j(U_i \cap U_j) \to \varphi_i(U_i \cap U_j)$$
    sending $z \mapsto z + \lambda$ is biholomorphic.
\end{ex}

\begin{thm}[Riemann Extension Theorem][label=thm:riemann extension]
    Let $X$ be a Riemann surface, and $a \in X$. Let $f \in \mc O(X\setminus \{a\})$ be such that $f$ is bounded on $U \setminus \{a\}$ for some neighborhood $U$ of $a$. Then, there exists a function $F \in \mc O(X)$ such that $F|_{X\setminus\{a\}}=f$. 
\end{thm}

\begin{thm}[Principle of Analytic Continuation]
    Let $X$ and $Y$ be Riemann surfaces, and $f : X \to Y$ be holomorphic. If there is a non-empty open set $U_0 \subset C$ such that $f|_{U_0}$ is a constant $y_0 \in Y$, then $f(x) =y_0$ for all $x \in X$.
\end{thm}

\begin{proof}
    Let 
    $$E:= \brax{x \in X \setbar \exists U\ni x \text{ open, } f(U) =\{y_0\} }.$$
    Then, $E$ is open and $U_0 \subset E$. Let $x_0 \in \bar E$ and $(U,\varphi),(V,\psi)$ be charts on $X$ and $Y$ respectively with $x_0 \in U$ and $f(x_0) \in V$. Let $\Omega$ be the connected component of $\varphi\paren{U \cap f^{-1}(V)}$ containing $\varphi(x_0)$. Now, $\varphi(E \cap U) \cap \Omega$ is a non-empty open set. Moreover, $F:= \psi \circ f \circ \varphi^{-1}|_\Omega$ is holomorphic on $\Omega$, and 
    $$F(\varphi(E \cap U) \cap \Omega) = \{\psi(y_0)\}.$$
    Thus, by analytic continuation on $\C$, $F(\Omega) = \{\psi(y_0)\}$, hence $f(x)=y_0$ for all $x$ in a neighborhood of $x_0$. Therefore, $x_0 \in E$, so $E$ is closed. Since $X$ is connected, and $E$ is both open and closed, we must have $E=X$.
\end{proof}

\begin{exercise}
    Show that the principle of analytic continuation stated above is equivalent to the formulation involving accumulation points in $X$.
\end{exercise}

\begin{defn}
    A \termnt{meromorphic function} on a Riemann surface $X$ is a holomorphic function $f:X'\to \C$, where $X'\subset X$ is an open subset such that
    \begin{enumerate}[(i)]
        \item $X \setminus X'$ is discrete in $X$,
        \item For any point $p \in X'$, 
        $$\lim_{x \to p}|f(x)| = \infty$$
    \end{enumerate}
    The points of $X\setminus X'$ are called the \term[pole]{poles} of $f$. Let $\mc M(X)$ denote the $\C$-algebra of all meromorphic functions on $X$. We'll soon show that $\mc M(x)$ is a field.
\end{defn}

\begin{rmk}
    If $(U,z)$ is a coordinate neighborhood of a pole $p$ of $f$, with $z(p)=0$, then $f$ has a Laurent expansion 
    $$f(z) = \sum_{v=-k}^{+\infty} c_vz^v$$
    in a neighborhood of $p$, with $c_{-k} \neq 0$. The integer $-k:=\ord_p(f)$ is well-defined, and independent of the choice of local coordinates around $p$.
\end{rmk}

Our definition of meromorphic functions does not work in higher dimensions, since isolated zeros/poles don't exist in complex dimension $n\geq 2$. The following is a better, more generalized definition:
\begin{defn}
    A \term[meromorphic functions]{meromorphic function} $f$ is one that locally can be written as $\frac{g}{h}$, where $g$ and $h$ are holomorphic functions.
\end{defn}

\begin{thm}[][label=thm:mer_is_hol]
    Let $X$ be a Riemann surface, and $f \in \mc M(X)$. For each pole $p$ of $f$, let $f(p):= \infty$. Then, the extended function
    $$f:X \to \P^1$$
    is holomorphic. Conversely, if $f:X \to \P^1$ is holomorphic, either $f$ is identically $\infty$, or $f^{-1}(\infty)$ consists of isolated points in $X$, and
    $$f:X\setminus f^{-1}(\infty) \to \C$$
    is a meromorphic function on $X$. Hence, we can identify $f \in \mc M(X)$ with the corresponding holomorphic map $f:X \to \P^1$
\end{thm}

\begin{proof}
\hfill \break
    $(\implies)$ Let $f \in \mc M(X)$ be meromorphic, and let $P$ be the set of poles of $f$. Then, the induced map $f:X \to \mb P^1$ is continuous. Suppose $\varphi:U \to V$ and $\psi:U'\to V'$ are holomorphic charts on $X$ and $\mb P^1$ with $f(U) \subset U'$. We must show that $g:= \psi \circ f \circ \varphi^{-1}:V \to V'$ is holomorphic. Indeed, since $f \in \mc O(X \setminus P)$, we get $g \in \mc O(V\setminus \varphi(P))$. By the Riemann Extension Theorem (\Cref{thm:riemann extension}), $g$ is holomorphic on all of $V$.

    $(\impliedby)$ Easy consequence of analytic continuation.
\end{proof}

\begin{cor}
    Analytic continuation also holds for meromorphic functions on a Riemann surface $X$. So, any $f \in \mc M(X)$ which is not identically zero has isolated zeros on $X$. This implies that $1/f\in \mc M(x)$, thus $\mc M(X)$ is a field.
\end{cor}

\begin{thm}[Local Behavior of Holomorphic Maps][label=thm:local_behav]
    Suppose $X,Y$ are Riemann surfaces, and $f:X \to Y$ is a non-constant holomorphic map. Let $a \in X$ and $b=f(a)$. Then, there exists $k \in \Z^+$ and charts $\varphi:U \to V$ on $X$, $\psi:U' \to V'$ on $Y$, such that
    \begin{enumerate}[(i)]
        \item $a \in U, \varphi(a) =0, b \in U', \psi(b)=0, f(U) \subset U'$.
        \item $F:= \psi \circ f \circ \varphi^{-1}:V \to V'$ is $F(z) =z^k$ for all $z \in V$.
    \end{enumerate}
\end{thm}

Hence, the question is local, and it suffices to show that if $f: \Omega \to \C$ is holomorphic and non-constant in a neighborhood $\Omega$ of $0 \in \C$, then there exists a neighborhood $U$ of $0$ and $\psi : U \to U'$ biholomorphic such that $f \circ \psi(z) = z^k$ for some $k \geq 1$.

    Since $f$ is holomorphic and non-constant, we may write $f(z) = z^k g(z)$ for some $k \geq 1$, and $g \in \mc O(0)$ with $g(0) \neq 0$. We claim that there exists a neighborhood $U''$ of $0$ such that $g(z)=h(z)^k$ for some $h\in \mc O(U'')$, with $h(0) \neq 0$. Indeed there exists $\tilde g$ such that

\begin{rmk}
    The integer $k$ may be characterized as follows: for any neighborhood $U_0$ of $a$, there exists neighborhoods $U \subset U_0$ of $a$ and $W$ of $b=f(a)$ such that $\#f^{-1}(y) \cap U = k$ for all $y \in W$, $y \neq b$. This $k$ is the \term{multiplicity} of $f$ at $k$.
    \end{rmk}

\begin{notethm*}[Notation]
    If $K \subset \C$ is compact, we write $\mc O(K)$ to mean functions $f:K \to \C$ which are holomorphic in an open neighborhood of $K$.
\end{notethm*}

\begin{cor}[][label=cor:openhol]
    Let $f:X \to Y$ be a non-constant holomorphic map. Then, $f$ is an open map.
\end{cor}

\begin{proof}
    By \Cref{thm:local_behav}, it is clear that if $U$ is a neighborhood of $a \in X$, then $f(U)$ is a neighborhood of $f(a)$.
\end{proof}

\begin{thm}
    Suppose that $X$ and $Y$ are Riemann surfaces with $X$ compact and $Y$ connected, and $f:X \to Y$ is a non-constant holomorphic map. Then, $f$ is surjective, and $Y$ is compact.
\end{thm}

\begin{proof}
    By \Cref{cor:openhol}, $f$ is an open map, hence $f(X)$ is both an open and closed subset of $Y$ which is connected. Hence, $f(X) =Y$, and $Y$ is compact. 
\end{proof}

\begin{cor}[][label=cor:compact_hol_constant]
    If $X$ is a compact Riemann surface, and $f:X \to \C$ is holomorphic, then $X$ is constant.
\end{cor}

\begin{cor}[Maximum Principle]
    Let $X$ be a Riemann surface, and $f:X \to \C$ a non-constant holomorphic map. Then, $|f|$ does not attain a maximum.
\end{cor}

\begin{proof}
    As $f$ is non-constant, $f(X)$ is an open subset of $\C$, so if $f(X) \subset \bar D(x_0,R)$ for some $x_0 \in \C,R>0,$ then $f(X) \subset D(x_0,R)$. So, the complex maximum principle applies.
\end{proof}

\begin{cor}
    Every meromorphic function on $\P^1$ is rational, that is, can be written as a quotient of two polynomials.
\end{cor}

\begin{proof}
    If $f \in \mc M(\P^1)$, then $f$ induces a holomorphic map $\tilde f:\P^1 \to \P^1$ by \Cref{thm:mer_is_hol}, so $f$ has finitely many poles. We may assume $\infty$ is not a pole of $f$, otherwise consider $1/f$ instead of $f$.

    Let $a_1,\dots,a_n \in \C$ be the poles of $f$, and 
    $$h_\nu(z) := \sum_{j=-k_\nu}^{-1}c_{\nu_j}(z-a_\nu)^j$$
    be the principle part of $f$ at the pole $a_\nu$, for $\nu=1,2,\dots,n$. Then,
    $$f-h_1-h_2 - \cdots - h_n$$
    is holomorphic on $\P^1$, and is therefore a constant.
\end{proof}

We've shown that $\mc M(\P^1) \simeq \C(z)$, the field of rational functions in $z$.

\end{exercise}

\section{Covering Maps}
In this section, $X$ and $Y$ are topological manifolds.

\begin{defn}
    Let $X$ and $Y$ be topological manifolds (more generally, locally compact Hausdorff spaces). A continuous map $f:X \to Y$ is \term{proper} if $f^{-1}(K)$ is compact for any compact $K \subset Y$. We say that $f$ is a \term[local homeomorphisms]{local homeomorphism} if for all $x \in X$, there exists an open neighborhood $U \ni x$ such that $f(U)$ is open in $Y$, and $f|_U:U \to f(U)$ is a homeomorphism. We say that $f$ is a \term[covering maps]{covering map} if for all $y \in Y$, there exists a neighborhood $V \ni y$ which is \term{evenly covered} by $f$, meaning 
    $$f^{-1}(V) = \bigsqcup_{j \in J}U_j$$
    is a disjoint union of open sets $U_j$ such that $f|_{U_j}:U_j \to V$ is a homeomorphism onto $V$, for all $j \in J$. We say $X$ is a \term[coverings]{covering} of $Y$ if a covering map exists.
\end{defn}

\begin{rmk}
    \begin{enumerate}
        \item Since the cardinality $|J|$ of $f^{-1}(y)$ is a locally constant function on $Y$ for any covering map $f$, if $Y$ is connected then $\# f^{-1}(y)$ is independent of $y \in Y$. The covering is said to be \term{finite} (resp. \term{infinite}) if $f^{-1}(y)$ is constant (resp. infinite). The covered is said to be an \term[sheeted cover,sheeted covering]{n-sheeted covering} if $\#f^{-1}(y) = n$.
        \item If $f:X \to Y$ and $\tilde f:\tilde X \to Y$ are two coverings of $Y$, they are said to be \term{isomorphic coverings} if there exists a homeomorphism $\varphi :\tilde X \to X$ such that $f \circ \varphi = \tilde \varphi$. That is, the diagram

\end{rmk}

\begin{ex}
    Any $n$-sheeted covering of $\mb D^*:= \brax{z \in \C \setbar 0 < \abs{z} < 1}$ is isomorphic to $f_n(z):=z^n$.
\end{ex}

\begin{lem}[][label=lem:proper_closed]
    If $f:X \to Y$ is proper, then $f$ is closed.
\end{lem}

\begin{proof}
    If $F \subset X$ is closed, and $K \subset Y$ is compact, it suffices to show $f(F) \cap K$ is closed. Indeed, $f(F) \cap K = f(F \cap f^{-1}(K))$, and since $f$ is proper we know $f^{-1}(K)$ is compact, hence $F \cap f^{-1}(K)$ is compact. Thus, the image of $F \cap f^{-1}(K)$ is closed in $K$.
\end{proof}

\begin{prop}
    Suppose $p:X \to Y$ is a local homeomorphism with $Y$ connected. Then, $p$ is a finite covering map if and only if $p$ is proper.
\end{prop}

\begin{proof}
    \hfill \break
    ($\implies$) If $p$ is a finite covering, $y_0 \in Y$, and $V\supset y_0$ evenly covered, then $p|_{p^{-1}(V)}:p^{-1}(V) \to V$ is clearly proper, since $p$ is a finite cover. In general, if $K$ is any compact set, then $K$ can be covered by finitely many compact neighborhoods, each evenly covered by $p$, so it follows that $p$ is proper.  

    ($\impliedby$) Let $p$ be a proper local homeomorphism. Then, $p$ is closed by \Cref{lem:proper_closed}, so $p(X)$ is both open and closed in $Y$, which is connected, hence $p$ is surjective. Now, let $y_0 \in Y$ and $p^{-1}(y_0)=\{x_1,\dots,x_n\}$. Let $U_j'$ be a neighborhood of $x_j$ such that $p|_{U_j'}:U_j' \to p(U_j')=:V_j$ is a homeomorphism, and the $U_j'$ are disjoint.

    We want to find our evenly covered neighborhood. If $X \setminus \bigcup U_j'$ is closed in $X$, and $p$ is proper, then $F:=p(X \setminus \bigcup U_j')$ is closed in $Y$, and $y_0 \not \in F$. Hence, $V':=Y\setminus F$ is an open neighborhood of $y_0$, and $p^{-1}(V') \subset \bigsqcup U_j'$. Define $V:= V' \cap V_1 \cap \cdots \cap V_n$. Then, $V$ is our desired neighborhood: it is an open neighborhood such that $p^{-1}(V) \subset U_1' \cup \cdots \cup U_n'$. For $1 \leq i \leq n$, define
    $$U_i:= p^{-1}(V) \cap U_u' = \paren{p|_{U_i'}}^{-1}(V).$$
    Then, $U_i$ is mapped homeomorphically onto $V$ by $p$, and $U_1,\dots,U_n$ are disjoint. So, $V$ is evenly covered by $p$.
\end{proof}

\section{Branched and Unbranched Coverings}

\begin{defn}
    Let $f:X \to Y$ be a non-constant holomorphic map of Riemann surfaces. A point $x \in X$ is called a \term[branch points]{branch point} or \term[ramification points]{ramification point} of $f$ if there is no neighborhood $U$ of $x$ such that $f|_U$ is injective. We say that $f$ is an \term{unbranched holomorphic map} if it has no branch points. This happens if and only if $f$ is a local homeomorphism. 
\end{defn}

\begin{rmk}
    The set of branch points of $f$ in $X$ is closed and discrete in $X$.
\end{rmk}

\begin{defn}
Assume $f:X \to Y$ is a proper map, and let $A$ be the set of all branch points of $f$. Then, the set $B:= f(A)$ is also closed and discrete. $B$ is called the set of \term[critical value]{critical values} of $f$. 
\end{defn}

Let $Y' :=Y \setminus B$ and $X' := X \setminus f^{-1}(B) \subset X \setminus A$. Then, $f|_X : X' \to Y'$ is a proper unbranched holomorphic covering map, with a well-defined finite number of sheets in $n$ (that is, every $c \in Y'$ is achieved exactly $n$ times). We will extend this statement to include the critical values $b \in B$ by considering multiplicities. For $x \in X$, let $v(f,x)$ be the multiplicity with which $f$ takes the value $f(x)$ at the point $x$. We say that $f$ takes the value $c \in Y$, counting multiplicities, $m$ times on $X$, if
$$m := \sum_{x \in f^{-1}(c)}v(f,x)$$

\begin{thm}[][label=thm:sheet_counting]
    Suppose that $X$ and $Y$ are two Riemann surfaces, and $f:X \to Y$ is a proper, non-constant, holomorphic map. Then, there exists $n \in \Z^+$ such that $f$ takes every value $c \in Y$, counting multiplicities, $n$ times.
\end{thm}

\begin{proof}
    Let $n$ be the number of sheets of the unbranched cover $f:X' \to Y'$ defined earlier. Suppose $b \in B$ is a critical value, $f^{-1}(b) = \{x_1,\dots,x_r\}$, and let $k_j:= v(f,x_j)$ for $j=1,\dots,r$. Then, there exists neighborhood $U_j$ of $x_j$, and $V_j$ of $b$, such that for all $c \in V_j\setminus \{b\}$, the set $f^{-1}(c) \cap U_j$ has cardinality $k_j$ for $1 \leq j \leq r$. 
    \begin{claim}
        We can find a neighborhood $V \subset V_1 \cap \cdots \cap V_r$ of $b$ such that $f^{-1}(V) \subset U_1 \cap \cdots \cap U_r$.
    \end{claim}
    \begin{proof}
        Indeed, $f$ is proper, hence closed, so $f(X \setminus (U_1 \cup \cdots \cup U_r))$ is a closed set in $Y$ not containing $b$. So, take $V:=V'\cap V_1 \cap \cdots V_n$, where $V' :=Y\setminus f(X\setminus(U_1 \cup \cdots \cup U_r)).$
    \end{proof}
    Now, for every $c \in V \cap Y'$, $f^{-1}(c)$ consists of $k_1 + \cdots + k_r$ points, and this number is exactly $n$ since $\#f^{-1}(c)=n$ for all $c \in Y'$.
\end{proof}


\begin{cor}
    Let $X$ be a compact Riemann surface. Then, every non-constant meromorphic function $f:X \to \P^1$ has as many zeros as poles, where each is counted with multiplicities.
\end{cor}

\begin{proof}
    $f$ is proper, so \Cref{thm:sheet_counting} applies.
\end{proof}

The main difficulty is that if $X$ is a compact Riemann surface, we can construct any non-constant $f \in \mc M(X)$. We examine the case of $\mc M(\C\setminus \Lambda)$ where $\Lambda \subset \C$ is a lattice. These are elliptic functions.

\section{Elliptic Functions}

\begin{defn}
    Let $\Omega:= \Z\omega_1  +\Z\omega_2$ be a lattice in $\C$. A meromorphic function $f:\C \to \P^1$ is \term{elliptic} with respect to $\Omega$ if $f(z+\omega)=f(z)$ for all $\omega \in \Omega$. Such a function corresponds to a holomorphic map $f:T \to \P^1$, where $T:=\C/\Omega$.
\end{defn}

\begin{defn}
    The \term{order} $\ord(f)$ of an elliptic function $f$ is the number of solutions $f(z) = \infty$, i.e. the number of poles of $f$ on $T$, counted with multiplicity.
\end{defn}

\begin{rmk}
    An elliptic function $f$ is constant $\iff \ord(f)=0$.
\end{rmk}

Assume that $P$ is a fundamental parallelogram for $\Omega$ with vertices at $t,t+\omega_1,t+\omega_1+\omega_2$, and $t+\omega_2$, where $t \in \C$ is such that $\partial P$ has no zeros or poles of $f$. In the homework, we showed
$$\int_{\partial P}f(z)dz =0,$$
so by the residue theorem, the sum of the residues of $f$ inside $P$ is $0$.

\begin{cor}
    There are no elliptic functions of order $N=1$.
\end{cor}

\begin{cor}
    If an elliptic function has order $N>0$, then $f$ takes every value $c \in \P^1$ exactly $N$ times on $P$, counting multiplicities.
\end{cor}

\begin{cor}[][label=cor:f_eq_lamg]
    Let $f,g$ be elliptic functions with respect to $\Omega$, with zeroes and poles of the same orders, and at the same points. Then, $f(z) = \lambda g(z)$ for some $\lambda \in \C^*$.
\end{cor}

\begin{proof}
    The map $f/g:T \to \P^1$ is holomorphic on compact Riemann surfaces, hence constant.
\end{proof}

\begin{thm}
    Let the congruence classes mod $\Omega$ of zeros and poles of an elliptic function $f$ be $[a_1],\dots,[a_r]$ and $[b_1],\dots,[b_s]$, with multiplicities $k_1,\dots,k_r$ and $\ell_1,\dots,\ell_s$ respectively. Then,
    $$\sum_{j=1}^rk_ja_j \equiv \sum_{j=1}^s \ell_jb_j \pmod \Omega.$$
\end{thm}
This means we cannot just choose any set of zeros/poles, and assume an elliptic function has them.

\section{\texorpdfstring{The Weierstrass $\wp$-function}{The Weierstrass p-function}}

\begin{lem}
    If $s \in \R$, then
    $$\sum_{\omega \in \Omega\setminus\{0\}}\frac{1}{|\omega|^s} \text{ converges} \iff s>2$$
\end{lem}

\begin{proof}
    Let 
    $$P_r:=\{a\omega_1+b\omega_2 \mid a,b\in \R, \max \{|a|,|b|\} =r\}$$ 
    for $r \in \Z^+$, and $\Omega_r:= \Omega \cap P_r$. Then,
    $$\Omega =\{0\} \sqcup \bigsqcup_{r \geq 1}\Omega_r,$$
    and $|\Omega_r| = 8r$. Let $m$ and $M$ be the smallest and largest moduli of all elements of $P_1$. Then, since $\Omega_r \subset P_r = r\cdot P_1$, we have
    $$rM \ge |\omega| \geq rm, \quad \forall \omega \in \Omega_r.$$
    Thus,
    $$\sigma _r := \sum_{\omega \in \Omega_r}\frac{1}{|\omega|^s}\implies 8r(rM)^{-s}\leq \sigma_r \leq 8r(rm)^{-s},$$
    thus
    $$\sum_{r=1}^\infty \sigma_r < \infty \iff \sum_{r=1}^\infty r^{1-s}<\infty \iff s>2.$$
\end{proof}

\begin{cor}
    For each integer $N \geq 3$, the function 
    $$F_N(z):= \sum_{\omega \in \Omega}\frac{1}{(z-\omega)^N}$$
    is elliptic of order $N$.
\end{cor}

\begin{proof}
    Let $K$ be a compact subset of $\C \setminus \Omega$. Then, for all but finitely many $\omega \in \Omega$, say for all $\Omega'\subset \Omega$, we have $|\omega| \geq 2z$ for all $z \in K$. Hence,
    $$|z-\omega| \geq |\omega| - |z| \geq \frac{1}{2}|\omega|, \quad \forall z \in K,\forall \omega \in \Omega'.$$
    Thus,
    $$\norm{\frac{1}{(z-\omega)^N}}_k \leq 2^N \frac{1}{|\omega|^N}, \quad \forall \omega \in \Omega'.$$
    Hence, if $N \geq 3$, then by the previous lemma,
    $$\sum_{\omega \in \Omega}\norm{\frac{1}{(z-\omega)^N}}_k \text{ converges,}$$
    so $F_N(z)$ converges to a meromorphic function with a pole of order $N$ at each $\omega \in \Omega$, while $F_N|_{\C\setminus \Omega}$ is holomorphic. Since the series converges absolutely, it can be rearrared, so if $\omega_0 \in \Omega$,
    $$F_n(z+\omega_0) = \sum_{\omega \in \Omega}\frac{1}{(z+\omega_0-\omega)^N}=\sum_{\omega' \in \Omega}\frac{1}{(z-\omega')^N}=F_N(z)$$
    and we're done.
\end{proof}

\begin{defn}
    The \term[\wp]{Weierstrass $\wp$-function} for a lattice $\Omega$ is
    $$\wp(z) := \frac{1}{z^2} + \sum_{\omega \in \Omega^*}\paren{\frac{1}{(z-\omega)^2}-\frac{1}{\omega^2}}.$$
    For $z$ bounded and staying away from the lattice points, $1/(z-\omega)^2 - 1/\omega^2$ has order of magnitude $1/|\omega^3|$. Hence, $p(z)$ is meromorphic with a pole of order $2$ at each lattice point.
\end{defn}

\begin{thm}
    $\wp$ is periodic, hence an elliptic function.
\end{thm}

\begin{proof}
    First, note that $\wp$ is an even function: $\wp(-z)=\wp(z)$, and $\wp'$ is odd. In particular,
    $$\wp'(z) = -2\sum_{\omega \in \Omega}\frac{1}{(z-\omega)^3},$$
    which we know is an elliptic function. From the periodicity, we obtain $c \in \C$ such that
    $$\wp(z+\omega_1) - \wp(z) =c.$$
    Let $z:=-\omega_1/2$. Then,
    $$\wp(\omega_1/2) = \wp(-\omega_1/2) + c,$$
    thus $c=0$ because $p$ is even. Similarly $\wp(z+\omega_2)=\wp(z)$, so $\wp$ is elliptic.
\end{proof}

Let $\mc M(\Omega) = \mc M(T)$ denote the field of all elliptic functions with respect to $\Omega$. Our goal is to classify $\mc M(\Omega)$. We will show $\mc M(\Omega) =\C (\wp, \wp')$.

\begin{thm}
    If $P$ is a fundamental parallelogram with $0,\omega_1/2,\omega_2/2,$ and $\omega_3/2:=(\omega_1+\omega_2)/2$ in its interior, then
    $\omega_1/2,\omega_2/2,\omega_3/2$ are the zeros of $\wp'$ in $P$.
\end{thm}

\begin{proof}
    Note that $\wp'(z)$ has order $3$, hence $\wp'$ has three zeros in $P$. If $\omega \in \Omega,$ we have
    $$\wp '(\omega/2) = \wp'(-\omega/2) = -\wp'(\omega/2)$$
    since $\omega/2 \equiv -\omega/2 \pmod \Omega$. Therefore, $\wp'(\omega/2)$ equals $0$ or $\infty$. Since the only pole of $\wp'$ is the triple pole at $0$, we deduce that $\wp'(\omega/2)=0$ for $\omega =\omega_1,\omega_2,\omega_3$.
\end{proof}


\begin{rmk}
  Let $e_i := \wp (\omega_i/2)$ for $i=1,2,3$. Since 
  $$S := \braks{\frac{\omega_1}{2}} \cup \braks{\frac{\omega_2}{2}} \cup \braks{\frac{\omega_3}{2}}$$
  is the set of all zeros of $\wp'$ on $\C$, we see that
  $$\{e_1,e_2,e_3\} = \wp(S)$$
  is independent of the choice of basis $\omega_1,\omega_2$ for $\Omega$.
\end{rmk}

\begin{cor}[][label=cor:solutions_of_p]
    For each point $c \in \P^1 \setminus \{e_1,e_2,e_3,\infty\}$, the equation $\wp(z)=c$ has 2 simple solutions, and for $c \in \{e_1,e_2,e_3,\infty\}$, the equation $\wp (z) = c$ has a double solution.
\end{cor}

\begin{proof}
    This is clear, since $e_1,e_2,e_3,\infty$ are the critical values of the branched cover $\wp:T \to \P^1$ given by $\wp$, while $0,\omega_1/2,\omega_2/2,\omega_3/2$ are the branch points. 
\end{proof}

\begin{thm}
    If $f$ is an even elliptic function, then $f=R_1(\wp )$ for some rational function $R_1 \in \C(z)$. Thus,
    $$\mc M^{\text{even}}(\Omega) = \C(\wp).$$
    Moreover, if $f$ is any elliptic function, then
    $$f=R_1(\wp) + \wp' R_2(\wp),$$
    where $R_1,R_2 \in \C(z)$. Hence,
    $$\mc M(\Omega) = \C(\wp,\wp').$$
\end{thm}

\begin{proof}
    We begin with the even elliptic functions. Let $f$ be a non-constant even elliptic function of order $N>0$. If $k \in \C$, the equation $f(z)=k$ has multiple roots where $f'(z)=0$, and this occurs at only finitely many points of $P$. Hence, $f(z)=k$ has only simple roots for all but finitely many values of $k$. Therefore, we can choose $c,d \in \C$ with $c \neq d$ such that
    \begin{enumerate}
        \item[(a)] the roots of $f(z)=c$ and $f(z)=d$ are all simple,
        \item[(b)] none of these roots is congruent to $0$ or $\omega_i/2$ modulo $\Omega$, $i=1,2,3$.
    \end{enumerate}
    Since $f$ is even, a complete set of solutions to $f(z)=c$ looks like $A:=\{a_1,-a_1,\dots,a_n,a_n\}$, these being simple and distinct modulo $\Omega$. We say the same for the roots $B:=\{b_1,-b_1,\dots,b_n,-b_n\}$ of $f(z)=d$. Therefore,
    $$g(z):=\frac{f(z)-c}{f(z)-d}$$
    has simple zeros at $A$, and simple poles at $B$. By \Cref{cor:solutions_of_p}, the equation $\wp(z) = \wp(a_i)$ and $\wp(z)=\wp(b_i)$ have simple roots at $z = \pm a_i$ and $z = \pm b_i$ respectively, for $1 \leq i \leq n$. Hence,
    $$h(z):= \frac{\prod_{i=1}^n(\wp(z) - \wp(a_i))}{\prod_{i=1}^n(\wp(z)-\wp(b_i))}$$
    has the same zeros and poles as $g(z)$ all of them simple. Therefore, by \Cref{cor:f_eq_lamg}, $g=\lambda h$ for some $\lambda \in \C^*$. Now, solve
    $$\frac{f(z)-c}{f(z)-d}=\lambda \frac{\prod_{i=1}^n(\wp(z)-\wp(a_i))}{\prod_{i=1}^n(\wp(z)-\wp(b_i))}$$
    for $f(z)$, and we get $f(z) \in \C(\wp)$.

    Now the general case. If $f$ is odd, the $f/\wp'$ is even, so $f=\wp'R_2(\wp)$ by the even case. In general, for any $f \in \mc M(\Omega)$, we may write
    $$f(z) = \frac{f(z) + f(-z)}{2}+\frac{f(z)-f(-z)}{2},$$
    and the two terms are even and odd respectively. This concludes the proof.
\end{proof}

We've shown that $\mc M(\Omega) = \C(\wp, \wp')$. This is not to say that $\wp$ and $\wp'$ are algebraically independent: this would be absurd anyway. We now want to know understand their relationship.

\section{\texorpdfstring{The Differential Equation for $\wp(z)$}{The Differential Equation for p(z)}}

\begin{defn}
    We define the \term[\zeta]{Weierstrass $\zeta$-function}
    \begin{align} \label{eq:weierstrass_zeta}
        \zeta(z) := \frac{1}{z}+\sum_{\omega \in \Omega^*}\paren{\frac{1}{z-\omega}+\frac{1}{\omega}+\frac{z}{\omega^2}}.
    \end{align}
\end{defn}
    Note that
    $$\frac{1}{z-\omega}+\frac{1}{\omega}+\frac{z}{\omega^2}=\frac{z^2}{\omega^2(z-\omega)},$$
    so by comparison with 
    $$\sum_{\omega \in \Omega^*}\frac{1}{|\omega|^3}$$
    we see that \eqref{eq:weierstrass_zeta} converges absolutely for $z \in \C \setminus \Omega$.

    Let $D:=D(0,R)$ be the largest open disc centered at $0$ and containing no other lattice points. For $\omega \in \Omega^*$, the series
    $$\frac{1}{z-\omega}=-\frac{1}{\omega}-\frac{z}{\omega^2}-\frac{z^2}{\omega^3}-\cdots$$
    converges absolutely for $z \in D$. We can substitute this in \eqref{eq:weierstrass_zeta} and reverse the order of summation to get
    $$\zeta(z) = \frac{1}{z} +\sum_{\omega \in \Omega^*}\paren{ -\frac{z^2}{\omega^3} - \frac{z^3}{\omega^4}-\cdots }= \frac{1}{z}-G_3z^2 - G_4z^3-\cdots $$
    for all $z \in D$, where
    $$G_k:=G_k(\Omega):= \sum_{\omega \in \Omega^*}\frac{1}{\omega^k}$$
    are the \term{Eisenstein} series for $\Omega$.

    For $k$ odd, the terms $1/\omega^k$ and $1/(-\omega)^k$ cancel, so $G_k=0$. Therefore, the Laurent series for $\zeta(z)$ at $0$ is
    $$\zeta(z) = \frac{1}{z}-\sum_{n=2}^\infty G_{2n}z^{2n-1}$$
    for $z \in D$. Expanding, we get
    $$\wp(z) = -\zeta'(z) = \frac{1}{z^2} + 3G_4z^2 + 5G_6z^4 + \cdots,$$
    hence
    \begin{align*}
        \wp'(z) &= -\frac{2}{z^3}+6G_4z + 20 G_6z^3 + \cdots

        \\ 4\wp(z)^3 &= \frac{4}{z^6}+\frac{36 G_4}{z^2}+60G_6 + z^2 \varphi_2(z) \\ 60 G_4\wp(z) &= \frac{60G_4}{z^2} + z^2 \varphi_3(z)
    \end{align*}
    for $\varphi_1,\varphi_2,\varphi_3 \in \mc O(0)$. We deduce that
    \begin{align} \label{eq:weirdiffeq}
        \wp'(z)^2 - 4\wp(z)^3 + 60 G_4 \wp(z) + 140G_6 = z^2 \varphi(z),
    \end{align}
    where $\varphi(z) = \varphi_1(z) - \varphi_2(z) + \varphi_3(z)$ converges in $D$. Since $\wp',\wp \in \mc M (\Omega),$ the LHS in \eqref{eq:weirdiffeq} is elliptic, and equals $z^2 \varphi(z)$ on $D$. Hence, the LHS has no poles, so is constant. This constant is zero since $z^2 \varphi(z)|_{z=0}=0$.
    \begin{thm}
        The Weierstrass $\wp$ function satisfies the differential equation
        \begin{align*}
            \wp'(z)^2 = 4\wp(z)^3 -60G_4\wp(z) - 140G_6.
        \end{align*}
        Then, writing $g_2:=60G_4=60\sum_{\omega \neq 0}\frac{1}{\omega^4}$, and $g_3:= 140G_6=140\sum_{\omega \neq 0}\frac{1}{\omega^6}$, we have
        \begin{align*}
            \wp'(z)^2 = 4\wp(z)^3 - g_2\wp(z) - g_3.
        \end{align*}
        Thus, the points $(\wp(z),\wp'(z))$ lie in the algebraic curve with equation
        $$y^2 = 4x^3 - g_2x -g_3,$$
        called an \term{elliptic curve}. The RHS is a cubic with discriminant $\Delta = 16(g_2^3-27g_3^2)$.
    \end{thm}

\section{\texorpdfstring{Weierstrass' $\sigma$-Function}{Weierstrass' Sigma Function}}

\subsection{Background on Infinite Products}

If $\{b_n\}_{n \geq 1}$ is a sequence of non-zero complex numbers, we say that the infinite product $\prod_{n=1}^\infty b_n$ converges to $p \in \C$ if $p \neq 0$, and
$$\lim_{N \to \infty}\prod_{n=1}^Nb_n=p.$$
In this case, we must have $\lim_{n \to \infty}b_n =1$, so putting $b_n = 1+c_n$, then $\lim_{n \to \infty}c_n =0$.

Since $\lim_{n \to \infty}b_n=1$, we may assume $\Re b_n>0$, and choose a branch $\Log$ of $\log z$ defined on $\{\Re z>0\}$, for example $\Log(z) := \log|z| + i\arg (z)$, where $-\pi < \arg(z) < \pi$. Assuming $b_n \neq 0$ for all $n$, we can show the following prop


\begin{prop}
Suppose $b_n \neq 0$ for all $n$.
    \begin{enumerate}
    \item $\prod_{n=1}^\infty b_n$ converges $\iff \sum_{n=1}^\infty \Log(b_n)$ converges, in which case
    $$\prod_{n=1}^\infty b_n = \exp\paren{\sum_{n=1}^\infty \Log(b_n)}$$
    \item We say $\prod_{n=1}^\infty b_n$ \term{converges absolutely} if $\sum_{n=1}^\infty \Log(b_n)$ converges absolutely. Then, the product $\prod_{n=1}^\infty (1+c_n)$ converges absolutely $\iff \sum_{n=1}^\infty |c_n| < \infty$.
    \item If $\{f_n\}$ is a sequence of holomorphic functions on an open $U \subset \C$, and if $f_n = 1+F_n$, we say that the product $\prod_{n=1}^\infty f_n$ \term[converges uniformly]{converges uniformly on $U$} if and only if the series $\sum_{n=1}^\infty F_n$ converges uniformly on $U$. Moreover, uniform convergence on compact subsets of $U$ implies $\prod_{n=1}^\infty f_n$ is holomorphic on $U$.
\end{enumerate}
\end{prop}

\begin{proof}
    To prove (2), we use the inequality
    $$\frac{|z|}{2}\leq \abs{\Log(1+z)}\leq 2|z|$$
    for small $z$, which holds since
    $$\lim_{z \to 0}\frac{\Log(1+z)}{z}=1.$$
\end{proof}

Now, we're going to want some terms to be zero in an infinite product. 
\begin{defn}
We extend our definition of convergence to mean $\prod_{n=1}^\infty b_n$ converges when there exists $N \in \N$ such that $b_n \neq 0$ for all $n \geq N$, and $\prod_{n=N}^\infty b_n$ converges to a non-zero complex number $p$. The value of $\prod_{n=1}^\infty b_n$ is then defined to be $b_1\cdots b_{N-1}\cdot p$.    
\end{defn}

\subsection{\texorpdfstring{The $\sigma$-Function}{The sigma-Function}}

\begin{defn}
    Given a lattice $\Omega = \Z\omega_1 + \Z\omega_2 \subset \C$, define the \term[\sigma]{Weierstrass $\sigma$-function} to be
    $$\sigma(z):=z\prod_{\omega \in \Omega^*}\paren{1-\frac{z}{\omega}}\exp\paren{\frac{z}{\omega
    }+\frac{z^2}{2\omega^2} }.$$
    The exponential factor is included to ensure convergence.
\end{defn}

\begin{prop}
    $\sigma(z)$ converges.
\end{prop}

\begin{proof}
    Let $K \subset \C$ be any compact set, and for $\omega \neq 0$ 
    $$g_\omega(z) :=\paren{1 - \frac zw}\exp\paren{\frac{z}{\omega} + \frac{z^2}{2\omega^2} }.$$
    Then, as $|\omega| \to \infty$, we have $g_\omega(z) \to 1$ converges uniformly on $K$. Now, for $|\omega|$ large --- say $|\omega| > 2|z|$ for $z \in K$ --- we have
    \begin{align*}
        \abs{\Log g_\omega(z)} &= \abs{\Log\paren{1-\frac{z}{\omega}}+\frac{z}{\omega}+\frac{z^2}{2\omega^2}}=\abs{\frac{1}{3}\frac{z^3}{\omega^3} +\frac{1}{4}\frac{z^4}{\omega^4}+\cdots}
        \\ & \leq \frac{1}{3}\abs{\frac{z}{\omega}}^3 \paren{1+ 1/2 + 1/4 + \cdots} \leq \abs{\frac{z}{\omega}}^3
    \end{align*}
    for all $z \in K$. Then, since $\sum_{\omega \neq 0}\frac{1}{|\omega|^3}<\infty$, we see that 
    $$z\prod_{\omega \neq 0}g_\omega(z)$$
    converges uniformly on compact subsets of $\C$ to $\sigma(z)$.
\end{proof}

\begin{rmk}
    Observe that $\sigma(-z) = -\sigma(z)$, so $\sigma$ is an odd function with simple zeros at the points of $\Omega$. Also,
    $$\zeta(z) = \frac{\sigma'(z)}{\sigma(z)}=\frac{d}{dz}\paren{\Log\sigma (z)}.$$
    Since $\zeta'(z)=-\wp(z)$ and $\wp$ is periodic, there is a constant $\eta(\omega)$ such that
    $$\zeta(z+\omega)=\zeta(z) + \eta(\omega)$$
    for all $\omega \in \Omega$. Clearly, $\eta(\omega)$ is $\Z$-linear, so if $\eta_1:=\eta(\omega_1)$ and $\eta_2:=\eta(\omega_2)$, then
    $$\eta(\omega) = \eta(m \omega_1 + n\omega_2) = m\eta_1 + n \eta_2.$$
\end{rmk}

\begin{thm}[Legendre's Relation][label=thm:legrelation]
    $$\eta_1\omega_2 - \eta_2\omega_1 = 2\pi i.$$
\end{thm}

\begin{proof}
    Since $\zeta(z)$ is meromorphic with a single pole at $0$ with residue $1$, we get that
    \begin{align} \label{eq:int_zeta}
        \int_{\partial P}\zeta(z) dz = 2\pi i.
    \end{align}
    Break $\partial P$ into the four sides $\gamma_1,\dots,\gamma_4$. Then, observe
    \begin{align*}
        \int_{\gamma_3}\zeta(z)dz = -\int_{\gamma_1} \zeta(z+\omega_w)dz = -\int_{\gamma_1}\braks{\zeta(z) + \eta_2}dz,
    \end{align*}
    hence
    $$\int_{\gamma_1}\zeta + \int_{\gamma_3}\zeta = -\int_{\gamma_1}\eta_2dz = -\eta_2\omega_1.$$
    Similarly,
    $$\int_{\gamma_2}\zeta + \int_{\gamma_4}\zeta = \eta_1\omega_2,$$
    so plugging into \eqref{eq:int_zeta} yields the result.
\end{proof}

Now, we examine how $\sigma(z)$ behaves under translation. Using
$$\frac{\sigma'(z)}{\sigma(z)}=\zeta(z)$$
gives
$$\frac{\sigma'(z+\omega)}{\sigma(z+\omega)} = \frac{\sigma'(z)}{\sigma(z)} +\eta(\omega)$$
for all $\omega \in \Omega$. Integrating yields 
$$\log \sigma(z+\omega) = \log \sigma(z) + \eta(\omega)z  +c$$
where $c$ depends only on $\omega$. Hence,
\begin{align} \label{eq:sigma_frac}
    \frac{\sigma(z+\omega)}{\sigma(z)}=e^{\eta(\omega)z + c.}
\end{align}
Now evaluating $c$: assume that $\omega/2 \not \in \Omega$, so $\sigma(\omega/2) \neq 0$. Put $z:= -\omega/2$ in \eqref{eq:sigma_frac}, and we use the fact that $\sigma$ is odd to obtain
\begin{align*}
    \sigma(\omega/2) = -\sigma(\omega/2)e^{-\frac{1}{2}\eta\omega + c}\implies e^c = -e^{\frac{1}{2}\eta\omega},
\end{align*}
where $\eta:= \eta(\omega)$. Hence, if $\omega/2 \not \in \Omega$, we have
$$\frac{\sigma(z+\omega)}{\sigma(z)}=-e^{\eta (z+\omega/2)}.$$
On the other hand, for any $\omega \in \Omega$, let
$$\veps (\omega):= \frac{\sigma(z+\omega)}{\sigma(z)}e^{-\eta(z+\omega/2)}$$
so that $\veps(\omega)=-1$ if $\omega/2 \not \in \Omega$. Then,
\begin{align*}
    \frac{\sigma(z+2\omega)}{\sigma(z)}=\frac{\sigma(z+2\omega)}{\sigma (z+\omega)}\frac{\sigma(z+\omega)}{\sigma(z)} \implies \veps(2\omega) = \veps(\omega)^2
\end{align*}
for all $\omega \in \Omega$. Therefore, if $\omega/2 \in \Omega$, then
$$\veps(\omega) = \veps(\omega/2)^2 =\veps(\omega/4)^4 = \cdots = (-1)^{2n}=1.$$

\begin{thm}
    We have
    \begin{align*}
        \frac{\sigma(z+\omega)}{\sigma(z)}=\veps(\omega) \exp\paren{\eta (\omega)\cdot (z+\frac{\omega}{2})},
    \end{align*}
    where
    $$\veps(\omega) := \case{1 & \omega/2 \in \Omega \\ -1 & \omega/2 \not \in \Omega.}$$
    Moreover, if $\omega = m\omega_1 + n\omega_2$, then
    $$\veps(\omega ) = (-1)^{mn+m+n}.$$
\end{thm}

\begin{proof}
    Above.
\end{proof}

We return to the problem of finding $f \in \mc M(\Omega)$ with $[a_1],\dots,[a_r]$ and $[b_1],\dots,[b_s]$ as its congruence classes of zeros and poles, with multiplicities $k_1,\dots,k_r$ and $\ell_1,\dots,\ell_s$ respectively.  We already have the necessary conditions 
\begin{enumerate}[(i)]
    \item $\sum_{j=1}^r k_j = \sum_{i=1}^s \ell_i$.
    \item The sets $[a_1],\dots,[a_r]$ and $[b_1],\dots, [b_s]$ are disjoint.
    \item $\sum_{j=1}^rk_ja_j \equiv \sum_{j=1}^s \ell_jb_j \pmod \Omega$.
\end{enumerate}

\begin{thm}
    If conditions (i)-(iii) above hold, then there exists an elliptic function, unique up to scalar multiple, with exactly the wanted zeros and poles.
\end{thm}

\begin{proof}
    Let $n = \sum_{j=1}^rk_j=\sum_{i=1}^s\ell_i$. Write $u_1,\dots,u_n$ for the classes $[a_1],\dots,[a_r]$, with $a_i$ taken $k_i$ times, and similarly write $v_1,\dots,v_n$ for $[b_1],\dots,[b_s]$. Then, condition (iii) says
    $$\sum_{j=1}^n(u_j-v_j) = \omega \in \Omega,$$
    so we may replace $u_1$ by $u_1+\omega$, so that
    \begin{align*}
        \sum_{j=1}^nu_j=\sum_{j=1}^nv_j.
    \end{align*}
    Let
    $$f(z) := \frac{\sigma(z-u_1)\cdots \sigma(z-u_n)}{\sigma(z-v_1)\cdots \sigma(z-v_n)},$$
    where $\sigma$ is the $\sigma$-function associated to the lattice $\Omega$. Note that $f$ is meromorphic on $\C$. Now, since $\omega_1/2 \not \in \Omega$, we know
    \begin{align*}
       \sigma(z-u_j+\omega_1) &= -\sigma(z-u_j)\exp\paren{ \eta_1\cdot (z-u_j+\omega_1/2 )}
       \\ \sigma(z-v_j+\omega_1) &= -\sigma(z-v_j)\exp\paren{ \eta_1\cdot (z-v_j+\omega_1/2 )}
    \end{align*}
    for all $j=1,\dots,n$, and similarly for $\omega_2$. Therefore, for $i=1,2$,
    $$f(z+\omega_i) = f(z)\exp \paren{\eta_i \sum_{j=1}^n(v_j-u_j)}=f(z),$$
    and so $f(z) \in \mc M(\Omega).$
\end{proof}

\begin{cor}
    For any $a \in \C$ not in $\Omega$, we have
    $$\wp(z)-\wp(a) = -\frac{\sigma(z-a)\sigma(z+a)}{\sigma^2(z)\sigma^2(a)}$$
\end{cor}

\begin{proof}
    The function $\wp(z)-\wp(a)$ has zeros at $a$ and $-a$, and a double pole at $0$. Hence, by the theorem and its proof,
    $$\wp(z)-\wp(a) = c\frac{\sigma(z-a)\sigma(z+a)}{\sigma^2(z)}$$
    for some $c \in \C$. Multiply by $z^2$ and let $z \to 0$. Using that $\frac{\sigma^2(z)}{z^2}\to 1$ and $z^2\wp(z) \to 1$, we have
    $$c = \frac{1}{\sigma(a)\sigma(-a)}=-\frac{1}{\sigma^2(a)}$$
    as desired.
\end{proof}

\section{Several Complex Variables}

Recall the Cauchy-Riemann equations in 1-variable: if $f:U \to \C$ with is continuous on $U \subset \C$ open, and has all its first partial derivatives, then we set
$$\partder[]{f}{z}(a):=\frac{1}{2}\paren{\partder{f}{x}(a)-i\partder{f}{y}(a)} \quad \quad \partder{f}{\bar z}(a):= \frac{1}{2}\paren{\partder{f}{x}(a) +i\partder{f}{y}(a)}.$$
Then, $f$ is holomorphic on $U$ if and only if the Cauchy-Riemann equations hold, which we can write as
$$\partder{f}{\bar z}(a)=0, \quad \forall a \in U.$$
Note that size $z=x+iy$ and $\bar z=x-iy$, then $x=\frac{z+\bar z}{2}$ and $y=\frac{z-\bar z}{2i}$, so the formulas for $\partder{f}{z}$ and $\partder{f}{\bar z}$ are just the chain rule.


\begin{defn}
    L et $\Omega \subset \C^n$ be open. A function $f:\Omega \to \C$ is \term{holomorphic} if for every point $a:=(a_1,\dots,a_n) \in \Omega$, there exists a neighborhood $U$ of $a$ and constants $c_\alpha:=c_{\alpha_1,\dots,\alpha_n}, \alpha_j \in \Z_{\geq 0}$ such that
    \begin{align} \label{eq:holseries}
        f(z) = \sum_{\alpha}c_\alpha(z_1-a_1)^{\alpha_1}\cdots (z_n-a_n)^{\alpha_n}, \quad z \in U.
    \end{align}
\end{defn}

If this series \eqref{eq:holseries} converges for $z=w:=(w_1,\dots,w_n)$, then it converges uniformly on compact subsets of the polydisk
    $$\brax{z \setbar |z_j-a_j|<|w_j-a_j|,1\leq j \leq n}.$$
One can show that \eqref{eq:holseries} can be differentiated termwise. Hence,
$$c_\alpha = \frac{1}{\alpha!}D^\alpha f(a),$$
where $\alpha! = \alpha_1!\cdots \alpha_n!$, and $D^\alpha =\frac{\partial^{|\alpha |}}{\partial_{z_1}^{\alpha_1}\cdots \partial_{z_n}^{\alpha_n}}$, and $|\alpha|:=\alpha_1+\cdots +\alpha _n$.

\begin{thm}
    Let $\Omega \subset \C^n$ be open, and $f:\Omega \to \C$ be continuous. Suppose that for any $a \in \Omega$, the partial derivatives
    $$\partder{f}{\bar z_j}:=\frac{1}{2}\paren{\partder{f}{x_j}+i\partder{f}{y_j}}$$
    exists at $a$, and $\partder{f}{\bar z_j}(a)=0$ for $j=1,\dots,n$. Then $f$ is holomorphic on $\Omega$.
\end{thm}

\begin{proof}
    Let $r>0$ be such that the closed polydisk 
    $$\bar P(a,r):=\prod_{j=1}^n\bar D(a_j,r) = \brax{z \setbar |z_j-a_j|<r, 1 \leq j \leq n}$$
    is contained in $\Omega$. For $w_1 \in D(a_1,r)$ and $z_2,\dots,z_n$ fixed, Cauchy's formula gives
    $$f(w_1,z_2,\dots,z_n) = \frac{1}{2\pi i}\int_{C_1}\frac{f(z_1,\dots,z_n)}{z_1-w_1}dz_1,$$
    where $C_1$ is the circle $t \mapsto a_1 +re^{2\pi i t}$, $t \in [0,1]$. Let $C_j$ be the circle $t \mapsto a_j +re^{2\pi it}$, and iterate Cauchy's formula to get
    \begin{align*}
        f(w_1,\dots,w_n) = \frac{1}{(2\pi i )^n}\int_{C_1}\cdots \int_{C_n}\frac{f(z_1,\dots,z_n)}{\prod_{j=1}^n(z_j-w_j)}dz_n \cdots dz_1
    \end{align*}
    for $|w_j-a_j|<r$, $1 \leq j \leq n$. Since $f$ is continuous, $f$ is bounded on $\bar P(a,r)$. We may expand for each $j$, and see that
    $$-\frac{1}{z_j-w_j}=\frac{1}{(z_j-a_j)-(w_j-a_j)}=\sum_{m=0}^\infty \frac{(w_j-a_j)^m}{(z_j-a_j)^{m+1}},$$
    which converges uniformly on compact subsets of $D(a_j,r)$. Hence, $f(w) = \sum_{\alpha}c_\alpha(w-a)^{\alpha}$ with
    $$c_\alpha = \frac{1}{(2\pi i)^n}\int_{C_1}\cdots \int_{C_n}\frac{f(z_1,\dots, z_n)}{\prod_{j=1}^n(z_j-a_j)^{\alpha_j+1}}dz_1\cdots dz_n,$$
    so $f$ is holomorphic.
\end{proof}

\begin{prop}[Properties of Holomorphic Functions on $\C^n$]
    Holomorphic functions on $\C^n$ satisfy the principle of analytic continuation, the open mapping theorem, Cauchy's formula for a polydisk, Cauchy's inequalities, and theorems of Weierstrass and Montel.
\end{prop}

There are also some properties from single complex analysis that fail in several complex variables.

\begin{thm}[Hartogs' Theorem, 1905]
If $n\geq 2$, then a holomorphic function $f:\C^n\to \C$ has no isolated singularities.
\end{thm}

\begin{proof}
    In fact, something stronger is true: Let $\Omega = B_1 \setminus \bar B_2$, where $B_1,B_2$ are open balls with $\bar B_2 \subset B_1$. Then, if $f \in \mc O(\Omega)$ is holomorphic, then there exists unique $F \in \mc O(B_1)$ such that $F|_\Omega=f$. In particular, an isolated singularity $\bar B_2=\{w\}$ can always be filled in.
\end{proof}

\begin{thm}[The Riemann Mapping Theorem Fails]
    There does not exists a holomorphic non-constant $f$ from a ball to a polydisk in $\C^2$. In fact, there's not even a proper map.
\end{thm}

\section{The Riemann Surface of an Algebraic Function}

Why did we discuss several complex variables? Well, one of our main motivators is zero sets of polynomials. If our polynomial has more than one variable (which they usually do), then this induces a surface in several complex variables.

Suppose $p\in \C[x,y]$ is an irreducible polynomial with $\deg_yP =n>0$, so 
\begin{align} \label{eq:P(x,y)}
    P(x,y)=a_0(x)y^n+\cdots + a_{n-1}(x)y+a_n(x).
\end{align}
Intuitively, the equation $P(x,y)=0$ defines an algebraic function $y=y(x)$. We want to make this precise.

\begin{thm}[Implicit Function Theorem][label=thm:implicit_function_theorem]
    Let $f(x,y)$ be a holomorphic function of 2 complex variables $x$ and $y$, defined on the polydisk $$\brax{|x|<r_1,|y|<r_2}\subset \C_2$$ such that $f(0,0)=0$ and $\partder{f}{y}(0,0)\neq 0$. Then, there exists $\veps,\delta>0$ such that for any $x \in D(0,\veps)=\brax{z \in \C \setbar |z|<\veps}$, there is a unique solution $y(x)$ to the equation $f(x,y)=0$ with $|y(x)|<\delta$.  The function $x \mapsto y(x)$ is holomorphic.
\end{thm}

\begin{proof}
    Since $\partder{f}{y}(0,0)\neq 0$, we can choose $\delta>0$ such that $f(0,y)\neq 0$ for $0<|y| \le \delta$. Now, choose $\veps>0$ such that $f(x,y)\neq 0$ for all $|x| \leq \veps$ and $|y|=\delta$. This is possible, since $f$ is continuous and does not vanish on the compact set $\{0\} \times \{|y|=\delta\}$. By the argument principle, if $|x|<\veps$, then the integral
    \begin{align} \label{eq:implicitfuncintegral}
        \frac{1}{2\pi i}\int_{|y|=\delta}\frac{f_y(x,y)}{f(x,y)}dy
    \end{align}
    is an integer $n(x)$ equal to the number of zeros of the function $y \mapsto f(x,y)$ in $\{|y|<\delta\}$. Moreover, our initial choice of $\delta$ implies $n(0)=1$. Since $f(x,y) \neq 0$ for $|x| \leq \veps$ and $|y|=\delta$, the integral \eqref{eq:implicitfuncintegral} is a continuous function on $x$ for $|x|<\veps$. Hence, $n(x)=1$ for $|x|<\veps$. Therefore, there exists a unique zero $y(x)$ of $f(x,y)$ with $|y(x)|<\delta$. Finally, the residue theorem implies that 
    $$y(x) = \frac{1}{2\pi i}\int_{|y|=\delta}y\frac{ \frac{\partial f}{\partial y}(x,y) }{f(x,y)}dy,$$
    and it follows that $x \mapsto y(x)$ is a holomorphic map.
\end{proof}

\begin{lem}
    Let $P(x,y) = a_0(x)y^n+\cdots + a_n(x)$, and $a \in \C$ be such that $a_0(a) \neq 0$, and there is no $b \in \C$ such that $P(a,b) = \partder{P}{y}(a,b)=0$. Then, there exists $\veps >0$ and $n$ holomorphic functions $y_1(x),\dots,y_n(x)$ for $x \in D(a,\veps)$ such that
    \begin{enumerate}[(i)]
        \item $y_i(x) \neq y_j(x')$ if $i \neq j$, $x,x' \in D(a,\veps)$, and $P(x,y_j(x))=0$ for $x \in D(a,\veps),$ $j=1,\dots,n$.
        \item If $z \in \C$ and $P(x,z)=0$ for some $x \in D(0,\veps)$, then $z = y_j(x)$ for a unique $j \in [1,n]$.
    \end{enumerate}
\end{lem}

\begin{proof}
    Since $\partder{P}{y}(a,b) \neq 0$ for all solutions $b$ of $P(a,b)=0$, the polynomial $P(a,y)$ has exactly $n$ roots, say $b_1,\dots,b_n$. Take $\veps>0$ small and for each $i$, let $y_i(x)$ be the holomorphic function on $D(a,\veps)$ with $y_i(a) = b_i$ and $P(x,y_i(x))=0$ (which exists by the implicit function theorem \Cref{thm:implicit_function_theorem}). Then, for small enough $\veps>0$, the $y_i$ have the sequence properties (the second follows since the equation $P(x,y)=0$ has $\leq n$ solutions).
\end{proof}

\begin{prop}[][label=prop:simsolutions]
    Let $P(x,y)$ be as in \eqref{eq:P(x,y)}. Then, there are only finitely many $x \in \C$ such that the equations $P(x,y)=0$ and $\partder{P}{y}(x,y)=0$ have a simultaneous solutions $y\in \C$.
\end{prop}

\begin{proof}
    Set $Q:=\partder{P}{y}$. By Euclidean division, there exists $k_0 \geq 1$ so that if $b_0(x)=a_0(x)^{k_0}$, we have
    $$b_0(x)P=h_0(x,y)Q+Q_1,$$
    where $\deg_yQ_1 < \deg _yQ = n-1$. Similarly, there exists polynomials $b_j(x)$ such that
    \begin{align*}
        b_1(x)Q &= h_1(x,y)Q_1 + Q_2, \quad \deg_yQ_2 < \deg_yQ_1,\dots
        \\ b_r(x)Q_{r-1} &=h_r(x,y) Q_r + Q_{r+1} \quad \deg_y Q_{r+1}< \deg _yQ_r.
    \end{align*}
    We may assume $\deg_y Q_{r+1}=0$, otherwise we continue the division process.
    \begin{claim}
        $Q_r+1 \not \equiv 0$.
    \end{claim}
    \begin{proof}
        Indeed, if $Q_{r+1}=0$, then $Q_r \mid b_r(x)Q_{r-1}$, so if $f$ is an irreducible factor of $Q_r$ with $\deg_yf>0$, $f$ must divide $Q_{r-1}$. Iteration shows that $f\mid Q$ and $f \mid P$, contradicting the irreducibility of $P$.
    \end{proof}
    Finally, if $(a,b) \in \C^2$ is such that $P(a,b)=0=Q(a,b)$, then $Q_{r+1}(a)=0$. As $Q_{r+1} \not \equiv 0$, we have
    $$\brax{a \in \C \setbar \exists b \in \C \text{ with } P(a,b)=0=Q(a,b)}\subset \{a\in \C \mid Q_{r+1}(a)=0\}$$
    is finite.
\end{proof}

\begin{defn}
    We write,
    \begin{align*}
        S_0 &:=\{ x \in \C \mid a_0(x)=0\},
        \\ S_1 &:= \brax{x \in \C \mid \exists y \in \C, P(x,y)=\partder{P}{y}(x,y)=0},
        \\ S &:= S_0\cup S_1 \cup \{\infty\}.
    \end{align*}
    Then, $S \subset \P^1$ is called the set of \term[critical point]{critical points} of $x$. Note that $S$ is finite by \Cref{prop:simsolutions}.
\end{defn}

\begin{prop}[][label=prop:projproper]
    Let $P(x,y)$ be as in \eqref{eq:P(x,y)}, and assume $a_0(x) \not \equiv 0$. Let
    $$V := V(P):= \{(x,y) \in \C^2 \mid P(x,y)=0\},$$
    and $\pi:V \to \C$ sending $(x,y) \mapsto x$ be the first projection. Then, the map
    $$\pi|_{\pi^{-1}(\C\setminus S_0)}:\pi^{-1}(\C\setminus S_0) \to \C\setminus S_0$$
    is proper.
\end{prop}

\begin{proof}
We use a lemma which gives a bound on the roots of a polynomial in terms of its coefficients.
\begin{lem}[][label=lem:rootbound]
    Let $c_1,\dots,c_n \in \C$, and suppose $w^n+c_1w^{n-1}+\cdots + c_n=0$. Then, $|w|<2 \max _{i}|c_i|^{1/i}$, unless $c_i=0$ for all $i$.
\end{lem}

\begin{proof}
    Let $c:= \max_i |c_i|^{1/i}>0$, and set $z:=w/c$. Then, we have $z^n+\frac{c_1}{c}z^{n-1}+\cdots + \frac{c_n}{c^n}=0$. It now suffices to show $|z|<2$.

    Indeed, since $|c_i| \leq c^i$ for all $i$, we have $|z|^n \leq |z|^{n-1}+\cdots + |z|+1$. If $|z| \geq 2$. Thus, if $|z| \geq 2$, then
    $$1 \leq \frac{1}{|z|}+\cdots + \frac{1}{|z|^n}\leq \frac{1}{2}+\cdots + \frac{1}{2^n}<1,$$
    a contradiction.
\end{proof}
    Now, suppose $K \subset \C \setminus S_0$ is compact. Then, there exists $\delta>0$ such that $|a_0(x)| \geq \delta$, and $|a_i(x)| \leq 1/\delta$ for all $i \geq 1$ and $x \in K$. If $(x,y) \in V$ and $x \in K$, then
    $$y^n+\frac{a_1(x)}{a_0(x)}y^{n-1} +\cdots + \frac{a_n(x)}{a_0(x)}=0,$$
    and $\abs{\frac{a_i(x)}{a_0(x)}}\leq 1/\delta^2$ for all $i$. By \Cref{lem:rootbound}, $|y| \leq 2\max_i(1/\delta^{2/i})$, hence $\pi^{-1}(K)$ is bounded. Moreover, $\pi^{-1}(K) =(K \times \C) \cap V$ is closed in $\C^2$, so $\pi^{-1}(K)$ is compact.
\end{proof}

\begin{cor}
    Let $P(x,y) \in \C[x,y]$ be irreducible, $V:=V(P) \subset \C^2$, and $\pi :=\w{pr}_1|_V$. Then, for $S$ the set of critical points of $x$, the map
    $$\pi|_{\pi^{-1}(\P^1\setminus S) }:\pi^{-1}(\P^1 \setminus S) \to \P^1 \setminus S$$
    is a finite sheeted covering map.
\end{cor}

\begin{proof}
    By \Cref{prop:projproper}, $\pi$ is a proper map. The result then follows from the Implicit Function Theorem (\Cref{thm:implicit_function_theorem}), which gives that $\pi$ is a local homeomorphism.
\end{proof}

\begin{rmk}
The space $Y:= \pi^{-1}(\P^1\setminus S)$ carries the structure of a Riemann surface in a natural way, so that $\pi$ is holomorphic. 
\end{rmk}
Indeed, if $Z$ is a Riemann surface, $X$ is a connected real 2 dimensional manifold, and $p:X \to Z$ is a local homeomorphism, then there is a unique complex structure on $X$ such that $p$ is holomorphic.

For $U$ any open set in $X$ such that $p|_U$ is a homeomorphism onto an open set $V \subset Z$ with $V \subset V_j$ for a chart $(\psi_j,V_j)$ of $Z$, let $\varphi_U:U \to \C$ be the map $\varphi_U:=\psi_j \circ p$. One checks that $(U,\varphi_U)$ and $(U',\varphi_{U'})$ are compatible, so we get a complex structure on $X$ for which $p$ is holomorphic.

\begin{claim}
    $f$ extends to a meromorphic function on $X$.
\end{claim}

\begin{proof}
    Let $a \in S, x \in X$ with $p(x)=a$, and choose local coordinates $z$ at $x$ and $w$ at $a$ so that $p$ becomes $z \mapsto z^m=w$. If $U$ is a neighborhood by $x$, then by definition of $V$ and $f$, we have $$f(z)^n+\frac{a_1(w)}{a_0(w)}f(z)^{n-1}+\cdots + \frac{a_n(w)}{a_0(w)}=0$$
    for $z \neq 0, w=p(z)$. Also, the functions $a_i(w)/a_0(w)$ are clearly meromorphic at $w=0$, so there exists constant $C,N>0$ such that 
    $$\abs{\frac{a_i(w)}{a_0(w)}}\leq \frac{C}{|w|^N}$$
    near $w=0$. Then, by \Cref{lem:rootbound}, 
    $$|f(z)| \leq 2 \max_i \frac{C^{1/i}}{|w|^{N/i}}\leq \frac{C_1}{|z|^k}$$
    for some constants $C_1$ and $k$. Hence, $f$ has a meromorphic extension to all of $X$.
\end{proof}

\begin{prop}
    $Y$ is connected, so $X$ is as well. Therefore, $X$ is a compact Riemann surface which carries a meromorphic function $f:X \to \P^1$. Moreover, if $p:X \to \P^1$ is the extension of $\pi:Y \to \P^1 \setminus S$, then we have $P(p(x),f(x))=0$ on $X$.
\end{prop}

\begin{proof}
    Suppose $Y$ is not connected, and let $Y_1$ be a connected component of $Y$, $Y_1 \neq Y$. Then, $\pi_1:= \pi|_{Y_1}:Y_1 \to \P^1 \setminus S$ is an $r$-sheeted covering map for some $r$ with $1 \leq r < n$. Let $f_1:=f|_{Y_1}$.

    For any $x \in \P^1 \setminus S$, let $e_\nu$ for $\nu=1,\dots,r$ be the $\nu^{\text{th}}$ elementary symmetric function $e_\nu(x):=e_\nu(y_1,\dots,y_r)$, where $y_1,\dots,y_r$ are the values taken by $f_1$ at the $r$ points of $\pi^{-1}_1(x)$. By definition of $f_1$, the $y_j$ are are values of the $2^{\text{nd}}$ projection $\w{pr}_2$ at the points $(x,y) \in Y$, so we have $P(x,y_j)=0$ for $1 \leq j \leq r$.

    \begin{claim}
        The $e_\nu(x)$ extend to meromorphic functions on $\P^1$.
    \end{claim}

    \begin{proof}
        Indeed, since the $y_j$ are values of the function $f_1$, which is meromorphic on $Y_1$, in a neighborhood of any $a \in S$, since $\pi_1^{-1}(x) = \{P_1,\dots,P_r\}$, we have an estimate
        $$|e_\nu(x)| = \abs{\sum_{i_1<\cdots < i_\nu} f_1(P_{i_1})\cdots f_1(P_{i_\nu})} \leq C \norm{z}^{-\ell}.$$
        Here, $z=(z_1,\dots,z_r)$, where $z_i$ is a coordinate function near $P_i$ for $i=1,\dots,r$. Thus,
        $$|e_\nu(x)| \leq C_1 |x-a|^{-\ell'}$$
        (or $C_1|x|^{-\ell}$ if $a = \infty$), proving that the $e_\nu$ are meromorphic on $\P^1$.
    \end{proof}
    This claim implies that $e_1(x),\dots,e_r(x)$ are rational functions of $x$. Let
    $$Q(x,y) := y^r-e_1(x)y^{r-1}+\cdots + (-1)^re_r(x).$$
    Then, if $x \in \P^1 \setminus S$, the roots $y_1,\dots,y_r$ of $Q(x,y)$ are also roots of $P(x,y)$. Hence, $Q$ divides $P$ in $\C(x)[y]$, and since $\deg_y Q\geq 1$, Gauss's Lemma implies that $P$ is not irreducible in $\C[x][y]$, a contradiction.
\end{proof}

\section{\texorpdfstring{The Equation $\partial u/\partial \bar z=f$}{The Equation du/dzbar = f}}

Let $R$ be a closed rectangle in $\C$, and $f \in C^{\infty}(R)$ (that is, $f$ is $C^\infty$ in an open neighborhood of $R$).

\begin{prop}[][label=prop:sortastokes]
    $$\iint_R\partder{f}{\bar z}dz\land d\bar z = -\int_{\partial R}f dz,$$
    where $dz \land d \bar z = (dx+idy)\land (dx - idy)=-2i\,dx \land dy$
\end{prop}

\begin{proof}
    Suppose $R=[a,b]\times [c,d]$, and parametrize the four sides of $\partial R$ by $\gamma_1,\gamma_2,\gamma_3,\gamma_4$. Then,
    \begin{align*}
        \iint_R \partder{f}{\bar z}dz \land d \bar z &=-i \int_{c}^d\int_a^b\paren{\partder{f}{x}+i\partder{f}{y}}dxdy
        \\ &= -i\int_c^d \braks{f(b,y)-f(a,y)}dy
        \\ &=-\int_{\gamma_2}fdz -\int_{\gamma_4}fdz.
    \end{align*}
    The rest is an exercise.
\end{proof}

Next, let $R'$ and $R$ be closed rectangles with $R' \subset \Int(R)$. Choose $f$ in some closed annulus $C^\infty (R\setminus \Int(R'))$.
\begin{claim}
    We claim
    $$\iint_{R\setminus \Int(R')}\partder{f}{\bar z}dz \land d\bar z = -\int_{\partial R}fdz + \int_{\partial R'}fdz.$$
\end{claim}
\begin{proof}
    We have $f \in C^\infty (U)$ for some open set $U \supset R\setminus \Int (R')$. Let $\alpha \in C^\infty _0(U)$ with $\alpha \equiv 1$ in a neighborhood of $R\setminus \Int(R')$ (the $0$ in the subscript means compactly supported). Consider
    $$g(z) :=\case{\alpha(z)f(z) & z \in U \\ 0 & z \not \in U.}$$
    Then, $g \in C^\infty (\C)$. Applying \Cref{prop:sortastokes} to $g$ on $R$ and $R'$, we obtain
    $$\iint_R \partder{g}{\bar z}dz \land d\bar z = -\int_{\partial R}f, \quad \iint_{R'}\partder{g}{\bar z}dz \land d\bar z=-\int_{\partial R'}f,$$
    and subtracting gives the claim.
\end{proof}

\begin{thm}[][label=thm:fisdoubleint]
    Let $f \in C_0^\infty (\C)$. Then, for any $\omega \in \C$,
    \begin{align} \label{eq:fisdoubleint}
        f(w) = \frac{1}{2\pi i }\iint_\C \partder{f}{\bar z}(z)\cdot  \frac{1}{z-w}dz \land d \bar z
    \end{align}
\end{thm}

\begin{proof}
    Observe that the function $z \mapsto \frac{1}{z}$ for $0 < |z| \leq 1$ lies in $L^1(\bar D(0,1))$, i.e.
    $$\iint_{|z|\leq 1}\frac{1}{|z|}dxdy < \infty.$$
    (Indeed, passing to polar coordinates shows that the integral equals $2\pi$). We now change variables in \eqref{eq:fisdoubleint} and consider
    $$I:= \iint_\C \partder{f}{\bar z}(z+w)\frac{1}{z}dz \land d \bar z.$$
    We claim $I = 2\pi i f(w)$.

    Indeed, fix $w$, and choose a closed rectangle $R$ such that $\supp(z \mapsto f(z+w)) \subset \Int (R)$ and $0 \in R$. Let $R_\veps$ be a small square centered at $0$ with side length $\veps$, and $R_\veps \subset R$. Using Lebesgue's dominated convergence theorem, we have
    \begin{align*}
        \iint_\C \partder{f}{\bar z}(z+w) \frac{dz \land d \bar z}{z}&=\iint_\C \partder{}{\bar z} \paren{\frac{f(z+w)}{z}}dz \land d \bar z
        \\ &= \lim_{\veps \to 0}\iint_{R \setminus \Int (R_\veps)}\partder{}{\bar z}\paren{\frac{f(z+w)}{z}}dz \land d\bar z
        \\ &= \lim_{\veps \to 0}\int_{\partial R_\veps}\frac{f(z+w)}{z}dz
        \\ &= 2\pi if(w) + \lim_{\veps \to 0}\int_{\partial R_\veps}\frac{f(z+w)-f(w)}{z}dz.
    \end{align*}
    Then, since $f \in C_0^\infty (\C)$, there exists $M>0$ such that $|f(z+w)-f(w)| \leq M|z|$ for $|z|$ small, hence
    $$\abs{\int_{\partial R_\veps}\frac{f(z+w)-f(w)}{z}dz}\leq M\cdot 4\veps \to 0,$$
    proving the claim.
\end{proof}

\begin{thm}[][label=thm:fispartder]
    Let $f \in C_0^\infty (\C)$, and define 
    $$u(w) := \frac{1}{2\pi i }\iint_\C \frac{f(z)}{z-w}dz \land d \bar z.$$
    Then, $u \in C^\infty (\C)$, and $\partder{u}{\bar z}=f$ on $\C$.
\end{thm}

\begin{proof}
    We change variables in the integral so that
    $$u(w) = \frac{1}{2\pi i } \iint _\C \frac{f(z+w)}{z}dz\land d \bar z.$$
    We claim $$\partder{u}{x}(w) =\frac{1}{2\pi i }\iint _\C \frac{\partder{f}{x}(z+w)}{z}dz \land d \bar z.$$
    Indeed, for $h \in \R$,
    $$\frac{u(w+h)-u(w)}{h}=\frac{1}{2\pi i}\iint_\C \frac{f(z+w+h)-f(z+w)}{h}\cdot \frac{1}{z} dz \land d \bar z.$$
    Since $f$ is differentiable, the first factor in the integral is a bounded function of $h$, and $\frac{1}{z}dz \land d\bar z$ is $L^1$ on any compact set, so we obtain the claim by taking the limit as $h \to 0$ (the exchange of limit and integral is justified by the dominated convergence theorem).

    Arguing in the same way shows that $\partder{u}{y}(w) $, and in fact all derivatives of $u$, exist. Hence, $u \in C^\infty (\C)$. Moreover, since $z=x+iy$, and $\partder{}{\bar z}=\frac{1}{2}\paren{\partder{}{x}+i\partder{}{y}}$, we have
    $$\partder{u}{\bar z}(w) = \frac{1}{2 \pi i} \iint_\C \partder{f}{\bar z}(z+w)\frac{1}{z}dz \land d \bar z = f(w)$$
    by \Cref{thm:fisdoubleint}.
\end{proof}

\begin{prop}

\begin{lem}[][label=lem:compact_seq]
    Let $\Omega \subset \C$ be open. Then, there exists a sequence $\{K_n\}_{n \geq 1}$ of compact subsets of $\Omega$ such that 
    \begin{enumerate}[(i)]
        \item $K_n \subset K_{n+1}$,
        \item $\bigcup_{n=1}^\infty K_n = \Omega$,
        \item $K_n = \paren{\hat {K_n}}_\Omega$, i.e. $\Omega \setminus K_n$ has no connected components relatively compact to $\Omega$.
    \end{enumerate}
\end{lem}

\begin{proof}
    We first construct a sequence $\{K_n'\}$ satisfying the first two conditions. 

    If $\Omega = \C$, just use disks $K_n = \bar D(0,n)$. If $\Omega \neq \C$, then $\partial \Omega \neq \emptyset$, so set 
    $$K_n =\brax{x \in \Omega \setbar \w{dist}(x,\partial \Omega) \geq \frac{1}{n}}\cap \bar D(0,n),$$
    and it's easy to show that these satisfy the first two conditions.

    Now, set $K_n := \hat{\paren{K_n'}}_\Omega$ for all $n \geq 1$, and we're done.
\end{proof}

\begin{thm}[Runge's Theorem]
    Let $\Omega \subset \C$ be open and $K \subset \Omega$ be compact. Then, any function $f \in \mc O(\hat K_\Omega)$ can be approximated uniformly on $\hat K_\Omega$, by functions holomorphic on $\Omega$.
\end{thm}

\end{thm}

\begin{proof}[Proof (Grothendieck)]
Let $\{K_n\}_{n \geq1}$ be a sequence of compact subsets of $\Omega$ as in \Cref{lem:compact_seq}. For each $n$, there exists $u_n \in C_0^\infty (\Omega)$ such that $\partder{u_n}{\bar z}=f$ in a neighborhood of $K_n$. Then,
$$\partder{}{\bar z}(u_{n+1}-u_n)=f-f=0$$
in a neighborhood of $K_n$, so $u_{n+1}-u_n \in \mc O(K_n)$. By Runge's Theorem, there exists $h_n \in \mc O(\Omega)$ such that
$$\norm{u_{n+1}-u_n-h_n}_{K_n} < \frac{1}{2^n}$$
for all $n \geq 1$. Now, define $u$ on $K_n$ by
\begin{align} \label{eq:def-of-u}
    u:=u_{n+1}+\sum_{\ell = n}^\infty (u_{\ell+1}-u_{\ell}-h_{\ell})-h_1-h_2-\cdots- h_{n-1},
\end{align}
and observe that the series converges uniformly on $K_n$. Moreover, $u$ is well defined on $\Omega$, since
\begin{align*}
    u&=u_k + \sum_{\ell \geq k}(u_{\ell+1}-u_{\ell}-h_\ell)-\sum_{j<k}h_j
    \\ &= u_k+(u_{k+1}-u_k-h_k)+\sum_{\ell \geq k+1}(u_{\ell+1}-u_{\ell}-h_{\ell})-\sum_{j < k}h_k
    \\ &= u_{k+1}+\sum_{\ell \geq k+1}(u_{\ell+1}-u_\ell -h_\ell) - \sum_{j < k+1}h_j,
\end{align*}
i.e. $u$ is independent of the choice of $n$ for $K_n$. Since the series converges uniformly on $K_n$, we have
$$u-u_{n+1}\in \mc O(K_n)$$
and therefore
$$\partder{u}{\bar z}=\partder{u_{n+1}}{\bar z}=f$$
on $\Int(K_n)$. Since $n$ is arbitrary, and $\bigcup_{n}K_n = \Omega$, we're done.
\end{proof}

\section{Cohomology and the Mittag-Leffler Problem}

\begin{notebox}[The Mittag-Leffler Problem]
   Let $\Omega \subset \C$ be open, $E \subset \Omega$ be discrete, and suppose that for all $a \in E$, we're given a principal part
$$p_a:=\sum_{\nu=1}^\infty \frac{c_\nu}{(z-a)^\nu} \in \mc O(\C \setminus \{a\})$$
where the $c_\nu$ are some given constants.

We ask to find a function $f\in \mc O(\Omega \setminus E)$ such that $f - p_a \in \mc O(a)$ for all $a \in E$.
\end{notebox}

Here is a cohomological approach to the problem: Let $\mc U=\{U_i\}_{i \in I}$ be a covering of $\Omega$ by open sets such that $|U_i \cap E| \leq 1$ for all $i \in I$, and let $f_i$ be any meromorphic function that solves the problem on $U_i$. Note that we can indeed solve the problem on $U_i$, because if $U_i$ contains one point $a \in E$, we can just take $f=p_a$, and otherwise $f$ can be anything holomorphic.

Now, define
$$f_{ij}:=f_i|_{U_i \cap U_j} - f_j|_{U_i \cap U_j} \in \mc O(U_i \cap U_j).$$
Then, on $U_i \cap U_j \cap U_k$, we have $f_{ij}+f_{jk}+f_{ki}=0$. Thus, solving the problem globally on $\Omega$ is equivalent to finding a family of functions $\{g_i \in \mc O(U_i)\}_{i \in I}$ such that $f_{ij}=g_j-g_i$ on $U_i \cap U_j$. for all $i,j \in I$. Indeed, given such a family $\{g_i\}$, we can define $f:= f_i+g_i$ on $U_i$, which is in fact a globally defined function on $\Omega$, because the definition agrees on overlaps. Then, $f$ satisfies the Mittag-Leffler conditions, and conversely a global solution induces such $\{g_i\}$.

In the \v{C}ech cohomology theory, $\{f_{ij}\} : f_{ij}+ f_{jk}+f_{ki}=0$ is a cocycle in $Z^1(\mc U,\mc O)$, and
$$\brax{\{f_{ij}\} \setbar f_{ij}=g_j-g_i \text{ for some } \{g_i\}  }=B^1(\mc U,\mc O)$$
is a coboundary group. The \term{first \v{C}ech cohomology group}
$$H^1(\mc U,\mc O):=Z^1(\mc U,\mc O)/B^1 (\mc U,\mc O)$$
is the obstruction to solving the problem in general. The Mittag-Leffler theorem says that this group is zero, for all $\Omega \subset \C$ open, and for all open covers $\mc U$ of $\Omega$.

\section{Sheaves of Abelian Groups}
''Sheaves are friends"
\begin{defn}
    Let $X$ be a topological space. A \term[presheafs]{presheaf} $\mc F$ of abelian groups of $X$ is an assignment $U \mapsto \mc F(U)$ of abelian groups $\mc F(U)$ to each open set $U \subset X$, with $F(\emptyset)=0$, and for every pair $V \subset U$ of open sets, there are homomorphisms $r_{U,V}: \mc F(U) \to \mc F(V)$ called \term[restriction map]{restriction maps}, satisfying
    \begin{enumerate}[(i)]
        \item $r_{U,U}=\id_{\mc F(U)}$ for all $U \subset X$ open
        \item If $W \subset V \subset U$ are three open sets, then $r_{U,W}=r_{V,W} \circ r_{U,V}$.
    \end{enumerate}
     We usually denote $r_{U,V}(f)$ by $f|_V$, as if $f$ is a function of $U$ and $r_{U,V}(f)$ is its restriction to $V$.

     The presheaf $\mc F$ is called a \term[sheafs]{sheaf} if for any $U \subset X$ and open cover $U=\bigcup_{i \in I}U_i$, we have
     \begin{enumerate}[(i)]
        \item If $f,g \in \mc F(U)$ with $r_{U,U_i}(f) = r_{U,U_i}(g)$ for all $i \in I$, then $f=g$.
        \item If we are given, for each $i \in I$, an element $f_i \in \mc F(U_i)$ such that $r_{U_i,U_i \cap U_j}(f_i)=r_{U_j,U_i\cap U_j}(f_j)$ for all $i,j \in I$, then there exists $f \in \mc F(U)$ such that $r_{U,U_i}(f)=f_i$ for all $i \in I$.
     \end{enumerate}
     An element $f \in \mc F(U)$ is called a \term[sections]{section} of $\mc F$ over $U$.
\end{defn}

\begin{rmk}
    By property (i) of a sheaf, the $f$ in (ii) is unique.
\end{rmk}

\begin{ex}
    Let $X$ be a Riemann surface.
    \begin{enumerate}
        \item \term[sheaf of holomorphic functions,\cO]{The sheaf of holomorphic functions $\mc O$}: for any open set $U \subset X$, $\mc O(U)$ is the group of holomorphic functions $U \to \C$
        \item \term[sheaf of $C^\infty$ functions,\cA]{The sheaf of $C^\infty$ functions $\cA$}: for any open set $U \subset X$, \[ \mc A(U) = C^\infty (U):= \{f:U \to C \mid f \text{ is } C^\infty \} \]
        \item \term[presheaf of constant functions]{The presheaf of constant functions $\mc F$}: $\mc F(U) = \{ f:U \to \C \text{ constant}\}$. Note that $\mc F$ is not a sheaf, as property (ii) fails: if $U$ is disconnected, we can send one component to $0$, another to $1$, and then we can't get a constant function that restricts to $0$ and $1$. However, one gets a sheaf $\C$ by setting $\C(U) := \{ f:U \to \C, f$ is locally constant$\}$.
    \end{enumerate}
\end{ex}

\begin{defn}
    Let $\mc F$ be a presheaf on $X$, and fix $a \in X$. We define an equivalence relation $\sim$ on pairs $(U,f)$, where $U\subset X$ is open with $a \in U$, and $f \in \mc F(U)$ as follows: we say $(U, f) \sim (V,g)$ if there exists $W$ open with $a \in W \subset U \cap V$ such that $f|_W = g|_W$. The equivalence class of a pair $(U,f)$ is called the \term[germs]{germ} of the section $f$ at $a$, denoted $f_a$. 
    
    The set of all germs of sections at $a$ forms an abelian group $\mc F_a$ called the \term[stalks]{stalk} of $\mc F$ at $a$.
\end{defn}

\begin{notethm*}
    We in fact have
    \[
    \mc F_a := \coprod _{U \ni a} (\mc F(U)/\sim) =: \varinjlim_{U \ni a} \mc F(U) 
    \]
\end{notethm*}
We can associate a sheaf to any presheaf $\mc F$ as follows: Let \term{$|\cF|$}$:=\coprod_{a \in X}\mc F_a$ and $p:|\mc F| \to X$, $f_a \mapsto a$ be the natural projection. Define a topology on $|\mc F|$ as follows: for any $U \subset X$ and $f \in \mc F(U)$, let $N(U,f):=\brax{f_a \setbar a \in U}$, where $f_a$ is the equivalence class in $\mc F_a$ defined by $(U, f)$. The system of sets $\brax{N(U,f)}_{U \subset X}$ is a basis for a topology on $|\mc F|$ such that $p$ is a local homeomorphism (exercise).

A \termnt{section} of $|\mc F|$ over $U$ is a continuous map $s:U \to |\mc F|$ such that $p \circ s=\id_U$, i.e. $s(a) \in \mc F_a$, for all $a \in U$. Let $|\mc F|(U)$ denote the set of all sections of $|\mc F|$ over $U$.

\begin{exercise}
    $|\mc F|$, together with the natural restriction maps, is a sheaf.
\end{exercise}

\begin{exercise}
    For every $a \in X$, there is a natural isomorphism of stalks
    \[
    \mc F_a \arr[\sim] |\mc F|_a.
    \]
\end{exercise}

\begin{ex}
    If $X$ is a Riemann surface, and $p \in X$, we define the \term{skyscraper sheaf} $\C_p$ by
    \[
    \C_p(U) := \case{\C & p \in U, \\ 0 & p \not \in U.}
    \]
    The stalks are
    \[
    (\C_p)_a = \case{0 & a \not \neq p \\ \C & a =p.}
    \]
\end{ex}

\section{\v{C}ech Cohomology of Sheaves}


Let $\mc F$ be a sheaf on the topological space $X$, and $\mc U:= \{U_i\}_{i \in I}$ be an open cover of $X$.
\begin{defn}
    Define \term[cochain group]{cochain groups} to be 
    \begin{align*}
        C^0(\mc U,\mc F) &:= \prod_{i \in I}\mc F(U_i) = \brax{ (f_i)_{i \in I} \setbar f_i \in \mc F(U_i), \: \forall i \in I} \\
        C^1 (\mc U,\mc F) &:= \prod_{(i,j)}\mc F(U_i \cap U_j)=\brax{ (f_{ij}) \setbar f_{ij} \in \mc F (U_i \cap U_j), \forall i,j \in I}
    \end{align*}
    Let $Z^1(\mc U,\mc F) \subset C^1(\mc U, \mc F)$ be the subgroup of \term[cocycle,cocycles,1-cocycle]{1-cocycles}, consisting of families $(f_{ij})$ for which
    \begin{align} \label{eq:cechcochain}
        f_{ij}+f_{jk}=f_{ik} \quad \text{on } U_i \cap U_j \cap U_k, \: \forall i,j,k \in I
    \end{align}
    Note that \eqref{eq:cechcochain} implies $f_{ii}=0$, and $f_{ij}=-f_{ji}$ for any cocycle $(f_{ij})$.

    Define a \term[coboundary maps,coboundary]{coboundary map} $\delta: C^0(\mc U,\mc F)\to C^1(\mc U,\mc F)$ by setting $\delta(f_i)_{i\in I}:=(f_{ij})$, where $f_{ij}:= f_j-f_i$ on $U_i \cap U_j$. Moreover, let
    \[ B^1(\mc U,\mc F) := \delta C^0(\mc U,\mc F) \subset Z^1(\mc U,\mc F). \]
    Then, the \term[H^1,\v{C}ech coholomogy group]{first \v{C}ech cohomology group $H^1(\mc U,\mc F)$} is
    \[
    H^1(\mc U,\mc F):= Z^1(\mc U,\mc F)/B^1 (\mc U, \mc F).
    \]
    Moreover, the \term[$H^0(\cU,\cF)$]{zeroth \v{C}ech cohomology group $H^0(\cU,\cF)$} is
    \[
    H^0(\mc U,\mc F):= \ker \delta \in C^0(\mc U,\mc F)
    \]
    A $0$-cochain $(f_i)$ belongs to $\ker \delta$ if and only if $f_{i}|_{U_i \cap U_j}=f_j|_{U _i \cap U_j}$ for all $i, j$. Since $\mc F$ is a sheaf, the $f_i$'s come from the restriction of a section $f \in \mc F(X)$. Hence, there exists a natural isomorphism
    \[
    H^0(\mc U,\mc F) \arr[\sim]\mc F(X),
    \]
    so $H^0(\mc U,\mc F)$ is independent of the open cover $\mc U$ of $X$.
\end{defn}

\begin{notethm*}[Goal]
    We aim to study, for any open sets $\Omega \subset \C$, the three groups $H^1(\mc U, \mc A), H^1(\mc U, \mc O)$, and $H^1(\mc U, \mc \C)$ for any open cover $\mc U$ of $\Omega$ by disks.
\end{notethm*}

\subsection{Partitions of Unity}

\begin{defn}
    A \term{partition of unity} relative to the open cover $\mc U:=\{U_i\}_{i \in I}$ of $\Omega \subset \C$ is a family $\{\varphi_i\}_{i \in I}$ of $C^\infty$ functions on $\Omega$, such that
    \begin{enumerate}[(i)]
        \item $\varphi_i:\Omega \to \R_{\geq 0}$ has $\supp \varphi_i \subset U_i$,
        \item the family of closed sets $\{\supp \varphi _i\}_{i \in I}$ is \term{locally finite}, i.e. for any compact set $K \subset \Omega$, the set $\brax{i \in I \setbar K \cap \supp \varphi_i \neq \emptyset}$ is finite
        \item $\sum_{i \in I}\varphi_i = 1$ on $\Omega$.
    \end{enumerate}
    Partitions of unity always exists for any open cover $\mc U$ of $\Omega$.
\end{defn}

\begin{ex}
    Let $F \subset \Omega$ be a closed subset and $U \subset \Omega$ be an open set containing $F$. Then, there exists $\varphi \in C^\infty (\Omega)$ such that $0 \leq \varphi \leq 1$, $\varphi|_F \equiv 1$, and $\varphi|_{\Omega \setminus U}\equiv 0$.
\end{ex}

\begin{proof}
    Set $U^1 := \Omega \setminus F$, and let $\{\varphi, \varphi'\}$ be a partition of unity relative to the cover $\{U,U'\}$ of $\Omega$. Then, $\varphi'|_F \equiv 0 \implies \varphi|_F \equiv 1$, so $\varphi$ satisfies the required conditions.
\end{proof}

\begin{thm}
    Let $\Omega \subset \C$ be open and $\mc U := \{U_i\}_{i \in I}$ be an open cover of $\Omega$. Then,
    \[
    H^1(\mc U,\mc A)=0
    \]
\end{thm}

\begin{proof}
    Suppose that $(f_{ij})\in Z^1(\mc U, \mc A)$, so $f_{ij}+ f_{jk}=f_{ik}$. We seek a family $(f_i)$ such that $f_{ij}=f_j-f_i$ on $U_i \cap U_j$ for all $i,j$.

    Let $(\varphi _i)_{i \in I}$ be a partition of unity relative to $\mc U$. Observe that the function $\varphi_if_{ij}$ can be defined on the set $U_j$ by setting $\varphi_if_{ij}(z):=0$ for $z \in U_j \setminus U_i$. Now, define
    \[
    f_j := \sum_{i \in I}\varphi_i f_{ij} \in \mc A(U_j).
    \]
    Then, for $j,k \in I$, we have on $U_j \cap U_k$ that
    \begin{align*}
        f_k-f_j = \sum_{i} \varphi_i(f_{ik}-f_{ij}) = \sum_{i}\varphi_i (f_{ji} + f_{ik})=f_{jk}\sum_i \varphi_i = f_{jk}.
    \end{align*}
    Hence, $\delta (f_i) = (f_{ij})$.
\end{proof}

\begin{defn}
A sheaf that admits of a partition of unity is called a \term{fine sheaf}. So, $\mc A$ is a fine sheaf.
\end{defn}


\begin{thm}[Mittag-Leffler's Theorem, or the 1$\st$ Cousin Problem]
    Let $\Omega$ be open in $\C$, and $\mc U$ an open cover of $\Omega$. Then,
    \[
    H^1(\mc U,\mc O)=0.
    \]
\end{thm}

\begin{proof}

Begin with a $1$-cocycle $(f_{ij})\in Z^1(\mc U,\mc O)$. Let $\alpha _i \in C^\infty (U_i)$ be such that $\alpha_j - \alpha _i = f_{ij}$ on $U_i \cap U_j$ for all $i,j$. Let $u_i := \partial \alpha _i / \partial \bar z$ on $U_i$. Then, $u_i - u_j=0$ on $U_i \cap U_j$, so there exists $u \in C^\infty (\Omega)$ such that $u|_{U_i}=u_i$ for all $i \in I$.

Now, by \Cref{thm:grothmagic}, we know there exists $A \in C^\infty (\Omega)$ such that $\partder{A}{\bar z}=u$ on $\Omega$. Define $(f_i)$ by $f_i := \alpha _i - A$ on $U_i$, for all $i \in I$. Then, on $U_i \cap U_j$, $f_j -f_i = \alpha _j -\alpha _i = f_{ij}$, and $\partder{f_i}{\bar z}=u_i-u=0$ on $U_i$ for all $i \in I$.
\end{proof}

Now, we aim to remove our choice of open cover from the definition of the \v{C}ech cohomology groups. Namely, we want to define $H^1(X, \mc F)$.

\begin{defn}
Let $\mc U:=\{U_i\}_{i \in I}$ and $\mc V:= \{V_i\}_{j \in J}$ be two open covers of $X$. Then, $\mc V$ is said to be a \term{refinement} of $\mc U$ if there exists a \term{refinement map} $\tau :J \to I$ such that $V_j \subset U_{\tau(j)}$ for all $j \in J$. Such a map $\tau$ induces a map
\[
\tau^* : H^1 (\mc U,\mc F) \to H^1 (\mc V,\mc F)
\]
sending $(f_{ij})\in Z^1(\mc U,\mc F)$ to $(g_{\alpha\beta})_{\alpha,\beta \in J}$, where $g_{\alpha \beta}:= f_{\tau(\alpha) \tau (\beta)}|_{V_\alpha \cap V_\beta}$.
\end{defn}

\begin{notethm*}[Check]
    $\tau^*$ depends only on the covering $\mc U$ and $\mc V$, and not the refinement map $\tau$.
\end{notethm*}

\begin{lem}
    If $\sigma,\tau : J \to I$ are two refinement maps, so $V_\beta \subset U_{\sigma(\beta)} \cap U_{\tau (\beta)}$ for all $\beta \in J$, then the induced maps
    \[
    \sigma^*,\tau^* : H^1(\mc U,\mc F) \to H^1(\mc V, \mc F)
    \]
    are equal. Moreover, if $\mc V$ is a refinement of $\mc U$, the (now natural) morphism
    \[
    H^1(\mc U,\mc F) \to H^1(\mc V ,\mc F)
    \]
    is injective.
\end{lem}

\begin{proof}
    Fix $(f_{ij}) \in Z^1(\mc U, \mc F)$. Then, on $V_{\beta _1}\cap V_{\beta_2}$, we have
    \[
    f_{\tau(\beta_1)\tau(\beta_2)}-f_{\sigma(\beta_1)\sigma(\beta_2)}= g_{\beta_1}-g_{\beta_2},
    \]
    where
    \[
    g_{\beta}=f_{\tau(\beta)\sigma(\beta)}|_{V_\beta}.
    \]
    Hence, we see that
    \[
    \brax{ f_{\tau(\beta _1)\tau(\beta_2)} - f_{\sigma(\beta _1)\sigma(\beta _2)}} \in B^1(\mc V, \mc F).
    \]
    This proves $\sigma^* = \tau^*$. Injectivity is left as an exercise.
\end{proof}

Suppose we have three coverings $\mc U,\mc V, \mc W$, with $\mc W < \mc V < \mc U$, where $<$ means refinement. Write, in general, $\tau_{\mc U,\mc V}$ for the induced map $\tau ^*$. Then,
\[
\tau_{\mc V,\mc W} \circ \tau_{\mc U, \mc V} = \tau_{\mc U, \mc W}.
\]
We can therefore define an equivalence relation $\sigma$ on the disjoint union $\coprod_{\mc U} H^1(\mc U,\mc F)$ over all open coverings by saying $\alpha \in H^1(\mc U,\mc F)$ is equivalent to $\beta \in H^1(\mc V, \mc F)$ if there exists a further refinement $\mc W < \mc U$ and $\mc W < \mc V$ such that
\[
\tau_{\mc U,\mc W}(\alpha)= \tau_{\mc V,\mc W}(\beta).
\]
The set of equivalence classes is in fact a group, and is the direct limit of the $H^1(\mc U,\mc F)$. We write this as
\[
H^1(X, \mc F):= \varinjlim_{\mc U} H^1(\mc U,\mc F)= \coprod_{\mc U}H^{1}(\mc U, \mc F)/\sim.
\]
Then, for any open covering $\mc U$ of $X$, the canonical map $H^1(\mc U,\mc F)\to H^1(X,\mc F)$ is injective. Thus, $H^1(X,\mc F)=0 \iff H^1(\mc U,\mc F)=0$ for all open coverings $\mc U$ of $X$.

There is a useful criterion for when an open cover $\mc U$ of $X$ is sufficient to compute $H^1(X,\mc F)$.

\begin{thm}[Leray's Theorem]
Let $\mc F$ be a sheaf of abelian groups on $X$, and $\mc U:= \{U_i\}_{i \in I}$ be an open cover of $X$ such that $H^{1}(U_i, \mc F)=0$ for all $i \in I$. Then,
\[
H^1(X,\mc F) \simeq H^1(\mc U,\mc F).
\]
\end{thm}

\begin{proof}
    It suffices to show that for any refinement $\mc V:= \{V_\alpha\}_{\alpha \in A}$ of $\mc U$, the induced map $\tau^*: H^1(\mc U,\mc F)\to H^1(\mc V,\mc F)$ is surjective. Here, $\tau:A \to I$ is a refinement map such that $V_\alpha \subset U_{\tau(\alpha)}$ for all $\alpha \in A$.

    Let $\{c_{\alpha \beta}\}_{\alpha,\beta \in A} \in Z^1(\mc V, \mc F)$. Then, 
    \[
    \brax{c_{\alpha \beta}|_{U_i}} \in Z^1(U_i \cap \mc V,\mc F).
    \]
    Since $H^1(U_i,\mc F)=0$, there exists a function $g_{i\alpha} \in \mc F(U_i \cap V_\alpha)$ such that $c_{\alpha \beta}=g_{i\alpha}-g_{i\beta}$ on $U_i \cap V_\alpha \cap V_\beta$. Then, on $U_i \cap U_j \cap V_\alpha \cap V_\beta$, we have
    \[
    g_{i\alpha}-g_{i\beta}=c_{\alpha \beta}=g_{j\alpha}-g_{j\beta} \implies g_{i\alpha}-g_{j\alpha}=g_{i\beta}-g_{j\beta}.
    \]
    Now, since $\mc F$ is a sheaf, there exists $\gamma_{ij} \in \mc F(U_i \cap U_j)$ such that $\gamma_{ij} = g_{i\alpha}-g_{i\beta}$ on $U_i \cap U_j \cap V_\alpha$ for all $\alpha \in A$. Clearly $\gamma_{ij}+\gamma_{jk}=\gamma_{ik}$ on $U_i \cap U_j \cap U_k$, and moreover 
    \[
    \gamma_{\tau(\alpha)\tau(\beta)}+c_{\alpha \beta}=\paren{g_{\tau(\alpha)\alpha}-g_{\tau(\beta)\beta}}+\paren{g_{\tau(b)\alpha }-g_{\tau (\beta)\beta}}= -g_{\tau(\beta)\beta} + g_{\tau(\alpha \alpha)}
    \]
    on $V_\alpha \cap V_\beta$, which is a subject of $U_{\tau(\beta)}$. Then, since $g_{\tau(\beta)\beta} \in \mc F(U_{\tau (\beta)}\cap V_\beta)=\mc F(V _\beta)$, this proves $(c_{\alpha \beta})$ and $(-\gamma_{\tau(\alpha)\tau(\beta)})$ induce the same element in $H^1(\mc V, \mc F)$. But, by definition,
    \[
    \paren{-\gamma _{\tau(\alpha)\tau(\beta)}}=\tau^*(-\gamma_{ij}).
    \]
    We conclude that $\tau^*$ is surjective, proving the theorem.
\end{proof}

\begin{cor}[Reformulated Mittag-Leffler][label=cor:mittag-leffler]
    For any open set $\Omega \subset \C$, we have
    \[
    H^1(\Omega, \mc O) = 0.
    \]
\end{cor}

\begin{defn}

    \[
    \mc F \arr[\alpha] \mc G \arr[\beta] \mc H

    \[

\end{defn}

\begin{ex}
Let $X = \C^*$, and for $U \subset X$ open, let $\Z_U$ be the set of locally constant integer valued functions on $U$, let $\mc O(U)$ be the set of holomorphic functions on $U$, and $\mc O^*(U)$ be the set of nowhere zero holomorphic functions on $U$.

We claim that the exponential sequence
\[
0 \to \Z \to \mc O \arr[\exp(2 \pi iz)] \mc O^* \to 0
\]
is exact.
\end{ex}

\begin{proof}
    For any open set $U \subset X$, the sequence
    \[
    0 \to \Z(U) \inj \mc O(U) \arr[\exp(2\pi i z)] \mc O^*(U)
    \]
    is exact. However, note that $\mc O(U) \to \mc O^*(U)$ need not be surjectitve: for instance, let $U = X = \C^*$, and take $z \in \mc O^*(U)$, but we know that an entire logarithm does not exist.

    However, for all $z \in \C^*$, $\exp:\mc O_z \to \mc O^*_z$ is surjective: indeed, for $z \neq 0$, the germ of functions near $z$ are holomorphic functions in a small connected domain away from the origin, where we can define a complect logarithm.
\end{proof}

The previous example can be generalized with the following result:
    \begin{exercise}[][label=exer:shortexacttoshortexact]
        Use the sheaf axioms to show that for any short exact sequence
        \[
        0 \to \mc F \arr[\alpha]\mc G \arr[\beta]\mc H \to 0
        \]
        of sheaves on $X$, the induced sequence
        \[
        0 \to \mc F(U) \arr[\alpha _U]G(U) \arr[\beta_U]\mc H(U)
        \]
        is exact for any $U \subset X$ open.
    \end{exercise}

\begin{thm}[][label=thm:longexact]
Consider a short exact sequence
\begin{align} \label{eq:shortexactseq}
    0 \to \mc F \arr[\alpha]\mc G \arr[\beta]\mc H \to 0
\end{align}
of sheaves on $X$. Then, the sequence naturally induces a long exact sequence of cohomology groups:
\[
0 \to H^0(X,\mc F) \arr[\alpha_X] H^0(X,\mc G)\arr[\beta_X]H^0(X,\mc H) \arr[\delta] H^1(X,\mc F) \arr[\alpha_X^1] H^1(X,\mc G)\arr[\beta_X^1]H^1(X,\mc H).
\]
\end{thm}

\begin{proof}
    Since $H^0(X,\mc F)=\mc F(X)$, and similar for $\mc G,\mc H$, the maps $\alpha_X$ and $\beta_X$ are already defined. To define $\alpha^1_X$, note that $\alpha :\mc F \to \mc G$ induces, for any open cover $\mc U$ of $X$, a map
    \[
    \alpha_\mc U : C^1(\mc U, \mc F) \to C^1(\mc U,\mc G)
    \]
    which sends cocycles to cocycles, and coboundaries to coboundaries. Hence, we obtain a map
    \[
    \alpha_{\mc U}^{1}:H^1(\mc U,\mc F) \to H^1(\mc U,\mc G).
    \]
    This in turn induces a map $\alpha^1_X$, and $\beta^1_X$ is defined similarly. 

    Next, we define the connecting homomorphism $\delta : \mc H(X) \to H^1(X, \mc F)$ as follows: let $h \in \mc H(X)$, and since $\beta :\mc G \to \mc H$ is surjective, there exists an open cover $\mc U:=\{U_i\}_{i \in I}$ of $X$, and elements $g_i \in \mc G(U_i)$ such that $\beta_{U_i}(g_i)=h|_{U_i}$ for all $i \in I$. Now,
    \[
    g_i-g_j|_{U_i \cap U_j}\in \ker(\beta_{U_i\cap U_j}) = \Img(\alpha_{U_i \cap U_j}),
    \]
    with the last equality coming from \Cref{exer:shortexacttoshortexact}. Hence, there exists $f_{ij}\in \mc F(U_i \cap U_j)$ with
    \[
    \alpha_{U_i \cap U_j}(f_{ij})=g_i - g_j|_{U_i \cap U_j}.
    \]
    Clearly, $f_{ij}+f_{jk}+f_{ki}$ maps to zero under $\alpha_{U_i \cap U_j \cap U_k}$. Thus, again by \Cref{exer:shortexacttoshortexact}, $f_{ij}+f_{jk}+f_{ki}=0$ on $U_i \cap U_j \cap U_k$, since the map is injective. Therefore, $\{f_{ij}\} \in Z^1(\mc U,\mc F)$, so it defines and elements $\delta(h)$ in $H^1(X,\mc F)$.

    Now, we need to check that $\delta(h)$ is independent of the choices of $\mc U$ and $\{g_i\}$. Indeed, if $\mc V:= \{V_\alpha\}_{\alpha \in A}$ is a refinement of $\mc U$ given by the map $\tau:A \to I$, and we choose elements $g_\alpha'=g_{\tau(\alpha)}|_{V_\alpha}$, we clearly obtain the same element of $H^1(X,\mc F)$. This is just the restriction of the cocycle $\{f_{ij}\}$ defined earlier to $\mc V$. Hence, we may consider the same covering $\mc U$, and possibly different $g_i' \in \mc G(U_i)$ with $\beta_{u_i}(g_i')=h|_{U_i}$ for all $i \in I$, As above, there exists $f_i \in \mc F(U_i)$ such that $\alpha_{U_i}(f_i)=g_i'-g_i$. Now, if $f_{ij}' \in \mc F(U_i \cap U_j)$ maps to $g_i'-g_j'$ on $U_i \cap U_j$, then
    \[
    \alpha_{U_i \cap U_j}(f_i - f_j)=\alpha_{U_i \cap U_j}(f'_{ij}-f_{ij}) \implies f_i -f_j = f_{ij}' -f_{ij}.
    \]
    Hence, $\{f_{ij}\}$ and $\{f_{ij}'\}$ define the same element of $H^1(X ,\mc F)$, so $\delta$ is well-defined.

    Proving exactness of the long exact sequence at $H^0(X ,\mc H), H^1(X,\mc F),$ and $H^1(X,\mc G)$ is left as an exercise.
\end{proof}

\begin{que}
When is the differentiation map $\frac{d}{dz}:\mc O(\Omega) \to \mc O(\Omega)$ surjective, for $\Omega \subset \C$ open?
\end{que}

\begin{ex}[The Cohomology Form of Cauchy's Theorem]
    Let $\Omega \subset \C$ be connected and open. Then, we have an exact sequence of sheaves on $\Omega$:
    \[
    0 \to \C \to \mc O \arr[d/dz] \mc O \to 0.
    \]
Then, since $H^1(\Omega,\mc O)=0$ by Mittag-Leffler (\Cref{cor:mittag-leffler}), we get an induced long exact sequence
\[
0 \to \C \to \mc O(\Omega)\to \mc O(\Omega) \to H^1(\Omega,\C)\to 0
\]
by \Cref{thm:longexact}. Thus, $H^1(\Omega,\C)$ measures the obstruction to solving the equation $f'=g$ for $g \in \mc O(\Omega)$. If $\Omega$ is simply connected, then $d/dz$ is onto and $H^1(\Omega, \C)=0$. 

In fact, one can show
\[
H^1(\Omega,\C) \simeq H^1_{\w{sing}}(\Omega,\C),
\]
where the RHS is singular cohomology.
\end{ex}

\begin{rmk}
    See the textbook for a definition of higher cohomology groups $H^k(X,\mc F)$ for any $k \geq 0$. One can show that the long exact sequence of cohomology groups extends further:
\begin{align*}
    0 \to \mc F(X) \to \mc G(X) \to \mc H(X) \to H^1(X, \mc F) \to H^1 (X,\mc G) \to H^1(X, \mc H)
    \\ \to H^2(X,\mc F) \to H^2(X,\mc G) \to H^2(X,\mc H) \to H^3(X,\mc F)\to \cdots.
\end{align*}
\end{rmk}

\section{Vector Bundles, Line Bundles, and Divisors}

Let $X, E$ be topological spaces, and $p:E \to X$ a continuous map. We assume that for every $x \in X$, the fiber $E_x:=p^{-1}(x)$ has the structure of an $n$-dimensional complex vector space.

\begin{defn}
 The map $p:E \to X$ is called a \term[vector bundles]{vector bundle} of rank $n$ over $X$ if every point $a \in X$ has a neighborhood $U \ni a$ such that there is a homeomorphism $\varphi:p^{-1}(U) \to U\times \C^n$, called a \term[local trivializations]{local trivialization} of $E$ over $U$, which is
 \begin{enumerate}[(i)]
    \item \term{fiber preserving}: the diagram

\[\begin{tikzcd}[ampersand replacement=\&]

	\& U
	\arrow["\varphi", from=1-1, to=1-3]
	\arrow["p"', from=1-1, to=3-2]
	\arrow["{\w{pr}_1}", from=1-3, to=3-2]
\end{tikzcd}\]
commutes, and
\item for each $x \in U$ the map $\varphi|_{E_x}$ is a vector space isomorphism
\[
E_x \arr[\sim] \{x\} \times \C^n \equiv \C^n.
\]
 \end{enumerate}
An \term{atlas} for $E$ is a family $\brax{(U_i,\varphi_i)}_{i \in I}$ of local trivializations such that $\bigcup_{i \in I}U_i = X$.

A vector bundle of rank $1$ is called a \term[line bundles]{line bundle}.
\end{defn}

\begin{defn}
    A vector bundle $E \to X$ is said to be \term{trivial} if it has a linear trivialization  $E \arr[\sim] X \times \C^n$. Hence, any vector bundle is \textit{locally} trivial by definition.
\end{defn}

\begin{prop}[][label=prop:transition-funcs]
    Suppose that $E \arr[p]X$ is a rank $n$ vector bundle over a topological space $X$, and suppose
    \[
    \varphi_i :p^{-1}(U_i) \equiv E_{U_i}\to U_i \times \C^n
    \]
    is an atlas on $E$. Then, there exists unique continuous maps
    \[
    g_{ij}:U_i \cap U_j \to \GL_n(\C) \subset \C^{n^2}\equiv M_n(\C).
    \]
    such that the morphisms $\varphi_{ij}:= \varphi_i \circ \varphi_j^{-1}:(U_i \cap U_j)\times \C^n \to (U_i \cap U_j)\times \C^n$ satisfy
    \[
    \varphi_{ij}(x,t) = (x,g_{ij}(x)t),\qquad \forall (x,t) \in U_i \cap U_j \times \C^n.
    \]
    On $U_i \cap U_j \cap U_k$, the equation $g_{ij}g_{jk}=g_{ik}$ holds. The $\{g_{ij}\}$ are called \term[transition function]{transition functions} for the vector bundle.
\end{prop}

\begin{proof}
    The map $\varphi_i \circ \varphi_j^{-1}$ is a fiber preserving homeomorphism which, when restricted to each fiber, is a vector space isomorphism. Hence, for all $x \in U_i \cap U_j$, there exists a matrix $g_{ij}(x) \in \GL_n(\C)$ with $\varphi_{ij}(x,t)=(x,g_{ij}(x)t)$. Since $\varphi_{ij}$ is a homeomorphism, the map $x \mapsto g_{ij}(x)$ is continuous on $U_i \cap U_j$.
\end{proof}

\begin{rmk}
    The converse of \Cref{prop:transition-funcs} also holds. Namely, given a family $\{g_{ij}\}$ of continuous maps $U_i \cap U_j \to \GL_{n}\C$ such that $g_{ij}g_{jk}=g_{ik}$ on $U_i \cap U_j \cap U_k$, we can construct a vector bundle with these transition functions.
\end{rmk}

\begin{proof}
    Set $E':= \coprod_{i \in I}U_i \times \C^n \times \{i\} \subset X \times \C^n \times I$, with the topology induced from $X \times \C^n \times I$, where $I$ has the discrete topology. Let $(x,t,i)\sim (x',t',j)$ if and only if $x=x'$, and $t=g_{ij}(x)t'$. Then, the cocycle condition $g_{ij}g_{jk}=g_{ik}$ implies that $\sim$ is an equivalence relation on $E'$. Define $E:=E'/\sim$, with the projection $p:E \to X$. If $h: E' \to E$ is the natural quotient map $h|_{U_i \times \C^n \times I}\to E_{U_i}$ is a homeomorphism, and we define $\varphi_i$ to be the inverse of $h|_{U_i\times \C^n \times \{i\}}$.
\end{proof}

\begin{defn}
    Let $X$ be a Riemann surface (or more generally a complex manifold), and $E\to X$ be a rank $n$ vector bundle over $X$. Two local trivializations $(U_i, \varphi_i)$ and $(U_j, \varphi_j)$ are said to be \term{compatible} if the associated transition function $g_{ij}:U_i \cap U_j \to \GL_n(\C)$ is holomorphic.
\begin{notethm}[Remark]
    If $\Omega \subset \C^m$ is open, then a map $f: \Omega \to \C^n$ is \term{holomorphic} if $f=(f_1,\dots,f_n)$, with each $f_i :\Omega \to \C$ holomorphic.
\end{notethm}
If $X$ has a covering by a compatible trivializations for $E$, in which case we say $E$ has a \term{holomorphic atlas}, then $E$ is called a \term[holomorphic line bundle]{holomorphic vector bundle} over $X$.
\end{defn}

\begin{rmk}
    If $L \to X$ is a holomorphic line bundle with transition functions $\{g_{ij}\}$ with respect to an open cover $\cU := \{U_i\}_{i\in I}$, then we obtain an associated element of $H^1(X,\mc O^*)$. Indeed, $\GL_1\C \equiv \C^*$, and $g_{ij} \in \cO^*(U_i \cap U_j)$ satisfy the cocycle condition on $U_i \cap U_j \cap U_k$.
\end{rmk}

\begin{defn}
    Let $E \arr[p]X$ and $E' \arr[p']X$ be two vector bundles over a Riemann surface $X$. A \term{morphism} $u:E \to E'$ is a map such that the diagram
    % https://q.uiver.app/#q=WzAsMyxbMCwwLCJFIl0sWzEsMSwiWCJdLFsyLDAsIkUnIl0sWzAsMiwidSJdLFswLDEsInAiLDJdLFsyLDEsInAnIl1d
\[\begin{tikzcd}[ampersand replacement=\&]
	E \&\& {E'} \\
	\& X
	\arrow["u", from=1-1, to=1-3]
	\arrow["p"', from=1-1, to=2-2]
	\arrow["{p'}", from=1-3, to=2-2]
\end{tikzcd}\]
commutes (i.e. $p'u =p$), and $u:p^{-1}(x) \to (p')^{-1}(x)$ is $\C$-linear for all $x \in X$.

The morphism $u$ is called continuous/$C^\infty$/holomorphic if the function $u$ has this property. We use this notion to define \term{isomorphic vector bundles}. 
\end{defn}

\begin{ex}
    $E$ is trivial if and only if $E$ is isomorphic to the trivial vector bundle $X \times \C^n$.
\end{ex}

\section{Sections of Vector Bundles}

\begin{defn}
    If $p:E \to X$ is a vector bundle, a \term[sections]{section} $E$ is a morphism $s:X \to E$ such that $s(x) \in E_x$ for all $x \in X$. Such sections can be continuous/$C^\infty$/holomorphic as before.
\end{defn}

If $\varphi_i: p^{-1}(U_i) \to U_i \times \C^n$ are local trivialization maps for $E$, then $\varphi_i \circ s(x)=(x,f_i(x))$ for all $x \in U_i$, where $f_i :U_i \to \C^n$ is a continuous/$C^\infty$/holomorphic map.

On $U_i \cap U_j$, we have 
\begin{align}
    \varphi_i \circ s(x) = (x,f_i(x))=\varphi_i \circ \varphi_j^{-1}(x,f_j(x))=(x,g_{ij}(x)f_j(x)) \notag
    \\ \implies f_i(x)=g_{ij}(x)f_j(x), \qquad \forall x \in U_i \cap U_j. \label{eq:sectioncondition}
\end{align}
Hence, a section $s$ of $E$ may be identified with a family $\{f_i\}_{i \in I}$ of functions $f_i:U_i \to \C^n$ such that \eqref{eq:sectioncondition} holds for all $i,j \in I$. The sections of $E$ over any open set $U \subset X$ are defined analogously.

\begin{ex}
    A section of the trivial vector bundle $E \to X$ is just a function $f:X \to \C^n$.
\end{ex}

\begin{defn}
    Let $X$ be a Riemann surface and $\pi :E \to X$ be a holomorphic vector bundle on $X$. Let $S \subset X$ be a discrete set, and $s:X\setminus S \to E$ be a holomorphic section of $E$ over $X \setminus S$. Then, $s$ is called a \term[meromorphic sections]{meromorphic section} if, for all $a \in S$, there exists a neighborhood $U$ of $a$ and a coordinate $z$ on $U$ with $z(a)=0$ such that $U \cap S = \{a\}$, and for some $N>0$, $z^Ns$ is the restriction to $U \setminus \{a\}$ of a holomorphic section of $E$ over $U$.
\end{defn}

\begin{defn}
    The \term[\cE]{sheaf of holomorphic sections $\cE$} of $E$ is defined by the group 
    \[
    \cE(U)=\Gamma(U,E):=\{\text{holomorphic sections of } E \text{ over } U\}.
    \] 
\end{defn}

\section{Divisors and Line Bundles}

\begin{defn}
    Let $X$ be a Riemann surface. A \term[divisors]{divisor} on $X$ is a map $D: X \to \Z$ such that the support of $D$ is locally finite. That is, for every compact $K \subset X$,
    \[
    \{p \in X \mid D(p) \neq 0\} \cap K \text{ is finite.}
    \]
\end{defn}
We assume now that $X$ is compact so the support of $D$ is finite. 
\begin{defn}
    The sum and difference $D_1+D_2$ and $D_1-D_2$ of two divisors are defined in the obvious way.

    A divisor $D$ is said to be \term{effective} if $D(p) \geq 0$ for all $p \in X$. We write $D_1 \geq D_2$ if $D_1 -D_2$ is effective. The set $\{p \in X \mid D(p) \neq 0\}$ is called the \term{support} of $D$, denoted $|D|$ or $\supp (D)$.
\end{defn}

\begin{notethm}[Recall]
If $f:U \to \P^1$ is a meromorphic function on a neighborhood $U$ of a point $a \in X$, and $z$ is a local coordinate on $U$ with $z(a)=0$, we set 
\[
\ord_a(f):=\case{ \infty & f \equiv 0 \text{ near }a, \\ k & f(z) = \sum_{n=k}^\infty a_nz^n, a_k \neq 0. }
\]
\end{notethm}
Suppose now that $s$ is a meromorphic section of a holomorphic vector bundle $E$ on a Riemann surface $X$, with $s \not \equiv 0$. For $a \in X$, choose a coordinate neighborhood $(U,z)$ with $z(a)=0$, and a local trivialization $\varphi:E|_U \to U \times \C^n$. Then, $(\varphi\circ s)(x) = (x,f(x))$, $x \in U$, where $f=(f_1,\dots,f_n)$, is an $n$-tuple of meromorphic functions on $U$. There is a unique integer $k$ such that $f = z^kg$, where $g(a) \neq 0$, and $g=(g_1,\dots,g_n)$ with each $g_i$ holomorphic near $a$.

\begin{defn}
    We define $\ord_a(s):=k$, where $k$ is the constant above.
\end{defn}

\begin{exercise}
    The definition of $\ord_a(s)$ is well-defined, i.e. independent of the choice of chart and local trivialization.
\end{exercise}

\begin{defn}
    The \term{divisor} of $s$ is the function $a \mapsto \ord_a(s)$. That is, 
    \[
    \div(s) = \sum_{a \in X}\ord_a(s)[a]. 
    \]
    Note that this also defines the divisor $\div(f)$ of a meromorphic function $f$ on $X$, since any such $f$ is a meromorphic section of the trivial line bundle $X \times \C \to X$.
\end{defn}

\begin{lem}
    Let $D:= \sum n_p[p]$ be a divisor on a compact Riemann surface $X$. Then, $D=\div(s)$ for a meromorphic section $s$ of a line bundle on $X$, denoted $L(D)$.
\end{lem}

\begin{proof}
    To construct $L(D)$ and $s$ from $D$, let $S:= |D|$ and $(U_P,z_P)$ be a local coordinate chart at $P \in S$ with $z_p(P)=0$. Assume that $U_P \cap U_Q = \emptyset$ for $P \neq Q$ in $S$. Set $U_0 := X\setminus S$, $f_0 \equiv 1$ on $U_0$, and let $f_P:=z_P^{N_p}$ on $U_P$, for all $P \in S$. Define $I := \{0\} \sqcup S$, and let $g_{ij}:= f_i/f_j$ for all $i,j \in I$, with $g_{ij}$ defined on $U_i \cap U_j$. Then, for $i,j$ with $U_i \cap U_j \neq \emptyset$, $g_{ij} \in \cO^*(U_i \cap U_j)$, and $g_{ij}g_{jk}=g_{ik}$ on $U_i \cap U_j \cap U_k$. So, the $\{ g_{ij} \}$ are the transition functions for a line bundle $L(D)$ on $X$. Since $f_{i}=g_{ij}f_j$ on $U_i \cap U_j$, the $f_i$'s define the meromorphic section $s_D$ of $L(D)$.

    We now have $\ord_a (s_D) = \ord_a (f_0)=0$ if $a \in X\setminus S$, and $\ord_a(s_D)=\ord_p(z_P^{n_P})=n_P$, if $a = P \in S$. Hence, $\div (s_D)=D$.
\end{proof}

\begin{notebox*}[Overview]
    Let $L \to P$ be a line bundle on a Riemann surface, and $\{U_i\}_{i \in I}$ is a trivializing open cover of $X$. Then, we can identify $L$ with a family of transition functions $\{g_{ij}\}$, with $g_{ij} \in \cO^*(U_i \cap U_j)$, and $g_{ij}g_{jk}=g_{ik}$ on $U_i \cap U_j \cap U_k$. Moreover, if $s:X \to L$ is a section, we can identify $s$ with $\{f_{i}\}$ such that $f_i$ is a function on $U_i$, and $f_i = g_{ij}f_j$ on $U_i \cap U_j$.
\end{notebox*}

\begin{rmk}
    Let $D = \sum n_p[p]$ be a divisor on $X$. This associates to $L(D),s_D$ such that $\div(s_D)=D$. The line bundle $L(D)$ is unique up to isomorphism.
\end{rmk}

\begin{proof}
    If $(\mc U_P,\zeta_P)$ are different local coordinate charts, then $h_P:=(\zeta_P/z_P)^{n_p}\in \mc O^*(U_P)$ and $h_0:=1$ implies
    \[
    g_{ij}^{\w{new}}=h_ig_{ij}^{\w{old}}h_j^{-1}.
    \]
\end{proof}

\begin{defn}
    The section $s_D$ is called the \term{standard section} of $L(D)$.
\end{defn}

\begin{rmk}[][label=rmk:sheaf-on-divisor]
    \begin{enumerate}
        \item If $s$ is a meromorphic section of a vector bundle $E \to X$, $s \not \equiv 0$, then $s$ is holomorphic if and only if $\div(s)$ is effective (so $s$ may have zeros, but no poles).
        \item If $D$ is a divisor on $X$, and $U \subset X$ is open, we obtain a sheaf structure \term{$\cO_D$} on $X$ by defining \[
        \mc O_D(U):=\brax{ f \text{ meromorphic on } U \setbar \div(f) \geq -D   }
        \] Moreover, if $\Gamma(U,L(D))$ denotes the vector space of holomorphic sections of $L(D)$ over $U$, and $s_D$ is the standard section of $L(D)$ with $\div(s_D)=D$, then the map \begin{align*}
            \mc O_D(U) &\to \Gamma(U,L(D)) \\ f &\mapsto fs_D
        \end{align*}
        is an isomorphism. Hence the sheaf of holomorphic sections of $L(D)$ can be canonically identified with $\mc O_D$.
        \item If $L \to X$ is any holomorphic line bundle, and $s_0,s_1$ are two meromorphic sections of $L$, $s_0 \not \equiv 0$, then there exists a meromorphic function $f$ on $X$ with $s_1 =f s_0$. Indeed, $s_1/s_0$ is well-defined independent of any local trivialization, since $f_i^1 = g_{ij}f_j^1$ and $f_i^0=g_{ij}f_{j}^0$ on $U_i \cap U_j$, so $f_i^1/f_j^1 = f_i^0/f_j^0$.
    \end{enumerate}
\end{rmk}

\section{Linear Equivalence of Divisors}

\begin{defn}
    Two divisors $D_1$ and $D_2$ are \term{linearly equivalent} if there exists a meromorphic function $f$ on $X$, not identically zero, such that $\div f= D_1 - D_2$. We write $D_1 \sim D_2$ when this is the case. 
\end{defn}

\begin{prop}
    We have $D_1 \sim D_2$ if and only if the line bundles $L(D_1)$ and $L(D_2)$ are holomorphically isomorphic.
\end{prop}

\begin{proof}
    Suppose $D_1 -D_2 = \div f$ for some $f \in \mc M(X)^*$. Let $s_{D_1}$ and $s_{D_2}$ be the standard sections of $L(D_1)$ and $L(D_2)$. 
    \begin{claim}
        We claim that the map $u:=s_{D_1}\mapsto s_{D_2}:L(D_1)\to L(D_2)$ is a unique isomorphism.
    \end{claim}
    \begin{proof}
     Indeed first define $u$ outside $|\div f| \cup |D_1| \cup |D_2|$ by \[
    \lambda s_{D_1}(x) \mapsto \lambda f(x)s_{D_2}(x), \quad \lambda \in \C.
    \] 
    Then, $u$ extends holomorphically to all of $X$, because on any open set $U \subset X$, the section $x \mapsto \lambda (x)s_{D_1}(x)$ is holomorphic if and only if the section $x \mapsto \lambda (x)f(x)s_{2}(x)$ is holomorphic (since $\div(s_{D_1}) \cap U=\div (fs_{D_2})\cap U$).
    \end{proof}
    Conversely, if $u:L(D_1) \to L(D_2)$ is an isomorphism, then $u \circ s_{D_1}$ is a meromorphic section of $L(D_2)$. If $f$ is defined by $u \circ s_{D_1}=fs_{D_2}$, then $\div f = D_1 - D_2$ since $\ord_a(s_{D_1})=\ord_a(u \circ s_{D_1})$ for all $a \in X$.
\end{proof}

\begin{rmk}
    In terms of the sheaves $\mc O_{D_1}$ and $\mc O_{D_2}$, the above isomorphism is just the map $\mc O_{D_1}(U) \to \mc O_{D_2}(U)$ sending $\varphi \mapsto f _\varphi$.
\end{rmk}

\section{Linear Operations on Vector Bundles}

Linear operations on vector spaces/matrices can by bundle-ized, provided that they are functorial. This means that they extend to give operations on vector bundles/transition functions.

\begin{ex}
    Suppose $E \to X$ and $E' \to X$ are two vector bundles over the same base $X$, and $\{g_{ij}\}$ and $\{g'_{ij}\}$ are transition functions for $E$ and $E'$ with respect to the same trivializing open cover of $X$.
    \begin{enumerate}
        \item \term{Direct sum}: The direct sum $E \oplus E'$ of $E$ and $E'$ is a vector bundle on $X$ such that $(E \oplus E')_x = E_x \oplus E'_x$, and the transition functions $\{g_{ij}^{E \oplus E'} \}$ are given by \[
        g_{ij}^{E \oplus E'} = g_{ij} \oplus g'_ij = \matr{g_{ij} & 0 \\ 0 & g_{ij}'}
        \]
        \item \term{Tensor product}: The tensor product $E \otimes E'$ has $(E \times E')_x = E_x\otimes E'_x$ and $g_{ij}^{E \times E'}=g_{ij} \otimes g_{ij}'$. In particular, if $L$ and $L'$ are two line bundles on $X$, then $L \otimes L'$ has transition functions $\{g_{ij}\cdot g'_{ij}\}$. Observe that this is the group operation on $H^1(X,\cO^*)$.
        \item \term[dual vector bundle]{Dual vector bundle $E^*$}: If $E \arr[p] X$ is a holomorphic vector bundle, and $E^*:= \coprod_{a \in X}E_a^*$, we make $E^*$ into a vector bundle as follows: for a trivialization $\varphi_U:p^{-1}(U) \to U \times \C^n$ of $E$ over $U \subset X$, define $\tilde \varphi_U :\coprod_{a \in U}E_a^* \to U \times \C^n$ by 
        \[
        \tilde \varphi_U(v) = \paren{ a, (\varphi_{U,a})^{-1}(v)}, \quad v \in E_a^*.
        \] 
        It follows that if $\{g_{ij}\}$ are the transition functions of $E$ with respect to an open cover, then the transition functions for $E^*$ are $\brax{ (g_{ij}^t)^{-1} }$.
        \item \term[exterior power,symmetric power]{Exterior and symmetric powers}: We have the $k$'th symmetric power $S^kE$, and the $k$'th exterior power $\Wedge^kE$, with transition functions $\{S^kg_{ij}\},\brax{\Wedge ^kg_{ij}}$ respectively.
        \item \term[Hom bundle]{Hom bundle}: We have the vector bundle $\Hom(E,E')\to X$, defined via the isomorphism $\Hom(E,E') \simeq E^* \otimes E'$.
    \end{enumerate}
\end{ex}

\begin{exercise}
    Using local trivializations, show how a global section of $\Hom(E,E')$ can be identified with a morphism $E \to E'$ of vector bundles.
\end{exercise}

\begin{rmk}
For divisors $D,D'$, we have
\begin{align*}
    L(D)\otimes L(D') &\simeq L(D+D'), \\ L(D)^* &\equiv L(-D).
\end{align*}
\end{rmk}

\begin{lem}
    A line bundle $L \to X$ is trivial if and only if $L$ has a holomorphic section $s$ with $s(x) \neq 0$, for all $x \in X$.
\end{lem}
\begin{proof}
    The map $X \times \C \to L$ sending $(x,\lambda)\mapsto \lambda s(x)$ is an isomorphism.
\end{proof}

\begin{cor}
If $L \to X$ is any line bundle, then $L \otimes L^*$ is a trivial line bundle on $X$. 
\end{cor}
\begin{proof}
    If $v \in L_x$, $v \neq 0$, then there exists a unique $\ell \in L_x^*$ such that $\ell(v) = 1$. Then, $x \mapsto v \otimes \ell$ is a nowhere vanishing section of $L \otimes L^*$ (indeed, the linear form corresponding to $cv$ is $\frac{1}{c}\ell$ for $c \in \C^*$).
\end{proof}

\begin{rmk}
    Let $L$ be a holomorphic line bundle on $X$, and $s$ be a meromorphic section of $L$, $s \not \equiv 0$. Then $L \simeq L(D)$, where $D = \div(s)$.
\end{rmk}

\begin{proof}
    If $s_{-D}$ is the standard section of $L(-D)$, then $s \otimes s_{-D}$ is a nowhere vanishing holomorphic section of $L \otimes L(-D)$. Hence, $L \otimes L(-D)\simeq \mc O$ (the trivial line bundle), so $L \simeq \mc O \otimes L(D) \simeq L(D)$.

    Alternatively, note that the isomorphism $\lambda s(x) \mapsto \lambda s_D(x)$, which is defined outside $|D|$, extends to an isomorphism $L \to L(D)$.
\end{proof}

\begin{defn}
    Let $\bm{Z^1(X)}$ denote the abelian group of all divisors on $X$ (sometimes denoted $\Div(X)$), and let $\bm{A^1(X)} := Z^1(X)/\sim$, where $\sim$ denotes linear equivalence. Let $\bw{Pic}\bm{(X)}$ be the group of isomorphism classes of line bundles on $X$, under the tensor product, called the \term{Picard group}.
\end{defn}

\begin{rmk}[Homework exercise]
    There is a natural isomorphism
    \begin{align*}
        \Pic(X) &\arr[\sim] H^1(X, \cO^*) \\ [L] & \mapsto \{g_{ij}\}
    \end{align*}
    where $\{g_{ij}\}$ are the transition functions of $L$ with respect to a trivializing open cover of $X$.
\end{rmk}

\begin{rmk}
    There is a morphism 
    \begin{align*}
        \varphi: Z^1(X) & \to \Pic(X) \\ D & \mapsto [L(D)].
    \end{align*}
    This passes to linear equivalence, so we get an induced map
    \[
    \tilde \varphi :A^1(X) \to \Pic(X).
    \]
\end{rmk}

\begin{thm}
    The map $\tilde \varphi$ above is surjective, so $\Pic(X) \simeq A^1(X)$ for $X$ a compact Riemann surface.
\end{thm}

    The proof of the claim relies on proving that any holomorphic line bundle $L \to X$ has a not identically zero meromorphic section.
\end{proof}

\section{Finiteness Theorems}

We now want to prove some facts about finiteness in cohomology. We first need some facts from complex analysis.

\begin{thm}[Weierstrass's Theorem]
    Let $\Omega \subset \C$ be an open set and $\{f_n\}_{n \ge 1}$ be a sequence of holomorphic functions on $\Omega$. Suppose $\{f_n\}$ converge uniformly on compact subsets of $\Omega$ to a function $f:\Omega \to \C$. Then, $f \in \cO(\Omega)$, and $f_n' \to f'$ uniformly on compact subsets of $\Omega$.
\end{thm}

\begin{thm}[Montel's Theorem]
    Let $\Omega \subset \C$ be open, and $\cF \subset \cO(\Omega)$ be a bounded family, i.e. for any compact set $K \subset \C$, there exists $M=M_K$ such that $|f(z)| \leq M$, for all $z \in K$ and $f \in \cF$. Then, every sequence $\{f_n\}$ of elements of $\cF$ contains a subsequence $\{f_{n_k}\}$ that converges uniformly on compact subsets of $\Omega$. That is, $\cF$ is \term{precompact}.
\end{thm}

We can now state some important finiteness theorems.

\begin{thm}
Let $X$ be a compact Riemann surface, and $\pi:E \to X$ be a holomorphic vector bundle on $X$. Then, the space $H^0(X,E)$ of global holomorphic sections of $E$ on $X$:
\[
H^0(X,E):= \brax{ s:X \to E \setbar \pi \circ s=\id_X, s\text{ holomorphic}}
\]
is a finite dimensional $\C$-vector space.
\end{thm}

\begin{proof}
    Choose finitely many coordinate neighborhood $\{(U_i,z_i)\}_{1 \leq i \le N}$ on $X$ such that
    \begin{enumerate}[(a)]
        \item $z_i:U _i \to \Delta := \brax{z \in \C \setbar |z| < 1}$ is a biholomorphism.
        \item If $V_i := z_i^{-1}\paren{D(0,\frac{1}{2})}$, then $\bigcup_{i=1}^NV_i =X$
        \item There are neighborhoods $W_i$ of $\bar{U_i}$, and trivializations of $E$: \[
        \varphi_i:\pi^{-1}(W_i) \to W_i \times \C^n
        \]
        with corresponding transition functions $g_{ij}:W_i \cap W_j \to \GL_n\C$.
    \end{enumerate}
    Any section $s \in H^0(X,E)$ is represented by a family $\{f_i\}_{1 \leq i \leq n}$ of holomorphic functions $f_i:W_i \to \C^n$ such that $f_i(x) = g_{ij}(x)f_j(x)$, for all $x \in W_i \cap W_j$. Let
    \[
    \norm{s}_U:= \max_i\sup_{x \in U_i}\abs{f_i(x)}, \qquad \norm{s}_V := \max_i \sup_{x \in V_i}|f_i(x)|
    \]
    \begin{rmk}
        One can quickly conclude the proof as follows: the space $\mc X:=(H^0(X,E),\norm{\cdot}_U)$ is a Banach space, where completeness of the norm follows from Weierstrass's theorem. Then, by Montel's theorem, the unit ball
        \[
        B_{\mc X}(0,1) := \brax{ s \in H^0(X,E)\setbar \norm{s}_U\leq 1  }
        \]
        is sequentially compact, hence compact. But a Banach space with a compact unit ball is finite dimensional.
    \end{rmk}
    That finishes the proof, but there's a more explicit proof due to Siegel, which generalizes to higher dimensions as well:
    \begin{claim}
        There exists a constant $C>0$ such that $\norm{s}_U \leq C\norm{s}_V$, for all $s \in H^0(X,E)$.
    \end{claim}
    \begin{proof}
        Let $x_0 \in \bar{U_i}$ be such that $|f_i(x_0)|=\norm{s}_U$. Choose $j$ so that $x_0 \in V_j$. Then we have
        \[
        \abs{f_i(x)}=\abs{g_{ij}f_j(x_0)} \leq C\abs{f_j(x_0)} \leq C\norm{s}_V, \qquad C= \max_{1 \leq i,j \leq N}\brax{ \sup_{x \in U_i \cap U_j} \norm{g_{ij}(x)} },
        \]
        which proves the claim
    \end{proof}
    \begin{claim}[Schwartz' Lemma][label=lem:schwartz]
        Let $s \in H^0(X,E)$, and suppose that $\ord_{a_i}(s) \geq k$, where $k \geq 0$ is a fixed integer, and $a_i \in U_i$ is such that $z_i(a_i)=0$, for all $i \in \{1,\dots,n\}$. Then, $\norm{s}_V \leq 2^{-k}\norm{s}_U$.
    \end{claim}
    \begin{proof}
        In fact $f_i/z_i^k$ is holomorphic on $U_i$ for all $i$, hence by the maximum principle
        \[
        \sup_{V_i}\abs{\frac{f_i}{z_i^k}} \leq \sup_{\partial U_i} \abs{\frac{f_i}{z_i^k}} = \sup_{\partial U_i}|f_i| \leq \norm{s}_U.
        \]
        Hence if $x \in V_i$, then
        \[
        \abs{f_i(x)} \leq \sup_{z_i \in V_i}\paren{|z_i|^k\abs{\frac{f_i}{z_i^k}} } \leq 2^{-k}\cdot \norm{s}_U,
        \]
        proving the claim.
    \end{proof}
    We deduce that if $s \in H^0(X,E)$ satisfies $\ord_{a_i}(s) \ge k$ for $i \in \{1,\dots,n\}$, then \[
    \norm{s}_U \leq C\norm{s}_V \leq \frac{C}{2^k}\norm{s}_U.
    \]
    Thus, if $2^k>C$, we deduce that $s=0$. Now, let $\cO_{a_i}$ denote the ring of germs of holomorphic functions at $a_i$, and let $\mf m_i^k$ be the principal ideal in $\cO_{a_i}$ generated by $z_i^k$. Then $\cO_{a_i}/\mf m_i^k$ is a $\C$-vector space of dimension $k$. If $2^k>C$, the map
    \begin{align*}
        H^0(X,E) &\arr[\varphi]\bigoplus_{i=1}^N \C^n \otimes \paren{\cO_{a_i}/\mf m_i^k}
        \\ s &\mapsto \bigoplus _i \paren{f_i \mod z_i^k}
    \end{align*}
    is injective, since
    \[
    \ker \varphi = \brax{s \setbar \ord_{a_i}(s) \ge k, \forall \: i \in \{1,\dots,N\} }.
    \]
    This proves the theorem.
\end{proof}

\begin{thm}[][label=thm:H1-is-finite]
    Let $X$ be a compact Riemann surface, and $\pi:E \to X$ be a holomorphic vector bundle on $X$. Then, $H^1(X,E)$ is a finite dimensional $\C$-vector space.
\end{thm}

\begin{proof}
    Suppose that $(U,\varphi_U)$ is a local trivialization of $E$, so that $\varphi_U:\pi^{-1}(U) \to U \times \C^n$ is biholomorphic. Suppose $V \subset U$ is a relatively compact open subset of $U$ (meaning $\bar V \subset U$ is compact), and let $E_b(V)$ denote the space of bounded holomorphic sections of $E$ on $V$, i.e. all $s \in \Gamma(V,E)$ with $\varphi_u(s(x)) =(x,f(x))$ where $x \in V$, $f(x) \in \C^n$, and $\sup_{x \in V}\abs{f(x)}< \infty$. Let $\norm{s}_V:=\sup_{x \in V}\abs{f(x)}$ for $s \in E_b(V)$. This norm makes $E_b(V)$ into a Banach space.
    \begin{notethm*}
        A different local trivialization $\varphi_U^1:\pi^{-1}(U) \to U \times \C^n$ gives rise to an equivalent norm on $E_b(V)$, so our choice doesn't matter.
    \end{notethm*}
    \begin{claim}
        If $U \simeq \Omega \subset \C$, with $\simeq$ denoting biholomorphism, then $H^1(U,E)=0$.
    \end{claim}
    \begin{proof}
        Mittag-Leffler's theorem (\Cref{cor:mittag-leffler}) asserts that $H^1(\Omega, \cO)=0$, where $\cO$ is the sheaf of holomorphic functions. Now, if $U \simeq \Omega \subset \C$ and $\varphi_U: \pi^{-1}(U) \to U \times \C^n$ is a trivialization, then
        \[
        H^1(U,E) = H^1\paren{ U, \bigoplus^n\cO } \simeq \bigoplus^n H^1(\Omega, \cO) = 0
        \]
        since $E|_U \simeq \bigoplus^n \cO_U$.
    \end{proof}
    Now, the claim implies that we can use Leray's theorem to compute $H^1(X,E)$ via a suitable open cover of $X$: choose a family $\brax{\paren{W_i,z_i}}_{1 \leq i \leq N}$ of coordinate neighborhoods on $X$, and trivializations $\varphi_i: \pi^{-1}(W_i) \to W_i \times \C^n$ of $E$ on $W_i$ so that
    \begin{enumerate}[(i)]
        \item $z_i:W_i \to D(0,2)$ is a biholomorphism, and
        \item if $U_i(t):= z_i^{-1}(D(0,t))$, then $\bigcup_{i=1}^NU_i(1/2)=X$.
    \end{enumerate}
    For each $t \in [1/2,2]$, let $\mc U(t)$ denote the open cover $\{U_i(t)\}_{1 \leq i \leq N}$ of $X$. For every $x \in W_j$ and $v \in E_x:=\pi^{-1}(x)$, define $\abs{\varphi_j(v)}:=\abs{w}$ where $\varphi_j(v)=(x,w)$ with $w \in \C^n$. We now define the bounded cocycles/coboundaries as follows:
    \begin{align*}
            Z_b^1(t) &:=\brax{ \xi \in Z^1\paren{ \mc U(t), E } \setbar f_{ij}\in E_b \paren{ U_i(t) \cap U_j(t) }, \forall i,j}.
            \\ C_b^0(t) &:= \brax{ \eta \in C^0\paren{\mc U(t),E} \setbar \text{ if } \eta = (c_i), \text{ then } c_i \in E_b(U_i(t)), \forall i }
    \end{align*}
    Introduce norms $\norm{\cdot}_t$ on $Z_b^1(t)$ and $C_b^0(t)$ by 
    \begin{align*}
        \norm{\xi}_t &:= \max_{i,j}\sup_{x \in U_i(t)\cap U_j(t)}\abs{ \varphi_i(f_{ij}(x)) }, \quad \text{ for } \xi=(f_{ij})\in Z_b^1(t),
        \\ \norm{\eta}_t &:= \max_{i} \sup_{x \in U_i(t)}\abs{\varphi_i(c_i(x))}, \quad \text{ for } \eta=(c_i)\in C_b^0(t).
    \end{align*}
    With these normals, $Z_b^1(t)$ and $C_b^0(t)$ are Banach spaces. Let $\frac{1}{2} \leq \rho \leq t < 1$.

\subsection{Functional Analysis Background}

\begin{defn}
A \term{topological vector space} (over $\C$) is a topological space $V$ which is also a $\C$-vector space such that
\begin{enumerate}
    \item[(i)] every point in $V$ is closed, and
    \item[(ii)] the vector space operations are continuous .
\end{enumerate}
\end{defn}

\begin{ex}
    Normed vector spaces, and Banach spaces are topological vector spaces.
\end{ex}

\begin{notebox}[Facts]
    Let $V$ be a topological vector space.
    \begin{enumerate}
        \item $\dim_\C V<\infty \iff V$ is locally compact.
        \item $V$ is Hausdorff.
        \item Any finite dimensional subspace $W \subset V$ is closed in $V$.
    \end{enumerate}
\end{notebox}

\begin{defn}
Let $V$ be a topological vector space, and $N$ be a closed linear subspace of $V$. Then, $V/N$ together with the quotient topology coming from $\pi:V \to V/N$ is a topological vector space such that $\pi$ is a continuous open map. We call $V/N$ the \term{quotient space}.
\end{defn}

\begin{proof}
The quotient topology makes $\pi$ continuous by definition. A subset $F \subset V/N$ is closed $\iff \pi^{-1}(F)$ is closed in $V$. Hence each point of $V/N$ is closed, since $\pi^{-1}(\pi(x))=x+N$ is closed (because $N$ is closed). Finally, if $U \subset V$ is open, then since $\pi^{-1}(\pi(U))=U+N=\bigcup_{n \in \N}(n+U)$, which is open in $V$, so we see that $\pi(U)$ is open in $V/N$.
\end{proof}

\begin{defn}
If $\paren{V,\norm{\cdot}}$ is a normed vector space, we get a \term{quotient norm} on $V/N$ by setting
\[
\norm{\pi(x)}:= \inf \brax{ \norm{x-n}:n\in \N }=:\w{dist}(x,N).
\]
\end{defn}

\begin{exercise}
Check that $(V/N,\norm{\cdot})$ is a normed vector space. If $(V,\norm{\cdot})$ is a Banach space, check that $(V/N,\norm{\cdot})$ is also Banach.
\end{exercise}

\begin{ex}
For $1 \leq p < \infty$, let $V$ be the space of all Lebesgue measurable functions $f:[0,1]\to \C$ such that $\int_{[0,1]}|f|^p < \infty$. Let $N$ be the subspace of $f \in V$ which are zero outside a set of measure zero. Then $N$ is closed in $V$, and $V/N$ is the Banach space $L^p[0,1]$.
\end{ex}

\begin{ex}[Non-example]
Let $\ell_2$ be the Hilbert space of all sequence $\{x_n\}_{n \geq 1}$ such that $\sum_{n=1}^\infty |x_n|^2 < \infty$. Let $N \subset \ell_2$ be the linear subspace of sequences $\{x_n\}$ such that $x_n=0$ for all $n \geq n_0$ (with $n_0$ depending on the sequence). Then, $N$ is a dense linear subspace of $\ell_2$, but not closed, so the quotient topology on $\ell_2/N$ is not Hausdorff.
\end{ex}

\begin{thm}[Open Mapping Theorem] \label{thm:open-mapping-thm}
If $X$ and $Y$ are Banach spaces, and $T:X \to Y$ is continuous, linear, and surjective, then $T$ is an open map.
\end{thm}

\begin{exercise}[Easy][label=exer:quotient-to-closed]
Suppose $N \subset E \subset V$ are linear subspaces of a topological vector space $V$ with $N$ closed. Show that if $E/N$ is closed in $V/N$, then $E$ is closed in $V$.
\end{exercise}

\subsection{\texorpdfstring{Concluding \Cref{thm:H1-is-finite}}{Concluding Second Finiteness Theorem}}

We have local trivialization neighborhood $\{(W_i,z_i)\}$ covering $X$. For $t \in [1/2,2]$, we set $U_i(t):= z_i^{-1}(D(0,t))$, and $\{\mc U_i(t)\}$ an open cover of $X$. We showed that for $t \in [1/2,1]$ the map
\[
\Phi:Z_b^1(1) \to H^1_b(t):= Z^1_b(t)/\delta C_b^0(t)
\]
is surjective. Moreover, $Z_b^1(t)$ and $C_b^0(t)$ are Banach spaces.

\begin{claim}
$\paren{H_b^1(t),\norm{\cdot}}$ is a Banach space.
\end{claim}
\begin{proof}
As we saw, this is equivalent to $\delta C_b^0(t)$ being a closed subspace of $Z_b^1(t)$. Fix an integer $N \geq 1$, and take $\rho$ such that $1/2 \leq \rho < t < 1$. Let $a_i \in W_i$ be the point where $z_i(a_i)=0$, and let
\[
C^0_b(t,N) := \brax{ \eta =(c_i)\in C_b^0(t) \setbar \ord_{a_i}(c_i) \geq N }.
\]
By Schwartz' Lemma, we know
\[
\norm{\eta}_\rho \leq \paren{\frac{\rho}{t}}^N\norm{\eta}_\rho
\]
if $\rho \in C_b^0(t,N)$, so pick $N$ such that $(\rho/t)^NC \leq 1/2$. Then, for $\rho \in C_b^0(t,N)$, we have
\[
\norm{\eta}_t \leq \norm{\delta_\eta}_t + C\norm{\eta}_\rho \leq \norm{\delta_\rho}_t + C\paren{\frac{\rho}{t}}^N\norm{\eta}_t \implies \norm{\eta}_t \leq 2 \norm{\delta_\eta}_t.
\] 
Now, a sequence $\{\delta_{\eta_n}\}_{n \geq 1}$ being Cauchy implies that $\{\eta_n\}$ is Cauchy, which implies $\{\eta_n\}$ converges because $C_b^0$ is a Banach space, which implies $\{\delta_{\eta_n}\}$ converges. Hence $\delta C_b^0(t,N) \subset Z_b^1(t)$ is a closed subspace, and it follows that the quotient
\[
\cH := Z_b^1(t)/\delta C_b^0(t,N)
\]
is a Banach space. Now the vector space $C_b^0(t)/C_b^0(t,N)$ is finite dimensional, so $\delta C_b^0(t)/ \delta C_b^0(t,N)$ is also finite dimensional, hence closed in $\cH$. Now, using \Cref{exer:quotient-to-closed}, it follows that $\delta C_b^0(t)$ is closed in $Z_b^1(t)$, hence $H_b^1(t)$ is a Banach space.
\end{proof}
We now complete the proof as follows:
\begin{proof}
    For $t<1$, $U_i(t)\cap U_j(t)$ is relatively compact in $U_i(1) \cap U_j(1)$. Montel's theorem implies that the restriction map $Z_b^1(1) \to Z_b^1(t)$ is compact when $t<1$ (i.e the image of the unit ball in $Z_b^1(1)$ is relatively compact). The induced map $\Phi:Z^1_b(1) \to H_b^1(t)$ is therefore both compact and surjective, so by the open mapping theorem, $\Phi$ is open, hence $H_b^1(t)$ has a relatively compact neighborhood of $0$, namely $\Phi(\text{unit ball})$. Therefore, $H_b^1(t)$ is locally compact, hence finite dimensional. We showed that $H^1(X,E) \simeq H_b^1(t)$, concluding the proof.
\end{proof}

\subsection{Applications}

\begin{thm}[][label=thm:exists-meromorphic]
Let $X$ be a compact Riemann surface, and suppose $\pi:L \to X$ is a holomorphic line bundle. Then $L$ has a meromorphic section which is not holomorphic. In particular:
\begin{enumerate}[(a)]
    \item any line bundle $L$ on $X$ is isomorphic to $L(D)$ for some divisor $D$ on $X$, and
    \item there exists a non-constant meromorphic function on $X$.
\end{enumerate}
\end{thm}

\begin{proof}
Choose $a \in X$, and $(U,z)$ a coordinate neighborhood of $a$ with $z(a)=0$, such that we have a holomorphic local trivialization $\varphi:\pi^{-1}(U) \to U \times \C$ of $L$. Let $k \geq 1$ and $s_k$ be the meromorphic section of $L$ over $U$ for which $\varphi(s_k(x))= (x,1/z(x)^k)$, $x \in U \setminus \{a\}$. 

Consider the open cover $\cU := \{U, X\setminus \{a\}\}$, and set
\begin{align*}
    f_{12}^{(k)}:= s_k|_{U \setminus \{a\}}, \qquad f_{21}^{(k)}=-f_{12}^{(k)}, \qquad f_{ij}^{(k)}=f_{ji}^{(k)}=0.
\end{align*}
This defines a cocycle $f^{(k)} \in Z^1(\cU,L)$. Since $H^1(X,L)$ is finite dimensional and $H^1(\cU,L)\hookrightarrow H^1(X,L)$ is injective, then for $d:= \dim_\C H^1(X,L)$ there exists constants $c_1,\dots,c_{d+1}$ not all zero such that 
\[
c_1f^{(1)}+\cdots +c_{d+1}f^{(d+1)}\in B^1(\cU,L),
\]
i.e. zero in cohomology. So, there exists holomorphic sections $u_1$ and $u_2$ of $L$ over $U$ and $X\setminus \{a\}$ respectively such that
\[
c_rs_r+\cdots + c_{d+1}s_{d+1}=u_1 -u_2
\]
on $U \setminus \{a\}$ with $c_r \neq 0$. The section $s:= u_2$ of $L$ on $X \setminus \{a\}$ is therefore meromorphic, and not holomorphic on $X$, since 
\[
s = -\sum_{\nu=r}^{d+1}c_\nu s_\nu + u_1
\]
on $U \setminus \{a\}$.
\end{proof}

\begin{cor}[][label=cor:pole-less-genus]
Let $g:=\dim_\C H^1(X,\mc O)$, called the \term{genus} of $X$, then there exists a non-constant meromorphic function on $X$, holomorphic on $X \setminus \{a\}$, with a pole of order $\leq g+1$ at $a$.
\end{cor}
\begin{proof}
    Use the trivial line bundle.
\end{proof}

\begin{rmk}
We'll see later that $g$ is indeed the topological genus of $X$, i.e. $g= \frac{1}{2}\rank H_1(X,\Z)$.
\end{rmk}

\begin{cor}
    Every Riemann surface of genus $0$ is isomorphic to $\P^1$.
\end{cor}
\begin{proof}
Let $a \in X$ and $f$ be a meromorphic function on $X$ with a single pole of order $\leq g+1=1$ at $a$, with $f$ not holomorphic, guaranteed by \Cref{cor:pole-less-genus}. Then, $f$ is a branched holomorphic covering map $X \to \P^1$, with only one sheet. Hence $f$ is a biholomorphism.
\end{proof}

\begin{thm}[][label=thm:mer-is-algebraic]
Let $X$ be a compact Riemann surface, and $f$ be a non-constant meromorphic function on $X$. Then there exists an integer $d \geq 1$ such that for any $g \in \mc M(X)$, there exist rational functions $a_1,\dots,a_d \in \C(x)$ such that
\begin{align} \label{eq:f-is-algebraic}
    g(x)^d+a_1(f(x))g(x)^{d-1}+\cdots + a_d(f(x))=0
\end{align}
for all $x \in X$. In other words, $g$ is algebraic over $\C(f)$.
\end{thm}

\begin{proof}
Let $f:X \to \P^1$ be a non-constant holomorphic map, let $A \subset X$ be the set of critical points of $f$, let $B:=f(A) \subset \P^1$ be the critical values, and let $Y:=X \setminus f^{-1}(B)$. Then, $f|_Y:Y \to \P^1\setminus B$ is a finite covering map. Let $d$ be the number of sheets.

Given any $g \in \cM (X)$, let $S\subset X$ denote the set of poles of $g$. For $z \in \P^1\setminus (B \cup f(S))$, define $a_i(z)$, $1\leq i \leq d$ by
\[
a_i(z)=e_i(g(x_1),\dots,g(x_d)),
\]
where $e_i$ is the $i$'th elementary symmetric polynomial, and $\{x_1,\dots,x_d\}=f^{-1}(z)$.

\begin{claim}
$a_i$ is holomorphic on $\P^1\setminus (B \cup f(S))$.
\end{claim}
\begin{proof}
Indeed, if $z_0 \in \P^1\setminus (B \cup f(S))$, we can choose a neighborhood $V$ of $z_0$ and disjoint open sets $U_1,\dots,U_d$ in $X \setminus A$ such that $f_j:=f|_{U_j}$ is a biholomorphic map onto $V$, and $f^{-1}(V)=\coprod_{j=1}^dU_j$. It follows that
\[
a_i(z)= e_i\paren{g (f_1^{-1}(z)),\dots, g(f_d^{-1}(z))}
\]
is holomorphic.
\end{proof}
Clearly \eqref{eq:f-is-algebraic} holds for any $x \in Y \setminus f^{-1} (f(S))$, hence we only need to check that the $a_i$ extend meromorphically to all of $\P^1$, since then $a_i(z)$ with be rational for all $i$.

Let $a \in B \cup f(S)$ and $U$ be a neighborhood of $a$ such that the only poles of $g$ on $f^{-1}(U)$ lie in $f^{-1}(a)$, and such that there exists a holomorphic function $w$ on $U$ with $w(a)=0$, $w \not \equiv 0$. Then, there exists an integer $N \geq 1$ such that $(w \circ f)^Ng$ is holomorphic on $f^{-1}(U)$. If $V$ is an open set such that $a \in V \subset U$, and $V$ is relatively compact in $U$, then $(w \circ f)^Ng$ is bounded on $f^{-1}(V)$. Hence, for $z \in V\setminus \{a\}$, define $b_i(z)$ to be the $i$'th elementary symmetric function of the values of $(w \circ f)^Ng$ at $f^{-1}(z)=\{x_1,\dots,x_d\}$, so then the $b_i$ are bounded. We deduce that the $b_i$ extend holomorphically to $a$ for all $i$. Now we're done, because $a_i=b_i/w^{iN}$, $1 \leq i \leq d$, so $a_i(z)$ has at most a pole at $a$.
\end{proof}

\begin{defn}
Let $K\supset \C$ be a field. We say that $K$ is an \term{algebraic function field} in $m$ variables if there exist $x_1,\dots,x_m$ in $K$ such that $x_1,\dots,x_m$ are algebraically independent over $\C$, and the extension $\C(x_1,\dots,x_m)\hookrightarrow K$ is finite.
\end{defn}

\begin{thm}
Let $X$ be a compact Riemann surface, and $\cM(X)$ denote the field of meromorphic functions on $X$. Then, $\cM(X)$ is an algebraic function field in $1$ variable.
\end{thm}

\begin{proof}
Let $f$ be a non-constant meromorphic function on $X$, which \Cref{thm:exists-meromorphic} guarantees. Then, $f$ is algebraically independent over $\C$, since if
\[
f^n+c_{n-1}f^{n-1}+\cdots + c_0=0, \quad c_i \in \C,
\]
then $f$ takes on finitely many values, which is not the case. So, $\C(f) \simeq \C(z)$. By \Cref{thm:mer-is-algebraic}, there exists $d \geq 1$ such that any $g \in \cM(X)$ is algebraic over $\C(f)$, of degree at most $d$. Choose $g_0 \in \cM(X)$ such that $[\C(f,g_0):\C(f)]$ is maximal.
\begin{claim}
    We claim that $\C(f,g_0)=\cM(X)$.
\end{claim}
\begin{proof}
Indeed, if there exists $h \in \cM(X)$, $h \not \in \C(f,g_0)$, then since $\C(f)$ has characteristic zero, the primitive element theorem guarantees some $g \in \cM(X)$ such that $\C(f)(g_0,h)=\C(f)(g)$. This gives a contradiciton, because
\[
[\C(f)(g):\C(f)] = \braks{\C(f)(g_0,h):\C(f)}> [\C(f)(g_0):\C(f)],
\]
proving the claim.
\end{proof}
This claim proves the theorem.
\end{proof}

\begin{que}
    If $X$ is a compact Riemann surface, does $\cM(X)$ determine $X$ up to isomorphism?
\end{que}

\begin{notethm}[Answer]
    Yes. This is the approach one takes in abstract algebraic geometry, which uses valuation theory: if we know $K:= \cM(X)$, then points $p \in X$ correspond to discrete valuations $\nu:K \to \Z$, $f \mapsto \ord_pf$.
\end{notethm}

\begin{rmk}
Let $X,Y$ be compact Riemann surfaces, and $f:Y \to X$ a (branched) holomorphic covering map. Then, the pullback $f^*\cM(X)$ given by meromorphic $g\circ f :Y \to \C$ for meromorphic $g \in \cM(X)$ is naturally a subfield of $\cM(Y)$. Identifying $f^*\cM(X)$ with $\cM(X)$, the Galois group $\Gal(\cM(Y)/\cM(X))$ is then naturally isomorphic to the group of Deck transformations $\w{Deck}(Y/X)$ of the covering map $f$. Moreover, the covering is Galois (or normal) if and only if the extension $\cM(Y)/\cM(X)$ is Galois.
\end{rmk}

\section{The Riemann-Roch Theorem}

\begin{defn}
If $D = \sum n_P[P]$ is a divisor on a compact Riemann surface $X$, then the \term{degree} of $D$ is $\deg D:= \sum n_P$. We think of $D$ as a function $D:X \to \Z$ with finite support (i.e. $D(P)=n_P$).
\end{defn}

\begin{prop}
If $D_1$ and $D_2$ are linearly equivalent, then $\deg D_1=\deg D_2$.
\end{prop}

\begin{proof}
Recall that $D_1 \sim D_2 \iff D_1 -D_2 = \div f$ for some $f \in \cM(X)$, hence
\[
\deg D_1 - \deg D_2 = \deg \div f=0.
\]
\end{proof}

We recall that for any divisor $D$, we introduced (\Cref{rmk:sheaf-on-divisor}) the sheaf $\cO_D$ on $X$ with
\[
\cO_D(U):=\brax{ f \in \cM(X) \setbar \div f \geq -D \text{ on } U}.
\]
Moreover, $\cO_D$ is isomorphic to the sheaf of holomorphic sections of the line bundle $L(D) \to X$, so
\[
\cO_D(U) = H^0(U,L(D))= \Gamma (U,L(D)).
\]

\begin{notebox}[The Riemann-Roch Problem]
    Riemann showed that
    \[
    h^0(D):= \dim_\C(H^0(X,\cO_D))=\dim_\C \cO_D(X)=\dim_\C \brax{f \in \cM(X) \setbar \div f \geq -D}
    \]
    is finite. Moreover, he showed that $h^0(D) \geq 1 -g + \deg D$, and assigned his student Roch to compute the difference.
\end{notebox}

An element of the stalk $\cO_{D,a}$ is given by a convergent Laurent series
\[
f = \sum_{n \geq -D(a)}c_nz^n,
\]
where $z$ is a local coordinate function near $a$ with $z(a)=0$. If $D_1,D_2$ are divisors with $D_1 \leq D_2$, then $-D_1 \geq -D_2$, and so clearly $\cO_{D_1}(U) \subset \cO_{D_2}(U)$, for all $U \subset X$ open. Hence, we get an injective map of sheaves $\cO_{D_1} \hookrightarrow \cO_{D_2}$. 

In terms of line bundles $L(D_1)$ and $L(D_2)$, the map is given by 
\begin{align*}
    \Gamma(U, L(D_1)) &\to \Gamma(U, L(D_2))
    \\ f & \mapsto f \otimes s_{D_2-D_1},
\end{align*}
where $s_{D_2-D_1}$ is the standard section of $L(D_2 - D_1)$ with $\div s = D_2-D_1$. This is because $L(D_2) \simeq L(D_1) \otimes L(D_2 - D_1)$.

In the following, \textbf{$X$ will always be a compact Riemann surface}.

\begin{prop}
If $D$ is a divisor on $X$ with $\deg D < 0$, then $H^0(X,\cO_D)=0$.
\end{prop}

\begin{proof}
If not, then there exists $f \neq 0$, $f \in \cO_D(X)$, so $\div f \geq -D$ implies $$0 = \deg \div f \geq -\deg D >0,$$ a contradiction.
\end{proof}

\begin{defn}[Recall]
Let $P$ be a point on $X$. The \termnt{skyscraper sheaf} $\C_P$ on $X$ satisfies
\[
\C_P(U)= \case{\C & P \in U \\ 0 & P \not \in U}
\]
with the obvious restriction maps. 
\end{defn}

\begin{prop}
\begin{enumerate}[(i)]
    \item $H^0(X,\C_p) \simeq \C$
    \item $H^1(X,\C_p)=0$.
\end{enumerate}
\end{prop}

\begin{proof}
(i) is trivial. For (ii), consider a $\xi \in H^1(X, \C_P)$ represented by a cocycle in $Z^1(\cU,\C_P)$. The open cover $\cU$ has a refinement $\mc V= \{V_\alpha\}_{\alpha \in A}$ so that $P \in V_\alpha$ for a unique $\alpha \in A$.  Then $Z^1(\mc V,\C_P)=0$, so $\xi =0$ in $H^1(X,\C_p)$.
\end{proof}

Now, it turns out that we can use the skyscraper sheaf to help us understand the Riemann-Roch problem on a compact Riemann surface $X$. For each divisor $D$ and point $P \in X$, we have a map $\cO_D \to \cO_{D+P}$. Let $(W,z)$ be a local coordinate chart on $X$ around $P$ such that $z(P)=0$. We define a sheaf homomorphism $\beta :\cO_{D+P} \to \C_P$ as follows: given $U \subset X$, if $P \not \in U$, then the homomorphism $\beta_U$ is zero. Otherwise, if $P \in U$ and $f \in \cO_{D+P}(U)$, expand $f = \sum_{n=-k-1}^\infty c_nz^n$ in the $z$-coordinate, where $k = D(P)$. We then set $\beta_U(f):= c_{-k-1} \in \C= \C_P(U)$.

We obtain a short exact sequence of sheaves on $X:$
\[
0 \to \cO_D \to \cO_{D+P}\arr[\beta] \C_P \to 0,
\]
inducing a long exact cohomology sequence
\begin{align} \label{eq:long-exact-RR-cohom}
0 \to H^0(X,\cO_D) \to H^0(X,\cO_{D+P}) \to \C \to H^1(X,\cO_D) \to H^1(X,\cO_{D+P})\to 0.    
\end{align}

\begin{cor}
    If $D_1 \leq D_2$ are two divisors on $X$, then the inclusion map $\cO_{D_1}\hookrightarrow \cO_{D_2}$ induces a surjective homomorphism
    \[
    H^1(X, \cO_{D_1}) \to H^1(X, \cO_{D_2}) \to 0.
    \]
\end{cor}

\begin{proof}
If $D_2 = D_1 + P$ for some $P \in X$, we've just proved this by the penultimate map in the sequence \eqref{eq:long-exact-RR-cohom}. Otherwise, $D_2 = D_1 + P_1 + P_2 + \cdots + P_r$, so we use induction.
\end{proof}

\begin{defn}
If $D$ is a divisor on $X$, define $h^i(D):= \dim_\C H^i(X,\cO_D)$ for $i = 0,1$. Then, define the \term{Euler characteristic} $\chi(D):= h^0(D)- h^1(D)$.
\end{defn}

\begin{notethm*}[Historical Note]
    This definition leaps us forward in history by 100 years. Riemann didn't have this notion of cohomology or Euler characteristic.
\end{notethm*}

\begin{lem}[][label=lem:chi-deg]
If $D_1$ and $D_2$ are divisors, then
\begin{align} \label{eq:chi-deg}
    \chi(D_2)-\deg(D_2)=\chi(D_1)-\deg(D_1). 
\end{align}
\end{lem}

\begin{proof}
    Suppose first that $D_2 = D_1 + P$. Then, the exact cohomology sequence
    \[
    0 \to \cO_{D_1}(X) \to \cO_{D_1 + P}(X) \to \C \to H_1(X, \cO_{D_1}) \to H^1(X, \cO_{D_2}) \to 0
    \]
    can be split into two short exact sequence as follows: set $V := \Img (\cO_{D_1 +P}(X) \to \C)$, and $W:= \C /V$. Then $$\dim V + \dim W = \dim \C = 1 = \deg D_2 - \deg D_1,$$ and the sequences
    \begin{align*}
        &0 \to \cO_{D_1}(X) \to \cO_{D_1 +P}(X) \to V \to 0, \\ &0 \to W \to H^1(X,\cO_{D_1}) \to H^1(X,\cO_{D_2}) \to 0
    \end{align*}
    are exact. Since all the vector spaces are finite dimensional, we get 
    \begin{align*}
        h^0(D_2) &= h^0(D_1)+ \dim V, \\
        h^1 ( D_1) &= h^1(D_2)+ \dim W.
    \end{align*}
    Adding yields $\chi (D_2)=\chi(D_1)+1=\chi(D_1)+\deg D_2 - \deg D_1.$ Iterating this calculation shows that this assertion holds whenever $D_1 \leq D_2$. Finally, if $D_1$ and $D_2$ are arbitrary, find a $D$ such that $D \leq D_1$ and $D \leq D_2$. Then,
    \[
    \chi(D_i)-\deg D_i = \chi(D)-\deg D, \quad i =1,2,
    \]
    proving the claim.
\end{proof}

\begin{thm}[Riemann-Roch Theorem (weak form)][label=thm:weak-riemann-roch]
If $D$ is a divisor on a compact Riemann surface $X$, then $h^0(D) - h^1(D)=1-g+ \deg D$, where $g$ is the genus of $X$.
\end{thm}

\begin{proof}
By \Cref{lem:chi-deg},
\[
h^0(D)-h^1(D)=\chi(D)\overset{\eqref{eq:chi-deg}}{=} \chi(0)+\deg D = h^0(\cO)-h^1(\cO) = 1-g+\deg D.
\]
\end{proof}

\begin{rmk}
Riemann-Roch shows that
\[
\dim H^0(X,\cO_D) - \dim H^1(X, \cO_D) = 1-g + \deg D.
\]
What's cool is that the LHS is an analytic invariant, while the RHS is a topological invariant. For instance, when $D=0$, $\dim H^1(X,\cO)=g$, the genus.
\end{rmk}

\section{Differential Forms on Riemann Surfaces}

\begin{defn}
Recall that if $X$ is a real $n$-dimensional $C^\infty$-manifold, then we have the \term[tangent bundles]{tangent bundle} $TX$. In a coordinate chart $\{x_i\}_{i=1}^n$ on $X$, a local basis of sections of $TX$ is given by vector fields $\partder[]{}{x_1},\dots,\partder[]{}{x_n}$. At any point $a \in X$, $\partder[]{}{x_i}|_a$ is a \term[derivations]{derivation}, i.e. a linear map $C_a^\infty \to \R$ obeying the product rule. The \term{cotangent space} $T_a^*X = (T_aX)^*$ has basis $dx_1|_a,\dotsm,dx_n|_a$.

We define the \term{complex tangent bundle} $T_\C X:=TX \otimes _\R \C$, the bundle of complex valued tangent vectors, and we define $T_\C^*X$ similarly.
\end{defn}

Suppose $X$ is a Riemann surface and $a \in X$. Let $\mc A = C^\infty$ denote the sheaf of smooth functions $U \to \C$ for $U \subset X$ open. On any coordinate neighborhood $(U,z)$, with $z = x+iy$, we have the differential operators
\[
\partder[]{}{x}, \quad \partder[]{}{y}, \quad \partder[]{}{z}=\frac{1}{2}\paren{\partder[]{}{x}-i \partder[]{}{y}},\quad \partder[]{}{\bar z}=\frac{1}{2}\paren{\partder[]{}{x}+i \partder[]{}{y}}.
\]
The stalk $\mc A_a$ consists of germs of smooth functions defined in a neighborhood of $a$. Let $m_a \subset \mc A_a$ be the maximal ideal of germs of functions which vanish at $a$.

\begin{defn}
Let $X$ be a Riemann surface. The \term{cotangent space} at $a \in X$ is $T_a^* X:= m_a/m_a^2$. For any $f \in \mc A(U)$, where $U$ is an open neighborhood of $a$, we define
\[
d_af:= (f-f(a)) \pmod {m_a^2}.
\]
\end{defn}

\begin{prop}
The dual space $(m_a/m_a^2)^*$ is the tangent space $T_aX$.
\end{prop}

\begin{proof}
Let $T: m_a/m_a^2 \to \C$ be a linear functional. We can define a derivation $D_T:\mc A_a \to \C$ by $D_T(f) = T(f-f(a))$. Note that $D_T$ is indeed a derivation, since for $f,g \in \mc A_a$, we have
\[
T\paren{ (f-f(a))(g-g(a)) }=0 \iff D_T(fg)=f(a)D_T(g)+ g(a)D_T(f).
\]
\end{proof}

Now for $(U,z)$ a coordinate neighborhood of $a \in X$ with $z=x+iy$, we have $\{d_ax,d_ay\}$ and $\{d_az,d_a\bar z\}$ for two bases of $T^*_{\C,a}X$. For any function $f \in C_a^\infty$, the Taylor formula
\[
f-f(a)=\partder[]{f}{x}(a)(x-x(a))+\partder[]{f}{y}(a)(y-y(a)) \pmod{m_a^2}
\]
translates to
\[
d_af = \partder[]{f}{x}(a)\cdot d_ax+\partder[]{f}{y}(a)\cdot d_ay.
\]
Similarly, we have
\[
df = \partder[]{f}{z}dz + \partder[]{f}{\bar z}d \bar z.
\]
Now, if $(U',z')$ is a second coordinate neighborhood at $a$, then
\[
\partder[]{z'}{z}(a)=c \in \C^*, \qquad \partder[] {\bar{z'}}{\bar z}(a)= \bar c,
\]
while
\[
\partder[]{z'}{\bar z}(a)=\partder[]{\bar{z'}}{z}(a)=0.
\]
These are both consequences of the Cauchy-Riemann equations. Therefore $d_az'= cd_az$ and $d_a\bar{z'}=\bar c d_a\bar z$. Hence the subspaces $T_a^{1,0}X,T_a^{0,1}x$ of $T^*_\C X$ spanned by $d_az$ and $d_a \bar z$ are independent of the choice of local coordinate at $a$. Thus we get a decomposition
\[
T^*_{\C,a}X=T_a^{1,0}X \bigoplus T_a^{0,1}X
\]

\begin{defn}
A \term{differential 1-form} on $X$ is a $C^\infty$-section of $T_\C^* X$.

Locally, any differential $1$-form equals $adz + bd \bar z$ for $a,b \in C^\infty(U)$. These patch together to form a sheaf. We denote the \term{sheaf of smooth 1-forms} on $X$ by \term[\cA^1]{$\cA^1$}. We also sometimes write $\mc A =\mc A^0$.

A 1-form $\varphi$ is of \term[(1,0)]{type (1,0)} if, for any local coordinate system $(U,z)$, $\varphi$ is of the form $adz$ (respectively $b d \bar z$ for \term[(0,1)]{type (0,1)}).

Let $\mc A^{1,0}$ and $\mc A^{0,1}$ denote the sheaves of $C^\infty$-forms of type $(1,0)$ and $(0,1)$. So for $U \subset X$ open,
\[
\mc A^{1,0}(U) = C^\infty (U,T^{1,0}U).
\]
\end{defn}

\begin{defn}
For $f \in \mc A(U)=\C^\infty (U)$, the \term{exterior derivative} $df$ has a unique expression $df = \partial f + \bar \partial f$ with $\partial f \in \mc A^{1,0}(U)$ and $\bar \partial f \in \mc A^{0,1}(U)$. In local coordinates, $\partial f = \partder[]{f}{z}dz$ and $\bar \partial f = \partder[]{f}{\bar z}d \bar z.$
\end{defn}

\begin{rmk}
Let $\mc A^k(U)$ denote smooth $k$-forms on $U$. Just as $d$ extends to a map $\mc A^k(U) \to \mc A^{k+1}(U)$, so do $\partial$ and $\bar \partial$.

For $\alpha \in \mc A^{1,0}(U)$, $\partial \alpha =0$ and $\bar \partial \alpha = d\alpha$. Similarly for $\beta \in \mc A^{0,1}(U)$, $\partial b = d \beta$ and $\bar \partial \beta =0$.

In local coordinates, if $\alpha = fdz$ and $\beta = g d \bar z$,
\begin{align*}
    \bar \partial \alpha &= \partder[]{f}{\bar z}d \bar z \land d z = -\partder[]{f}{\bar z}dz \land d \bar z, \\
    \partial \beta &= \partder[]{g}{z}dz \land d \bar z.
\end{align*} 
\end{rmk}

\section{\texorpdfstring{The Canonical Bundle $K_X$}{The Canonical Bundle KX}}

\begin{defn}
Let $W \subset X$ be open. A \term{holomorphic 1-form} on $W$ is an element $\omega \in \mc A^{1,0}(W)$ such that $\bar \partial w =0$ (equivalently, $dw=0$). If $(U,z)$ is a coordinate chart and $w|_{W \cap U} = fdz$, then $w$ is holomorphic if and only if $f$ is holomorphic. We can similarly define meromorphic $1$-forms: there are forms defined outside of a discrete set $A \subset W$, which, for any coordinate chart $(U,z)$ with $U \subset W$, are of the form $fdz$ with $f$ meromorphic on $U$.

Let $\Omega = \Omega^1_X$ be the \term{sheaf of holomorphic 1-forms} on $X$, with
\[
W \mapsto \Omega(W) := \brax{ \text{holomorphic 1-forms on }W}.
\]
The \term[canonical line bundle,K_X]{canonical line bundle $K_X$} on $X$ is a line bundle such that, for all $U \subset X$ open,
\[
H^0(U,K_X)=\Omega(U).
\]
In fact we have $(K_X)_a = T^{1,0}_aX$.
\end{defn}

We can give transition functions for $K_X$ as follows: Let $\{(U_i,z_i)\}$ be an open cover of $X$ by charts. Then by the above analysis, there exist $g_{ij} \in \mc O^*(U_i \cap U_j)$ such that $dz_j=g_{ij}d z_i$ on $U_i \cap U_j$. Thus $K_X$ is the line bundle given by the transition functions $\brax{g_{ij}}$.

Over any open set $W \subset X$, a section $s \in H^0(W,K_X)$ is gen by a family $\{f_i\}$ with $f_i \in \mc O(W \cap U_i)$ such that $f_i = g_{ij}f_j$ on $U_i \cap U_j \cap W$ for all $i,j$. Consider the holomorphic $1$-form $\omega_i := f_idz_i$ on $W \cap U_i$. Then on $W \cap U_i \cap U_j$, 
$$w_i = f_idz_i=g_{ij}f_jdz_i=f_jdz_j=w_j,$$ hence the $\{\omega_i\}$ patch to define $\omega \in \Omega (W)$. The map $s \mapsto \omega$ is an isomorphism.

\begin{rmk}
Under the above correspondence, meromorphic sections of $K_X$ correspond to meromorphic $1$-forms. Moreover, if we apply the construction to $C^\infty$-sections of $K_X$ over $W \subset X$, we obtain an isomorphism $C^\infty (W,K_X)\simeq \mc A^{1,0}(W)$.
\end{rmk}

\section{The Dolbeault Isomorphism}

\begin{defn}[Recall]
Let $E \arr[\pi]X$ be a holomorphic vector bundle on the Riemann surface $X$. The sheaf of holomorphic sections of $E$ is denoted $\mc E$, while $\mc A_E=C^\infty _E$ denotes the sheaf of smooth sections of $E$, so
\[
 \mc A_E(U)= \brax{C^\infty \text{ maps } s:U \to E, \pi\circ s = \id_U}.
\]
\end{defn}
\begin{rmk}
If $E|_U$ is trivial, then the map, then the map
\begin{align*}
    \mc E(U) \otimes_{\mc O(U)} \mc A(U) &\to \mc A_E(U) \\
    s \otimes f & \maps fs
\end{align*}
is an isomorphism. Set \term[$\cA_E^{0,1}$]{$\cA_E^{0,1}(W)$} $:= \mc A_E(W)\otimes _{\mc A(W)}\mc A^{0,1}(W)$, for $W$ open in $X$. If $E|_W$ is trivial, then $\mc A_E^{0,1}(W) = \mc E(W) \otimes _{\mc O (W)} \mc A^{0,1}(W)$. It follows that there exists a unique $\mc O(W)$-linear map
\[
\bar \partial_{E,W}: \mc A_E(W) \to \mc A_E^{0,1}(W)
\]
over each open set $W$ where $E|_W$ is trivial, induced by the map
\begin{align} \label{eq:big-tensor-stuff}
1 \otimes \bar \partial : \mc E(W) \otimes_{ \mc O(W) } \mc A(W) \to \mc E(W)\otimes _{\mc O(W)} \mc A^{0,1}(W).    
\end{align}
This map is well defined since $\bar \partial$ is $\mc O(W)$-linear.
\end{rmk}

\begin{defn}
 The map \eqref{eq:big-tensor-stuff} extends to a map
\begin{align*}
    \bar \partial_E : \mc A_E \to \mc A_E^{0,1}.
\end{align*}
\end{defn}

\begin{defn}
    We have an alternative definition of $\bar \partial _E$. Locally, a section $\alpha$ of $\mc A_E$ can be written a $f\otimes s$, for $f \in \mc A(U)$ and $s \in \mc E(U)$. We could try to define $d_\alpha = df \otimes s$, but this fails: for any holomorphic function $g:U \to \C^*$, we have $\alpha = fg \otimes g^{-1}s$, but if $g$ is not constant then
    \[
    d(fg) \otimes g^{-1}s \neq df \otimes s.
    \]
    However, setting $\bar \partial_E(f \otimes s):= \bar \partial f \otimes s$ makes sense, and satisfies the Leibniz rule
    \[
    \partial _E(f \cdot \alpha)= \bar \partial f \land \alpha+ f \bar \partial_E (\alpha).
    \]
\end{defn}

\begin{proof}
To justify this, fix a holomorphic trivialization of $E|_U$ and write $s=(s_1,\dots,s_r)$, while $\alpha = \sum_{i=1}^r f_i \otimes s_i$ for $s_i \in \mc A(U)$. Now set $\bar \partial_E \alpha := \sum_{i=1}^r \bar \partial f_i \otimes s_i$. Then if $s'=(s_1',\dots,s_r')$ is a different local trivialization, it gives rise to $\bar \partial '_E$.
\begin{claim}
$\bar \partial_E = \bar \partial '_E$.
\end{claim}
Indeed, if $\brax{\psi_{ij}}$ is. the transition matrix, then $s_i = \sum_j \psi_{ij}s_j'$
and
\[
\bar \partial _E'(\alpha) = \bar \partial'_E\paren{ \sum_{i,j}f_i \times \psi_{ij}s_j' } = \sum_{i,j}\bar \partial \paren{f_i \psi_{ij}} \otimes s_j' = \sum_{i,j} (\bar \partial f_i)\psi_{ij}\otimes s_j' = \bar \partial _E(\alpha).
\]
\end{proof}

\begin{thm}
For any holomorphic vector bundle $\pi:E \to X$ over a Riemann surface $X$, the sequence
\[
0 \to \mc E(X) \to \mc A_E(X) \arr[\bar \partial] \mc A_E^{0,1}(X) \to H^1(X,E)\to 0
\]
is exact.
\end{thm}

\begin{proof}
The main point is to show that $\coker (\bar \partial)= H^1(X,E)$, i.e
\[
\mc A_E(X) \arr[\bar \partial] \mc A_E^{0,1}(X) \to H^1(X,E)
\]
is exact.
\begin{lem} \label{lem:H1XAE-cohom-is-0}
We claim $H^1(X,\mc A_E)=0$.
\end{lem}
\begin{proof}
$\mc A_E$ is a fine sheaf, i.e. it admits partitions of unity. Let $\mc U = \{U_i\}_{i \in I}$ be an open cover, and take cocycles $s_{ij} \in \mc A_E(U_i \cap U_j)$ with $\{s_{ij}\} \in Z^1(\mc U, \mc A_E)$. If $\brax{\varphi_i}_{i \in I}$ is a partition of unity with respect to $\mc U$, define $s_i \in \mc A_E(U_i)$ by $s_i = \sum_{j \in I} \varphi_j s_{ij}$, where 
\[
(\varphi_js_{ij})(x)=\case{ \varphi_js_{ij}(x), & x \in U_i \cap U_j \\ 0, & x \in U_i \smallsetminus (U_i \cap U_j).}
\]
On $U_i \cap U_j$, we obtain
\[
s_k - s_\ell = \sum_j \varphi_j(s_{kj}-s_{\ell j})= \sum_j \varphi_j s_{k\ell}=s_{k\ell}.
\]
\end{proof}
Now, we apply the long exact cohomology sequence (\cref{thm:longexact}) to the exact sequence of sheaves:
\[
0 \to \mc E \to \mc A_E \arr[\bar \partial] \mc A_E^{0,1}\to 0
\]
to obtain the result. More explicitly, we can construct a map
\begin{align} \label{eq:H1-A-quotient-isom}
    D:H^1(X,E) \to \mc A_E^{0,1}(X)/\bar \partial \mc A_E(X)
\end{align}
as follows: let $\{s_{ij}\} \in Z^1(\mc U, E)$, so $s_{ij} \in H^0(U_i \cap U_j, E)$ for all $i,j \in I$. Using \Cref{lem:H1XAE-cohom-is-0}, find $\alpha \in \mc A_E(U_i)$ such that $\alpha _i - \alpha _j = s_{ij}$ on $U_i \cap U_j$. Then $\bar \partial \alpha _i = \bar \partial \alpha _j$ on $U_i \cap U_j$, so the $\brax{\bar \partial \alpha_i}$ define an element of $\mc A^{0,1}_E(X)$, whose image in the quotient $\mc A_E^{0,1}(X)/\bar \partial \mc A_E$ is $D(\brax{s_{ij}})$.
\begin{exercise}
Check that $D$ is independent of the choices made, and is injective.
\end{exercise}
The hard part is proving that $D$ is surjective. Given $\omega \in \cA^{0,1}_E(X)$, we can find an open cover $\mf U = \{U_i\}$ of $X$ and $\varphi_i \in \cA_E(U_i)$ with $\bar \partial \varphi_i = w|_{U_i}$, using our solution to $\bar \partial u =f$ locally. Set $s_{ij}:= \varphi_i-\varphi_j |_{U_i \cap U_j}$ for all $i,j$. Then $\bar \partial s_{ij}=0$ on $U_i \cap U_j$, so $s_{ij} \in \cE (U_i \cap U_j)=H^0(U_i \cap U_j,E)$. Clearly $\{s_{ij}\} \in Z^1(\mf U,E)$, and from construction it follows that $D(\{s_{ij}\}) = [w]$ in $\cA_E^{0,1}(X)/\bar \partial \cA_E(X)$.
\end{proof}

\begin{defn}
The \term{$(0,1)$-Dolbeault cohomology group} of $X$ is defined as the quotient group
\[
H_{\bar \partial}^{0,1}(X) := \mc A_E^{0,1}(X)/ \bar \partial  \mc A_E(X).
\]
\end{defn}

\begin{rmk}
We've shown that the map in \eqref{eq:H1-A-quotient-isom} is a vector space isomorphism. However, we will want to put topologies on the structures, and we'll want an isomorphism of the associated complete topological vector spaces.
\end{rmk}

\subsection{Frech\'{e}t Spaces}

\begin{defn}
A topological space $X$ is a \term[Freschet space]{Frech\'{e}t space} if 
\begin{enumerate}
    \item[(i)] the topology on $X$ may be induced by a countably family of seminorms $\brax{\norm{\cdot}_n}_{n \geq 1}$. That is, $U \subset X$ is open if and only if for all $x \in U$, there exists $N \geq 1$ and $\veps >0$ such that
\[
\brax{ y \setbar \norm{y-x}_n < \veps, \: \forall n \leq N } \subset U.
\]
    \item[(ii)] $X$ is complete with respect to the family $\brax{\norm{\cdot}_n}_{n \geq 1}$ of seminorms.
\end{enumerate}
\end{defn}

\begin{prop}
Any Frech\'{e}t space is metrizable: one defines a matrix on $X$ by
\[
d(x,y):= \sum_{n=1}^\infty \frac{1}{2^n}\frac{\norm{x-y}_n}{1+\norm{x-y}_n}.
\]
\end{prop}

\begin{rmk}
Since $X$ is a topological vector space, it's already Hausdorff, but not that the topology in a Frech\'{e}t space is Hausdorff if and only if
\[
\bigcap_{n=1}^\infty \brax{ x \in X \setbar \norm{x}_n=0 } = \{0\}.
\]
\end{rmk}

\begin{rmk}
Let $(U,z)$ be any coordinate chart on $X$. Then $C^\infty (U)$ carries a \term{$C^\infty$ topology} (or \term{Schwartz topology}) of uniform convergence of all derivatives $D^k$ on compact subsets of $U$.

Next, suppose $E\to X$ is a rank $n$ vector bundle where $E|_U$ trivial. If $\varphi \in \cA_E^{0,1}(X)$, then $\varphi|_U$ may be identified with an expression $(\varphi_1,\dots,\varphi_n)d\bar z$, where $\varphi \in C^\infty (U)$. Then we say that a sequence $\varphi^{(\nu)} \in \cA_E^{0,1}(X)$ converges if and only if, for any $U \subset X$ as above, the corresponding sequence $(\varphi_1^{(\nu)},\dots,\varphi_n^{(\nu)})$ with $(\varphi_1^{(\nu)},\dots,\varphi_n^{(\nu)})d\bar z = \varphi^{(\nu)}|_U$ converging in $\prod_{j=1}^nC^\infty (U)$. We also have a similar topology on $\cA_E(X)$.

In this way, $\cA_E^{0,1}(X)$ and $\cA_E(X)$ become Frech\'{e}t spaces, so in particular they are complete metric spaces.
\end{rmk}

\begin{thm}
$\bar \partial \cA_E(X)$ is a closed subspace of $\cA_E^{0,1}(X)$.
\end{thm}
\begin{proof}
Since $\codim \bar \partial \cA_E(X)<\infty$, this follows from \Cref{prop:frechet-func-anal}.
\end{proof}
\begin{prop} \label{prop:frechet-func-anal}
Let $X,Y$ be Frech\'{e}t spaces, and $T:X \to Y$ a continuous linear map. If $\dim _\C(Y/T(X))<\infty$, then $T(X)$ is closed in $Y$.
\end{prop}

\begin{proof}
$\ker T$ is closed, so $X/\ker T$ is a Frech\'{e}t space. Hence we may assume that $T$ is injective. Let $Y_0 \subset Y$ be a finite dimensional subspace such that the projection map $Y_0 \to Y/T(X)$ is an algebraic isomorphism.
\begin{claim}
We claim $Y_0$ is closed in $Y$.
\end{claim}
\begin{proof}
Indeed, if $y_1,\dots,y_n$ is a basis for $Y_0$, then the map $\C^n \to Y_0$ sending $(\lambda_1,\dots,\lambda_n) \mapsto \sum \lambda _iy_i$ is continuous and bijective. In fact, it is a homeomorphism since the image of
\[
\brax{\sum_{i=1}^n \abs{\lambda_i}^2=1}
\]
is a compact, hence closed, subset of $Y_0$. Then there exists a neighborhood of $0$ which is disjoint from this image, which implies that the map is open. We deduce that $Y_0$ is complete in the induced topology from $Y$, hence closed.
\end{proof}
Now consider the map $Y_0 \oplus X \to Y$ sending $(y,x) \mapsto y+T(x)$. This is a continuous bijection, so the open mapping theorem 
% (\Cref{thm:open-mapping-thm}) 
implies that it is a homeomorphism. As $X$ is closed in $Y_0 \oplus X$, its image $T(X)$ is closed in $Y$.
\end{proof}

\begin{cor}
Under the conditions of \Cref{prop:frechet-func-anal}, the quotient $Y/T(X)$ is a finite dimensional Hausdorff topological vector space.
\end{cor}

\begin{rmk}
It is not necessarily true that a finite codimension linear subspace of a Frech\'{e}t space is closed.
\end{rmk}



\section{Serre Duality}

\textit{``A differential form is something that you can integrate on a manifold''.}

Let $X$ be a compact Riemann surface, and $\pi:E \to X$ a holomorphic vector bundle on $X$.

We define a bilinear form 
\[
\angles{,}:H^0(X,E^*\otimes K_X)\otimes \cA_E^{0,1}(X) \to \C
\]
as follows: fix $s \in H^0(X,E^*\otimes K_X)$ and $\varphi \in \cA_E^{0,1}(X)$. Let $(U,z)$ be a coordinate neighborhood, and $\omega \in \Omega(U)$ be a holomorphic 1-form on $U$ without zeros (i.e. $w = fdz$, where $f(a) \neq 0$ for all $a \in U$). Then by restricting $s$ to $U$, we may write $s|_U = \lambda \otimes \omega$ for $\lambda \in H^0(U,E^*)$, and $\varphi|_U = \alpha \otimes \bar \omega$ for $\alpha \in \cA_E(U)$. Then we may define a pairing
\begin{align} \label{eq:serre-pairing}
    (s,\varphi)_U:= (\lambda,\alpha)\omega \land \bar \omega
\in \cA^2(U).
\end{align}
\begin{claim}
The pairing \eqref{eq:serre-pairing} is independent of the choice of $\omega$.
\end{claim}

\begin{proof}
If $\omega' = f\omega$ for $f \in \cO^*(U)$, then $s|_U = \paren{\frac{1}{f}\lambda}\otimes \omega '$ and $\varphi|_U = \paren{\frac{1}{\bar f}\alpha} \land \bar{\omega '}$, so
\[
\paren{\frac{1}{f}\lambda, \frac{1}{\bar f}\alpha} \omega' \land \bar{\omega '} = \frac{1}{\abs{f}^2}(\lambda, \alpha) f \bar f (\omega \land \bar{\omega}) = (\lambda,\alpha) \omega \land \bar{\omega}.
\]
\end{proof}
Therefore $(s,\varphi)$ is a well-defined 2-form on $X$.

\begin{defn}
We set \term{$\angles{s,\varphi}$} to be
\[
\angles{s,\varphi}:= \int_X(s,\varphi) \in \C.
\]
\end{defn}
Now note that if $W \subset X$ is any open set, and $\supp(\varphi) \subset W$, then $\angles{s,\varphi}$ is defined for all $s \in H^0(W,E^*\otimes K_X)$.

Let $(U,z)$ be a coordinate neighborhood on $X$ such that there exists a holomorphic trivialization $h_U: \pi^{-1}(U) \to U \times \C^n$, and let $h_u^*:\pi^{-1}(U) \to U \times \C^n$ be the corresponding trivialization of the dual bundle. For $x \in U,v \in E_x,v^*\in E_x^*$, and 
\[
h_u(v)=\paren{x,(u_1,\dots,u_n)}, \quad h_u^*(v^*)\paren{x,(u_1',\dots,u_n')},
\]
we have $v^*(v)= \sum _{i=1}^nu_iu_i'$. If $s \in H^0(X,E^* \otimes K_X)$ and $\varphi \in \cA_E^{0,1}(X)$ with $\supp(\varphi) \subset U$, write
\begin{align*}
    s|_U = \lambda \otimes dz, &\qquad h_u^*(\lambda(x))  = \paren{x,\lambda_1(x),\dots,\lambda_n(x)}, \\
    \varphi|_U = \alpha \otimes d \bar z, &\qquad h_u(\alpha (x)) = \paren{x, \alpha_1(x),\dots,\alpha_n(x)}.
\end{align*}
Then
\[
\angles{s, \varphi}=\int_U \paren{\sum_{k=1}^n \lambda _k(z)\alpha_k(z)}dz \land d \bar z.
\]

\begin{claim}
For any $f \in \cA_E(X)$ and $s \in H^0(X,E^*\otimes K_X)$, we have $\angles{s, \bar \partial f}=0$.
\end{claim}

\begin{proof}
Use a partition of unity to write $f = \sum_{\nu=1}^Nf_\nu,$ where each $f_\nu$ has support in a compact neighborhood $(U_\nu,z_\nu)$ on which $E|_{U_\nu}$ is trivial. We may thus assume that $N=1$, and $\supp(\varphi) \subset U$, with $(U,z)$ as above.

If $h_u(f(x))=\paren{x,(f_1(x),\dots,f_n(x))}$, then $\bar \partial f = \alpha \otimes d \bar z$ with
\[
h_u(\alpha(z)) = \paren{x,\paren{\partder[]{f_1}{\bar z},\dots,\partder[]{f_n}{\bar z}}}.
\]
Hence if $h_u^*(s(z))=\paren{z,\paren{\lambda_1(z),\dots,\lambda_n(z)}}$ on $U$, we have
\[
\angles{s,\varphi}=\int_U \sum_k \lambda _k(z) \partder[]{f_k}{\bar z}dz \land d \bar z = -\int_U \sum_k \partder[]{\lambda_k}{\bar z}f_k dz\land d\bar z=0
\]
by integrating by be parts.
\end{proof}

\begin{thm}[Weyl's Lemma (or the Regularity Theorem for $\bar \partial$).] \label{thm:weyl-lemma-regularity-thm}
Let $X$ be a compact Riemann surface and $\pi :E \to X$ be a holomorphic vector bundle on $X$. Suppose $F:\cA_E^{0,1}(X) \to \C$ is a continuous linear functional such that $F|_{\bar \partial \cA_E(X)}\equiv 0$. Then there exists a unique $s\in H^0(X,E^*\otimes K_x)$ such that
\[
F(\varphi) = \angles{s,\varphi}, \quad \forall \varphi \in \cA_E^{0,1}(X).
\]
\end{thm}

\begin{cor}[Serre Duality] \label{cor:serre-duality}
The induced pairing $\angles{,}_E:H^0(X,E^*\otimes K_X) \otimes H^1(X,E) \to \C$ is \term{perfect}, i.e it's a non-degenerate bilinear form. Hence the two vector spaces involved are dual to each other.
\end{cor}
